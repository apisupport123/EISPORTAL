var dataTable;
var currentDeleteId = null;

$(document).ready(function() {
    // Initialize DataTable WITHOUT server-side processing for export

                d.action = 'list';

            },

            { data: 'manufacturer' },
            { data: 'modelNo' },
            { data: 'ip' },
            { data: 'hostname' },
            { data: 'seatNo' },

// MANUAL EXCEL EXPORT (Fallback method)
function exportToExcelManual() {
    showLoading();
    
    try {
        // Get all data from DataTable
        var data = dataTable.rows({search: 'applied', page: 'all'}).data().toArray();
        
        if (data.length === 0) {
            hideLoading();
            showNotification('Info', 'No data to export', 'info');
            return;
        }
        
        // Create HTML table with all data
        var html = '<html xmlns:o="urn:schemas-microsoft-com:office:office" ' +
                  'xmlns:x="urn:schemas-microsoft-com:office:excel" ' +
                  'xmlns="http://www.w3.org/TR/REC-html40">' +
                  '<head><!--[if gte mso 9]><xml><x:ExcelWorkbook>' +
                  '<x:ExcelWorksheets><x:ExcelWorksheet><x:Name>Desktop Inventory</x:Name>' +
                  '<x:WorksheetOptions><x:DisplayGridlines/></x:WorksheetOptions></x:ExcelWorksheet>' +
                  '</x:ExcelWorksheets></x:ExcelWorkbook></xml><![endif]-->' +
                  '<style>td, th { mso-number-format:"General"; }</style>' +
                  '</head><body><table border="1">';
        
        // Headers
        html += '<tr>';
        html += '<th>SR NO</th>';
        html += '<th>Asset Serial No</th>';
        html += '<th>Manufacturer</th>';
        html += '<th>Model No</th>';
        html += '<th>IP</th>';
        html += '<th>Hostname</th>';
        html += '<th>Seat No</th>';
        html += '<th>Remark</th>';
        html += '</tr>';
        
        // Data rows
        for (var i = 0; i < data.length; i++) {
            var row = data[i];
            html += '<tr>';
            html += '<td>' + (i + 1) + '</td>';
            html += '<td>' + escapeForHtml(row.assetSerialNo || '') + '</td>';
            html += '<td>' + escapeForHtml(row.manufacturer || '') + '</td>';
            html += '<td>' + escapeForHtml(row.modelNo || '') + '</td>';
            html += '<td>' + escapeForHtml(row.ip || '') + '</td>';
            html += '<td>' + escapeForHtml(row.hostname || '') + '</td>';
            html += '<td>' + escapeForHtml(row.seatNo || '') + '</td>';
            html += '<td>' + escapeForHtml(row.remark || '') + '</td>';
            html += '</tr>';
        }
        
        html += '</table></body></html>';
        
        // Create blob and download
        var blob = new Blob(['\ufeff' + html], {
            type: 'application/vnd.ms-excel'
        });
        
        var url = URL.createObjectURL(blob);
        var link = document.createElement('a');
        var date = new Date();
        var filename = 'Desktop_Details_' + date.getFullYear() + '' + 
                      (date.getMonth() + 1).toString().padStart(2, '0') + '' + 
                      date.getDate().toString().padStart(2, '0') + '.xls';
        link.href = url;
        link.download = filename;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        URL.revokeObjectURL(url);
        
        hideLoading();
        showNotification('Success', 'Excel file exported successfully!', 'success');
        
    } catch (e) {
        hideLoading();
        console.error("Manual export error:", e);
        showNotification('Error', 'Failed to export: ' + e.message, 'error');
    }
}

// Add this helper function if not already present
function escapeForHtml(text) {
    if (!text) return '';
    return String(text)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
}

    if (e.which === 13 && !$(this).is('textarea')) {
        e.preventDefault();
        $('#btnSave').click();
    }
});
