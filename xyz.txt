        var currentAction = 'insert';
        
        function getValue() {
            var checkboxes = document.getElementsByName('laptop');
            var result = "";
            for (var i = 0; i < checkboxes.length; i++) {
                if (checkboxes[i].checked) {
                    if (result == "") {
                        result = checkboxes[i].value;
                    } else {
                        result += "," + checkboxes[i].value;
                    }
                }
            }
            if (result == "") {
                alert("Please select at least one record to delete");
                return;
            }
            if (confirm("Are you sure you want to delete selected record(s)?")) {
                var form = document.getElementById("myForm");
                form.action = '<%=request.getContextPath() + "/TxnContactDetail?action=delete&srno="%>' + result;
                form.submit();
            }
        }
        
        function deletesingle(srno) {
            if (confirm("Are you sure you want to delete this record?")) {
                var form = document.getElementById("myForm");
                form.action = '<%=request.getContextPath() + "/TxnContactDetail?action=delete&srno="%>' + srno;
                form.submit();
            }
        }
        
        function update(txn, lead, emailto, emailcc, primarySpoc, level2Contact, level3Contact, Project_Mgr, sr_no) {
            document.getElementById("myModal").style.display = "block";
            txn = decodeURIComponent(txn.replace(/\+/g, " "));
            lead = decodeURIComponent(lead.replace(/\+/g, " "));
            emailto = decodeURIComponent(emailto.replace(/\+/g, " "));
            emailcc = decodeURIComponent(emailcc.replace(/\+/g, " "));
            primarySpoc = decodeURIComponent(primarySpoc.replace(/\+/g, " "));
            level2Contact = decodeURIComponent(level2Contact.replace(/\+/g, " "));
            level3Contact = decodeURIComponent(level3Contact.replace(/\+/g, " "));
            Project_Mgr = decodeURIComponent(Project_Mgr.replace(/\+/g, " "));
            var form = document.getElementById("myForm");
            form.Txn_EndPoint.value = txn;
            form.Lead_Department.value = lead;
            form.Email_IDTo.value = emailto;
            form.Email_IDCC.value = emailcc;
            form.Contact_Details.value = primarySpoc;
            form.Level2_Contact.value = level2Contact;
            form.Level3_Contact.value = level3Contact;
            form.Project_Manager.value = Project_Mgr;
            form.sr_no.value = sr_no;
            currentAction = 'update';
            document.getElementById("updatebtn").style.display = "block";
            document.getElementById("submitbtn").style.display = "none";
        }
        
        function submitForm() {
            var form = document.getElementById("myForm");
            if (form.Txn_EndPoint.value.trim() == "") {
                alert("Please Enter TXN EndPoint Name");
                return false;
            }
            if (currentAction === 'update') {
                if (form.sr_no.value == "") {
                    alert("Error: No record selected for update");
                    return false;
                }
                form.action = '<%=request.getContextPath() + "/TxnContactDetail?action=update"%>';
            } else {
                form.sr_no.value = "";
                form.action = '<%=request.getContextPath() + "/TxnContactDetail?action=insert"%>';
            }
            form.submit();
            return true;
        }

        function exportToExcel() {
            var table = $('#example').DataTable();
            var data = table.rows({search: 'applied'}).data();
            if (data.length === 0) {
                alert("No data to export");
                return;
            }
            var wsData = [];
            var headers = [];
            $('#example thead tr th').each(function(index) {
                var headerText = $(this).find('b').text().trim();
                if (headerText === "") {
                    headerText = $(this).text().trim().replace('b', '').trim();
                }
                if (index > 0 && headerText !== "Actions") {
                    headers.push(headerText);
                } else if (index > 0 && headerText === "Actions") {
                    headers.push("Actions");
                }
            });
            wsData.push(headers);
            for (var i = 0; i < data.length; i++) {
                var rowData = [];
                var rowNode = table.row(i).node();
                $(rowNode).find('td').each(function(index) {
                    if (index > 0) {
                        var text = $(this).text().trim();
                        if (index === $(this).parent().find('td').length - 1) {
                            rowData.push("Update/Delete");
                        } else {
                            rowData.push(text);
                        }
                    }
                });
                wsData.push(rowData);
            }
            var ws = XLSX.utils.aoa_to_sheet(wsData);
            var wb = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(wb, ws, "TXNContactDetails");
            var filename = "TXNContact_details_" + new Date().toISOString().replace(/[\-\:\.]/g, "") + ".xlsx";
            XLSX.writeFile(wb, filename);
        }
        
        function exportToCSV() {
            var table = $('#example').DataTable();
            var data = table.rows({search: 'applied'}).data();
            if (data.length === 0) {
                alert("No data to export");
                return;
            }
            var csv = [];
            var headers = [];
            $('#example thead tr th').each(function(index) {
                var headerText = $(this).find('b').text().trim();
                if (headerText === "") headerText = $(this).text().trim().replace('b', '').trim();
                if (index > 0 && headerText !== "Actions") {
                    headers.push(headerText);
                }
            });
            csv.push(headers.join(","));
            for (var i = 0; i < data.length; i++) {
                var row = [];
                var rowNode = table.row(i).node();
                $(rowNode).find('td').each(function(index) {
                    if (index > 0 && index !== $(this).parent().find('td').length - 1) {
                        var text = $(this).text().trim();
                        if (text.indexOf(',') > -1 || text.indexOf('"') > -1) {
                            text = '"' + text.replace(/"/g, '""') + '"';
                        }
                        row.push(text);
                    }
                });
                csv.push(row.join(","));
            }
            var blob = new Blob(["\uFEFF" + csv.join("\n")], { type: "text/csv;charset=utf-8;" });
            var link = document.createElement("a");
            var url = URL.createObjectURL(blob);
            link.href = url;
            link.setAttribute("download", "TXNContact_details_" + new Date().toISOString().replace(/[\-\:\.]/g, "") + ".csv");
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            URL.revokeObjectURL(url);
        }
