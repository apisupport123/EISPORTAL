        int minute = cal.get(Calendar.MINUTE);

                } catch (Exception e) {

                            Thread.currentThread().interrupt();
                            break;
                        }
                    }
                    attempt++;
                }
            }
            if (!success) {
                System.err.println("[" + threadName + "] EXP Monitoring failed after " + MAX_RETRIES + " attempts.");
            }
        });

                        serverList.add(serverInfo);
                        hitValues.add(0);
                    }
                }

                int sumHits = 0;
                int validServerCount = 0;
                for (Integer hit : hitValues) {
                    if (hit > 0) {
                        sumHits += hit;
                        validServerCount++;
                    }
                }
                double averageHits = (validServerCount != 0) ? (double) sumHits / validServerCount : 0;
                double threshold = averageHits * 0.90;

                int criticalCount = 0;
                int warningCount = 0;
                int isolatedCount = 0;
                int healthyCount = 0;

                for (Map<String, String> server : serverList) {
                    String ip = server.get("ip");
                    int serverHit = Integer.parseInt(server.get("hits"));
                    String status = "healthy";
                    boolean isIsolated = false;
                    if (isolatedBrokers.containsKey(ip) && isolatedBrokers.get(ip).containsKey(broker)) {
                        isIsolated = isolatedBrokers.get(ip).get(broker);
                    }
                    if (isIsolated) {
                        status = "isolated";
                        isolatedCount++;
                        // Insert into impacted table (status = isolated)
                        insertImpactedStmt.setTimestamp(1, new Timestamp(startTime.getTime()));
                        insertImpactedStmt.setTimestamp(2, new Timestamp(endTime.getTime()));
                        insertImpactedStmt.setString(3, broker);
                        insertImpactedStmt.setString(4, ip);
                        insertImpactedStmt.setInt(5, serverHit);
                        insertImpactedStmt.setString(6, "isolated");
                        insertImpactedStmt.addBatch();
                        // Also add to isolated UI list
                        Map<String, String> isolatedServer = new HashMap<>();
                        isolatedServer.put("broker", broker);
                        isolatedServer.put("ip", ip);
                        isolatedServer.put("displayIP", server.get("displayIP"));
                        isolatedServer.put("hits", String.valueOf(serverHit));
                        isolatedServer.put("status", "isolated");
                        isolatedServersList.add(isolatedServer);
                    } else if (serverHit == 0) {
                        status = "zero";
                        criticalCount++;
                        // Insert into impacted table (status = critical)
                        insertImpactedStmt.setTimestamp(1, new Timestamp(startTime.getTime()));
                        insertImpactedStmt.setTimestamp(2, new Timestamp(endTime.getTime()));
                        insertImpactedStmt.setString(3, broker);
                        insertImpactedStmt.setString(4, ip);
                        insertImpactedStmt.setInt(5, serverHit);
                        insertImpactedStmt.setString(6, "critical");
                        insertImpactedStmt.addBatch();
                        // Add to impacted list
                        Map<String, String> impactedServer = new HashMap<>();
                        impactedServer.put("broker", broker);
                        impactedServer.put("ip", ip);
                        impactedServer.put("displayIP", server.get("displayIP"));
                        impactedServer.put("hits", String.valueOf(serverHit));
                        impactedServer.put("status", "critical");
                        impactedServersList.add(impactedServer);
                        // Build critical map for voice alerts
                        criticalServersByBroker.computeIfAbsent(broker, k -> new ArrayList<>()).add(server.get("displayIP"));
                    } else if (serverHit < threshold) {
                        status = "warning";
                        warningCount++;
                        // Insert into impacted table (status = warning)
                        insertImpactedStmt.setTimestamp(1, new Timestamp(startTime.getTime()));
                        insertImpactedStmt.setTimestamp(2, new Timestamp(endTime.getTime()));
                        insertImpactedStmt.setString(3, broker);
                        insertImpactedStmt.setString(4, ip);
                        insertImpactedStmt.setInt(5, serverHit);
                        insertImpactedStmt.setString(6, "warning");
                        insertImpactedStmt.addBatch();
                        // Add to impacted list
                        Map<String, String> impactedServer = new HashMap<>();
                        impactedServer.put("broker", broker);
                        impactedServer.put("ip", ip);
                        impactedServer.put("displayIP", server.get("displayIP"));
                        impactedServer.put("hits", String.valueOf(serverHit));
                        impactedServer.put("status", "warning");
                        impactedServersList.add(impactedServer);
                    } else {
                        // Healthy â€“ insert into impacted table with status 'healthy'
                        status = "healthy";
                        healthyCount++;
                        insertImpactedStmt.setTimestamp(1, new Timestamp(startTime.getTime()));
                        insertImpactedStmt.setTimestamp(2, new Timestamp(endTime.getTime()));
                        insertImpactedStmt.setString(3, broker);
                        insertImpactedStmt.setString(4, ip);
                        insertImpactedStmt.setInt(5, serverHit);
                        insertImpactedStmt.setString(6, "healthy");
                        insertImpactedStmt.addBatch();
                        // Not added to impacted/isolated UI lists
                    }
                    server.put("status", status);
                }

                // Determine broker status
                String brokerStatus = "healthy";
                if (isolatedCount > 0) {
                    brokerStatus = "isolated";
                } else if (criticalCount > 0) {
                    brokerStatus = "critical";
                } else if (warningCount > 0) {
                    brokerStatus = "warning";
                } else {
                    brokerStatus = "healthy";
                }

                // Insert broker stats
                insertBrokerStmt.setTimestamp(1, new Timestamp(startTime.getTime()));
                insertBrokerStmt.setTimestamp(2, new Timestamp(endTime.getTime()));
                insertBrokerStmt.setString(3, broker);
                insertBrokerStmt.setString(4, port);
                insertBrokerStmt.setInt(5, totalHits);
                insertBrokerStmt.setInt(6, healthyCount);
                insertBrokerStmt.setInt(7, warningCount);
                insertBrokerStmt.setInt(8, criticalCount);
                insertBrokerStmt.setInt(9, isolatedCount);
                insertBrokerStmt.setInt(10, serverList.size());
                insertBrokerStmt.setString(11, brokerStatus);
                insertBrokerStmt.setDouble(12, execDurationSec);
                insertBrokerStmt.addBatch();
            }

            // Execute batch inserts
            insertBrokerStmt.executeBatch();
            insertImpactedStmt.executeBatch();
            reportConn.commit();

            long totalDuration = System.currentTimeMillis() - methodStart;
            System.out.println("EXP Monitoring inserted " + brokerDataList.size() + " brokers and " +
                    (impactedServersList.size() + isolatedServersList.size()) + " impacted servers. Total time: " + totalDuration + " ms");

        } catch (SQLException e) {
            if (reportConn != null) {
                try { reportConn.rollback(); } catch (SQLException ex) {}
            }
            throw e;
        } catch (Exception e) {
            if (reportConn != null) {
                try { reportConn.rollback(); } catch (SQLException ex) {}
            }
            throw e;
        } finally {
            try { if (rs != null) rs.close(); } catch (SQLException e) {}
            try { if (ps != null) ps.close(); } catch (SQLException e) {}
            try { if (insertBrokerStmt != null) insertBrokerStmt.close(); } catch (SQLException e) {}
            try { if (insertImpactedStmt != null) insertImpactedStmt.close(); } catch (SQLException e) {}
            try { if (deleteBrokerStmt != null) deleteBrokerStmt.close(); } catch (SQLException e) {}
            try { if (deleteImpactedStmt != null) deleteImpactedStmt.close(); } catch (SQLException e) {}
            if (reportConn != null) {
                try { reportConn.setAutoCommit(true); reportConn.close(); } catch (SQLException e) {}
            }
            if (iibConn != null) {
                try { iibConn.setAutoCommit(true); iibConn.close(); } catch (SQLException e) {}
            }
        }
    }

    private static void fetchBrokerIPMappings(Connection con, String currentEnvironment) throws SQLException {
        String query = "SELECT DISTINCT a.EXP_IP, a.BROKER_NAME FROM EXP_IP_BROKER_DETAILS a ORDER BY a.BROKER_NAME, a.EXP_IP";
        try (Statement stmt = con.createStatement();
             ResultSet rs = stmt.executeQuery(query)) {
            while (rs.next()) {
                String brokerName = rs.getString("BROKER_NAME");
                String expIP = rs.getString("EXP_IP");
                if (brokerName == null || expIP == null) continue;
                brokerName = brokerName.trim();
                expIP = expIP.trim();
                boolean shouldInclude = false;
                if ("PR".equals(currentEnvironment)) {
                    shouldInclude = expIP.startsWith("10.188.24.") || expIP.startsWith("10.188.25.");
                } else if ("DR".equals(currentEnvironment)) {
                    shouldInclude = expIP.startsWith("10.177.40.") || expIP.startsWith("10.177.41.");
                }
                if (shouldInclude) {
                    brokerIPMapping.computeIfAbsent(brokerName, k -> new ArrayList<>()).add(expIP);
                }
            }
            for (List<String> ips : brokerIPMapping.values()) {
                Collections.sort(ips, (ip1, ip2) -> {
                    try {
                        String[] parts1 = ip1.split("\\.");
                        String[] parts2 = ip2.split("\\.");
                        int lastOctet1 = Integer.parseInt(parts1[3]);
                        int lastOctet2 = Integer.parseInt(parts2[3]);
                        return Integer.compare(lastOctet1, lastOctet2);
                    } catch (Exception e) {
                        return ip1.compareTo(ip2);
                    }
                });
            }
        }
    }

    private static JSONObject parseJsonWithDuplicates(String jsonString) {
        JSONObject result = new JSONObject();
        try {
            result = new JSONObject(jsonString);
            return result;
        } catch (Exception e) {
            // Custom parser â€“ same as before
            try {
                String content = jsonString.trim();
                if (content.startsWith("{")) {
                    content = content.substring(1, content.length() - 1);
                }
                List<String> pairs = new ArrayList<>();
                int braceLevel = 0, bracketLevel = 0;
                StringBuilder currentPair = new StringBuilder();
                for (int i = 0; i < content.length(); i++) {
                    char c = content.charAt(i);
                    currentPair.append(c);
                    if (c == '{') braceLevel++;
                    if (c == '}') braceLevel--;
                    if (c == '[') bracketLevel++;
                    if (c == ']') bracketLevel--;
                    if (c == ',' && braceLevel == 0 && bracketLevel == 0) {
                        String pair = currentPair.substring(0, currentPair.length() - 1);
                        pairs.add(pair);
                        currentPair = new StringBuilder();
                    }
                }
                if (currentPair.length() > 0) pairs.add(currentPair.toString());
                for (String pair : pairs) {
                    int colonIndex = pair.indexOf(':');
                    if (colonIndex > 0) {
                        String key = pair.substring(0, colonIndex).trim();
                        String value = pair.substring(colonIndex + 1).trim();
                        if (key.startsWith("\"") && key.endsWith("\"")) {
                            key = key.substring(1, key.length() - 1);
                        }
                        if (value.startsWith("{")) {
                            result.put(key, parseJsonWithDuplicates(value));
                        } else if (value.startsWith("[")) {
                            result.put(key, new JSONArray(value));
                        } else if (value.startsWith("\"")) {
                            result.put(key, value.substring(1, value.length() - 1));
                        } else if (value.equals("true") || value.equals("false")) {
                            result.put(key, Boolean.parseBoolean(value));
                        } else if (value.equals("null")) {
                            result.put(key, JSONObject.NULL);
                        } else {
                            try {
                                if (value.contains(".")) {
                                    result.put(key, Double.parseDouble(value));
                                } else {
                                    result.put(key, Long.parseLong(value));
                                }
                            } catch (NumberFormatException nfe) {
                                result.put(key, value);
                            }
                        }
                    }
                }
            } catch (Exception ex) {
                System.err.println("Custom JSON parsing failed: " + ex.getMessage());
            }
        }
        return result;
    }

    private static void detectIsolatedBrokers(JSONObject urlData) {
        isolatedBrokers.clear();
        if (urlData == null || urlData.length() == 0) return;
        Iterator<String> serverKeys = urlData.keys();
        while (serverKeys.hasNext()) {
            String serverKey = serverKeys.next();
            Object serverObj = urlData.opt(serverKey);
            if (serverObj instanceof JSONObject) {
                JSONObject serverData = (JSONObject) serverObj;
                String ip = serverData.optString("ip", "");
                if (ip.isEmpty()) continue;
                Object brokerObj = serverData.opt("BROKER");
                if (brokerObj instanceof JSONObject) {
                    JSONObject brokerData = (JSONObject) brokerObj;
                    if (!isolatedBrokers.containsKey(ip)) {
                        isolatedBrokers.put(ip, new HashMap<>());
                    }
                    Iterator<String> brokerKeys = brokerData.keys();
                    while (brokerKeys.hasNext()) {
                        String brokerName = brokerKeys.next();
                        Object brokerArrayObj = brokerData.opt(brokerName);
                        if (brokerArrayObj instanceof JSONArray) {
                            JSONArray brokerArray = (JSONArray) brokerArrayObj;
                            if (brokerArray.length() >= 2) {
                                int isolatedFlag = brokerArray.optInt(1, -1);
                                boolean isIsolated = (isolatedFlag == 0);
                                isolatedBrokers.get(ip).put(brokerName, isIsolated);
                            }
                        }
                    }
                }
            }
        }
    }

    private static String generateCaseStatements() {
        StringBuilder brokerCase = new StringBuilder("(\n CASE\n");
        StringBuilder portCase = new StringBuilder("(\n CASE\n");

        for (Map.Entry<String, String> entry : brokerMapping.entrySet()) {
            String brokerNames = entry.getKey();
            String expPorts = entry.getValue();
            String mappedPort = portMapping.get(entry.getKey());

            List<String> brokerList = parseCommaSeparatedValues(brokerNames);
            List<String> portList = parseCommaSeparatedValues(expPorts);

            if (brokerNames.contains("AADHAR")) {
                brokerCase.append(" WHEN TRIM(BROKER_NAME) IN (").append(formatList(brokerList))
                        .append(") OR TRIM(EXP_PORT) IN (").append(formatList(portList))
                        .append(") THEN 'AADHAR_EXP'\n");
                portCase.append(" WHEN TRIM(BROKER_NAME) IN (").append(formatList(brokerList))
                        .append(") OR TRIM(EXP_PORT) IN (").append(formatList(portList)).append(") THEN '")
                        .append(mappedPort).append("'\n");
            } else {
                brokerCase.append(" WHEN TRIM(BROKER_NAME) = '").append(brokerNames).append("' OR TRIM(EXP_PORT) = '")
                        .append(expPorts).append("' THEN '").append(brokerNames).append("'\n");
                portCase.append(" WHEN TRIM(BROKER_NAME) = '").append(brokerNames).append("' OR TRIM(EXP_PORT) = '")
                        .append(expPorts).append("' THEN '").append(mappedPort).append("'\n");
            }
        }

        brokerCase.append(" ELSE 'UNKNOWN_BROKER'\n END\n) AS broker,\n");
        portCase.append(" ELSE 'UNKNOWN_PORT'\n END\n) AS port");
        return brokerCase.toString() + portCase.toString();
    }

    private static String generateDynamicOrderByClause() {
        StringBuilder orderByClause = new StringBuilder("CASE\n");
        for (Map.Entry<String, Integer> entry : brokerOrderMapping.entrySet()) {
            String brokerName = entry.getKey();
            int orderNumber = entry.getValue();
            if (brokerName == null) continue;
            if (brokerName.contains("AADHAR")) orderNumber = 10;
            String[] brokerNames = brokerName.split(",");
            for (String broker : brokerNames) {
                String trimmedBroker = broker.trim();
                if (!trimmedBroker.isEmpty()) {
                    String aliasName = broker.contains("AADHAR") ? "AADHAR_EXP" : trimmedBroker;
                    orderByClause.append(" WHEN TRIM(broker) = '")
                            .append(aliasName)
                            .append("' THEN ")
                            .append(orderNumber)
                            .append("\n");
                }
            }
        }
        orderByClause.append(" ELSE 999\n");
        orderByClause.append("END");
        return orderByClause.toString();
    }

    private static List<String> parseCommaSeparatedValues(String input) {
        if (input == null) return new ArrayList<>();
        return Arrays.stream(input.split(","))
                .map(String::trim)
                .filter(s -> !s.isEmpty())
                .collect(Collectors.toList());
    }

    private static String formatList(List<String> values) {
        if (values.isEmpty()) return "";
        return String.join(", ", values.stream().map(val -> "'" + val + "'").toArray(String[]::new));
    }
