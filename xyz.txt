class TimeRangeHolder {
    String minTime = "";
    String maxTime = "";
    boolean loaded = false;
}

final ExpDataHolder expHolder = new ExpDataHolder();
final SysDataHolder sysHolder = new SysDataHolder();
final TimeRangeHolder timeRangeHolder = new TimeRangeHolder();

// Create thread to fetch only MAX insertion time from database
Thread timeRangeThread = new Thread(new Runnable() {
    public void run() {
        try {
            Connection conn = null;
            try {
                conn = DBUtilityAuto.getReportDBConnection();
                // Query to get max insertion time from both tables
                String timeSql = "SELECT MAX(INSERTION_TIME) as max_time FROM ("
                    + "SELECT INSERTION_TIME FROM a_exp_service_wise_td WHERE INSERTION_TIME >= SYSTIMESTAMP - INTERVAL '10' MINUTE "
                    + "UNION ALL "
                    + "SELECT INSERTION_TIME FROM a_sys_service_wise_td WHERE INSERTION_TIME >= SYSTIMESTAMP - INTERVAL '10' MINUTE"
                    + ")";
                
                System.out.println("Time Range Query: " + timeSql);
                Statement statement = conn.createStatement();
                ResultSet resultSet = statement.executeQuery(timeSql);
                
                if (resultSet.next()) {
                    Timestamp maxTime = resultSet.getTimestamp("max_time");
                    
                    if (maxTime != null) {
                        // Format max time
                        SimpleDateFormat sdf = new SimpleDateFormat("HH:mm:ss");
                        timeRangeHolder.maxTime = sdf.format(maxTime);
                        
                        // Subtract 10 minutes from max time to get min time
                        Calendar cal = Calendar.getInstance();
                        cal.setTime(maxTime);
                        cal.add(Calendar.MINUTE, -10);
                        timeRangeHolder.minTime = sdf.format(cal.getTime());
                    } else {
                        // Fallback to current time
                        ZonedDateTime now = ZonedDateTime.now(ZoneId.of("Asia/Kolkata"));
                        ZonedDateTime tenMinsAgo = now.minusMinutes(10);
                        DateTimeFormatter formatter = DateTimeFormatter.ofPattern("HH:mm:ss");
                        timeRangeHolder.minTime = tenMinsAgo.format(formatter);
                        timeRangeHolder.maxTime = now.format(formatter);
                    }
                }
                resultSet.close();
                statement.close();
            } catch (Exception e) {
                e.printStackTrace();
                // Fallback to current time range
                ZonedDateTime now = ZonedDateTime.now(ZoneId.of("Asia/Kolkata"));
                ZonedDateTime tenMinsAgo = now.minusMinutes(10);
                DateTimeFormatter formatter = DateTimeFormatter.ofPattern("HH:mm:ss");
                timeRangeHolder.minTime = tenMinsAgo.format(formatter);
                timeRangeHolder.maxTime = now.format(formatter);
            } finally {
                if (conn != null) try { conn.close(); } catch (Exception e) {}
            }
            timeRangeHolder.loaded = true;
        } catch(Exception e) {
            e.printStackTrace();
            timeRangeHolder.loaded = true;
        }
    }
});

// Time range message - using database max insertion time
document.addEventListener('DOMContentLoaded', function() {
    // Get times from server-side
    var minTime = '<%= request.getAttribute("minTime") != null ? request.getAttribute("minTime") : "" %>';
    var maxTime = '<%= request.getAttribute("maxTime") != null ? request.getAttribute("maxTime") : "" %>';
    
    // If we have valid times from database, use them
    if (minTime && maxTime && minTime !== "" && maxTime !== "") {
        var message = 'Session data from ' + minTime + ' to ' + maxTime + ' (DB Insertion Time)';
        document.getElementById('timeRangeMessage').innerText = message;
    } else {
        // Fallback to current time range
        var currentTime = new Date();
        var tenMinutesAgo = new Date();
        tenMinutesAgo.setMinutes(currentTime.getMinutes() - 10);
        var formatTime = function(date) {
            var hours = date.getHours();
            var minutes = date.getMinutes();
            hours = hours < 10 ? '0' + hours : hours;
            minutes = minutes < 10 ? '0' + minutes : minutes;
            return hours + ':' + minutes;
        };
        var message = 'Session data from ' + formatTime(tenMinutesAgo) + ' to ' + formatTime(currentTime);
        document.getElementById('timeRangeMessage').innerText = message;
    }
});
