<script>
    // ============================================================
    // GLOBALS
    // ============================================================
    var testResults = {};
    var totalTests = 0;
    var passCount = 0;
    var failCount = 0;
    var pendingCount = 0;
    var isRunning = false;

    // ============================================================
    // TOGGLE SECTION
    // ============================================================
    function toggleSection(header) {
        var body = header.nextElementSibling;
        var icon = header.querySelector('.toggle-icon');
        if (body.classList.contains('hidden')) {
            body.classList.remove('hidden');
            icon.className = 'fas fa-chevron-up toggle-icon';
        } else {
            body.classList.add('hidden');
            icon.className = 'fas fa-chevron-down toggle-icon';
        }
    }

    // ============================================================
    // ADD LOG
    // ============================================================
    function addLog(message, type) {
        var container = document.getElementById('logContainer');
        var time = new Date().toLocaleTimeString();
        var colorClass = type || 'info';
        var entry = document.createElement('div');
        entry.className = 'log-entry';
        entry.innerHTML = '<span class="time">[' + time + ']</span> <span class="' + colorClass + '">' + message + '</span>';
        container.appendChild(entry);
        container.scrollTop = container.scrollHeight;
    }

    // ============================================================
    // UPDATE STATS
    // ============================================================
    function updateStats() {
        document.getElementById('totalTests').textContent = totalTests;
        document.getElementById('passCount').textContent = passCount;
        document.getElementById('failCount').textContent = failCount;
        document.getElementById('pendingCount').textContent = pendingCount;
    }

    // ============================================================
    // UPDATE TEST STATUS
    // ============================================================
    function updateTestStatus(testId, status, message) {
        var item = document.getElementById('test-' + testId);
        if (!item) return;
        
        var icon = item.querySelector('.icon');
        var badge = item.querySelector('.status-badge');
        var resultDiv = document.getElementById('result-' + testId);
        
        // Update icon
        icon.className = 'icon ' + status;
        if (status === 'pass') {
            icon.innerHTML = '<i class="fas fa-check"></i>';
            badge.className = 'status-badge pass';
            badge.textContent = '✅ Passed';
        } else if (status === 'fail') {
            icon.innerHTML = '<i class="fas fa-times"></i>';
            badge.className = 'status-badge fail';
            badge.textContent = '❌ Failed';
        } else if (status === 'running') {
            icon.innerHTML = '<i class="fas fa-spinner"></i>';
            badge.className = 'status-badge running';
            badge.textContent = '⏳ Running...';
        } else {
            icon.innerHTML = '<i class="fas fa-clock"></i>';
            icon.className = 'icon pending';
            badge.className = 'status-badge pending';
            badge.textContent = '⏳ Pending';
        }
        
        // Show result
        if (message) {
            resultDiv.textContent = message;
            resultDiv.className = 'result-detail show';

        if (isRunning) {

            return;

        var type = parts[0];

        updateTestStatus(fullId, 'running');
        addLog('▶️ Running test: ' + fullId, 'info');

        switch(fullId) {
            case 'ui-1': testUILoad(fullId); break;

            case 'db-3': testSavedData(fullId); break;

    // ============================================================
    // RUN ALL TESTS
    // ============================================================
    function runAllTests() {
        if (isRunning) {
            alert('⚠️ Tests already running!');
            return;
        }
        isRunning = true;
        addLog('🚀 Starting all tests...', 'info');
        
        // Reset stats
        totalTests = 0;
        passCount = 0;
        failCount = 0;
        pendingCount = 0;
        
        // Get all test items
        var testItems = document.querySelectorAll('.test-item');
        var testIds = [];
        testItems.forEach(function(item) {
            var id = item.id.replace('test-', '');
            testIds.push(id);
        });
        
        totalTests = testIds.length;
        pendingCount = totalTests;
        updateStats();
        
        // Run each test with delay
        testIds.forEach(function(id, index) {
            setTimeout(function() {
                runTest(id);
            }, index * 1500);
        });
        
        // Update status after all tests
        setTimeout(function() {
            isRunning = false;
            addLog('✅ All tests completed!', 'pass');
            updateStats();
        }, testIds.length * 1500 + 2000);
    }
