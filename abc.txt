String getPartition() {
    Calendar cal = Calendar.getInstance();
    cal.add(Calendar.MONTH, -1); // Previous month
    String month = new SimpleDateFormat("MMM", Locale.US).format(cal.getTime()).toUpperCase();
    String year = new SimpleDateFormat("yy", Locale.US).format(cal.getTime());
    return month + "_" + year;
}

    function getPartition() {
        var now = new Date();
        now.setMonth(now.getMonth() - 1);
        var months = ['JAN','FEB','MAR','APR','MAY','JUN','JUL','AUG','SEP','OCT','NOV','DEC'];
        var month = months[now.getMonth()];
        var year = String(now.getFullYear()).slice(-2);
        return month + '_' + year;
    }

    function formatTime(seconds) {
        var mins = Math.floor(seconds / 60);
        var secs = seconds % 60;
        return String(mins).padStart(2, '0') + ':' + String(secs).padStart(2, '0');
    }

    function updateTimer() {
        timerSeconds--;
        if (timerSeconds <= 0) timerSeconds = 600;
        document.getElementById('timerDisplay').textContent = formatTime(timerSeconds);
    }

    function addActivity(sourceId, status) {
        var now = new Date();
        var timeStr = now.getHours().toString().padStart(2, '0') + ':' + 
                      now.getMinutes().toString().padStart(2, '0') + ':' + 
                      now.getSeconds().toString().padStart(2, '0');
        
        activityLog.unshift({ time: timeStr, source: sourceId, status: status });
        if (activityLog.length > MAX_ACTIVITY) activityLog.pop();
        renderActivity();
    }

    function renderActivity() {
        var container = document.getElementById('activityList');
        if (activityLog.length === 0) {
            container.innerHTML = `<div class="activity-empty"><i class="fas fa-inbox"></i><span>Waiting for activity...</span></div>`;
            return;
        }
        var html = '';
        for (var i = 0; i < Math.min(activityLog.length, 25); i++) {
            var item = activityLog[i];
            var cls = item.status === 'Y' ? 'on' : 'off';
            var txt = item.status === 'Y' ? 'ON' : 'OFF';
            html += `<div class="activity-item ${cls}">
                <span class="activity-time">${item.time}</span>
                <span class="activity-source">${item.source}</span>
                <span class="activity-status ${cls}">${txt}</span>
            </div>`;
        }
        container.innerHTML = html;
    }

    function hideLoading() {
        var overlay = document.getElementById('loadingOverlay');
        overlay.classList.add('hide');
        document.getElementById('liveBadge').style.display = 'flex';
        document.getElementById('timerBadge').style.display = 'flex';
        document.getElementById('statusBadge').style.display = 'flex';
        document.getElementById('statsRow').style.display = 'grid';
        document.getElementById('mainGrid').style.display = 'grid';
        
        // Set partition
        partitionName = getPartition();
        document.getElementById('partitionBadge').textContent = '📁 ' + partitionName;
        
        timerInterval = setInterval(updateTimer, 1000);
    }

    function fetchData() {
        fetch('?action=fetchSourceStatus')
            .then(function(response) { return response.json(); })
            .then(function(data) {
                renderSources(data);
                updateStats(data);
                updateActivityLog(data);
                document.getElementById('lastUpdate').textContent = 'Updated: ' + new Date().toLocaleTimeString();
                
                if (isFirstLoad) {
                    isFirstLoad = false;
                    hideLoading();
                }
            })
            .catch(function(error) {
                console.error('Error:', error);
                if (isFirstLoad) {
                    isFirstLoad = false;
                    hideLoading();
                    document.getElementById('sourcesGrid').innerHTML = `
                        <div style="text-align:center;padding:40px;color:var(--text3);grid-column:1/-1;">
                            <i class="fas fa-exclamation-triangle" style="font-size:2rem;display:block;margin-bottom:10px;"></i>
                            <p style="font-weight:600;">Failed to load data</p>
                            <p style="font-size:0.8rem;margin-top:4px;">Please refresh the page</p>
                        </div>
                    `;
                }
            });
    }

    function renderSources(data) {
        var container = document.getElementById('sourcesGrid');
        if (!data || data.length === 0) {
            container.innerHTML = `<div style="text-align:center;padding:40px;color:var(--text3);grid-column:1/-1;">
                <i class="fas fa-database" style="font-size:2rem;display:block;margin-bottom:10px;"></i>
                <p>No data available</p>
            </div>`;
            return;
        }

        var html = '';
        var activeCount = 0;
        var sourceIcons = {
            'SA': '🏦', 'FS': '📊', 'I4': '🔄', 'Y2': '📱', 'KA': '🏛️',
            'FA': '📈', 'WB': '🏧', 'ON': '🌐', 'CC': '💳', 'CK': '🔑',
            'CR': '📋', 'SQ': '📝', 'RL': '📌', 'UP': '⬆️', 'CA': '📁',
            'AK': '📂', 'NB': '🏦', 'IN': '📊', 'MB': '📱', 'SK': '🔒'
        };

        for (var i = 0; i < data.length; i++) {
            var item = data[i];
            var isActive = item.flag === 'Y';
            if (isActive) activeCount++;
            var cls = isActive ? 'active' : 'inactive';
            var icon = sourceIcons[item.source_id] || '📡';
            var status = isActive ? 'ONLINE' : 'OFFLINE';
            
            html += `<div class="source-card ${cls}" data-source="${item.source_id}">
                <span class="status-indicator"></span>
                <span class="source-icon">${icon}</span>
                <div class="source-id">${item.source_id}</div>
                <div class="source-status">${status}</div>
            </div>`;
        }
        container.innerHTML = html;
        document.getElementById('sourceCount').textContent = activeCount + '/' + data.length;
    }

    function updateStats(data) {
        var active = 0, inactive = 0;
        for (var i = 0; i < data.length; i++) {
            if (data[i].flag === 'Y') active++; else inactive++;
        }
        document.getElementById('activeCount').textContent = active;
        document.getElementById('inactiveCount').textContent = inactive;
        document.getElementById('totalSources').textContent = data.length;
        var uptime = data.length > 0 ? Math.round((active / data.length) * 100) : 0;
        document.getElementById('uptimePercent').textContent = uptime + '%';
    }

    function updateActivityLog(data) {
        var currentActive = {};
        for (var i = 0; i < data.length; i++) {
            if (data[i].flag === 'Y') currentActive[data[i].source_id] = true;
        }
        
        for (var sid in currentActive) {
            if (!sourceData[sid]) {
                addActivity(sid, 'Y');
            } else if (sourceData[sid] !== currentActive[sid]) {
                addActivity(sid, currentActive[sid] ? 'Y' : 'N');
            }
        }
        for (var old in sourceData) {
            if (sourceData[old] && !currentActive[old]) {
                addActivity(old, 'N');
            }
        }
        sourceData = currentActive;
    }

    function startDashboard() {
        document.getElementById('timerDisplay').textContent = formatTime(timerSeconds);
        fetchData();
        refreshInterval = setInterval(fetchData, 10000);
    }

    window.addEventListener('beforeunload', function() {
        if (refreshInterval) clearInterval(refreshInterval);
        if (timerInterval) clearInterval(timerInterval);
    });

    startDashboard();
</script>
