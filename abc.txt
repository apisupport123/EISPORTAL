<script>
    // ===== Premium Modal State =====
    var pmApi = '', pmCat = '', pmSource = '', pmError = '', pmHits = '', pmTD = '', pmTime = '';
    
    function openPremiumModal(msg, api, cat, src, err, hits, td, time) {
        pmApi = api || '';
        pmCat = cat || 'GEN-5';
        pmSource = src || '';
        pmError = err || '';
        pmHits = hits || '';
        pmTD = td || '';
        pmTime = time || '';
        
        document.getElementById('modalMsg').innerHTML = msg;
        document.getElementById('modalContainer').style.display = 'flex';
        document.getElementById('TextArea').value = '';
        document.getElementById('closeName').value = '';
    }
    
    function closePremiumModal() {
        document.getElementById('modalContainer').style.display = 'none';
    }
    
    // ===== Close Handler =====
    document.addEventListener('DOMContentLoaded', function() {
        document.getElementById('closeBtn').addEventListener('click', function() {
            var comment = document.getElementById('TextArea').value;
            var closer = document.getElementById('closeName').value;
            var pTime = document.getElementById('pasuseTime').value;
            
            if(comment.trim() == '' || closer == '') {
                alert('⚠️ Please fill Comment and Closed By');
                return;
            }
            
            var url = '<%=request.getContextPath() + "/jsp/ServiceWiseFailureAction.jsp?Api_Name="%>' 
                + encodeURIComponent(pmApi)
                + '&SourceID=' + encodeURIComponent(pmSource)
                + '&TextArea=' + encodeURIComponent(comment)
                + '&pasuseTime=' + encodeURIComponent(pTime)
                + '&ErrorCode=' + encodeURIComponent(pmError)
                + '&closeName=' + encodeURIComponent(closer)
                + '&ServiceCategory=' + encodeURIComponent(pmCat);
            
            closePremiumModal();
            location.replace(url);
        });
        
        document.getElementById('detailsBtn').addEventListener('click', function() {
            if(pmCat != 'SI' && pmCat != 'SIServices') {
                var url = '<%=request.getContextPath() + "/showFailureServicesWise.do?action=showFailureServicesWise&url="%>' 
                    + encodeURIComponent(pmApi) 
                    + '&ServiceCategory=' + encodeURIComponent(pmCat);
                window.open(url, '_blank');
            }
        });
        
        document.getElementById('openPopupButton').addEventListener('click', function() {
            window.open('<%=request.getContextPath() + "/jsp/DisplayHundredPercentFailure.jsp"%>', '_blank');
        });
        
        document.getElementById('openPopup').addEventListener('click', function() {
            window.open('DisplayTodayServiceWiseFailure.jsp', 'DataPopup', 'width=800,height=600');
        });
        
        // Update stats
        updateStats();
    });
    
    function updateStats() {
        var tiles = document.querySelectorAll('.metric-tile:not(.tile-empty)');
        var total = tiles.length;
        var critical = document.querySelectorAll('.tile-critical').length;
        var healthy = document.querySelectorAll('.tile-healthy').length;
        
        document.getElementById('totalServices').textContent = total;
        document.getElementById('totalCritical').textContent = critical;
        document.getElementById('totalHealthy').textContent = healthy;
        
        var now = new Date();
        document.getElementById('lastUpdate').textContent = now.toLocaleTimeString();
    }
    
    // ===== Timer =====
    function countdownPremium(el, minutes, seconds) {
        var element = document.getElementById(el);
        if(!element) return;
        var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
        function update() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var t = new Date(msLeft);
                var m = t.getUTCMinutes();
                var s = t.getUTCSeconds();
                element.innerHTML = (m ? m + ':' : '') + (m ? String(s).padStart(2,'0') : s);
                setTimeout(update, 100);
            }
        }
        update();
    }
    countdownPremium('ten-countdown', 5, 0);
    
    // ===== Click outside =====
    document.getElementById('modalContainer').addEventListener('click', function(e) {
        if(e.target === this) closePremiumModal();
    });
    
    document.addEventListener('keydown', function(e) {
        if(e.key === 'Escape') closePremiumModal();
    });
    
    // ===== Particles =====
    (function createParticles() {
        var container = document.getElementById('particles');
        for(var i = 0; i < 30; i++) {
            var p = document.createElement('div');
            p.className = 'particle';
            p.style.left = Math.random() * 100 + '%';
            p.style.width = (Math.random() * 2 + 1) + 'px';
            p.style.height = p.style.width;
            p.style.animationDuration = (Math.random() * 20 + 15) + 's';
            p.style.animationDelay = (Math.random() * 20) + 's';
            p.style.opacity = Math.random() * 0.3 + 0.1;
            container.appendChild(p);
        }
    })();
