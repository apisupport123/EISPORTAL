// ===== SLOT DATA =====
const slots = [
    { name: '10PM-11PM', endTime: '11PM', freezeMinute: 45 },
    { name: '11PM-12AM', endTime: '12AM', freezeMinute: 45 },
    { name: '12AM-1AM', endTime: '1AM', freezeMinute: 45 },
    { name: '1AM-2AM', endTime: '2AM', freezeMinute: 45 },
    { name: '2AM-3AM', endTime: '3AM', freezeMinute: 45 },
    { name: '3AM-4AM', endTime: '4AM', freezeMinute: 45 },
    { name: '4AM-5AM', endTime: '5AM', freezeMinute: 45 },
    { name: '5AM-6AM', endTime: '6AM', freezeMinute: 45 },
    { name: '6AM-7AM', endTime: '7AM', freezeMinute: 45 }
];

// ===== COMPUTE FREEZE TIME =====
function computeFreezeTime(slot, date) {
    let endStr = slot.endTime;
    let hour = 0, minute = 0;
    let isPM = endStr.includes('PM');
    let isAM = endStr.includes('AM');
    let timeStr = endStr.replace(/[AP]M/g, '').trim();
    
    if (timeStr === '12') {
        hour = isPM ? 12 : 0;
    } else {
        hour = parseInt(timeStr);
        if (isPM && hour !== 12) hour += 12;
        if (isAM && hour === 12) hour = 0;
    }
    
    let freezeTime = new Date(date);
    
    // ===== FIX: Only 12AM gets +1 day =====
    if (hour === 0 && minute === 0) {
        freezeTime.setDate(freezeTime.getDate() + 1);
    }
    
    freezeTime.setHours(hour, minute, 0, 0);
    freezeTime.setMinutes(freezeTime.getMinutes() + slot.freezeMinute);
    
    return freezeTime;
}

// ===== TEST FUNCTION =====
function testTime(timeStr) {
    let currentTime = new Date();
    
    // Parse time string (e.g., "11:30 PM")
    let parts = timeStr.split(' ');
    let timeParts = parts[0].split(':');
    let hour = parseInt(timeParts[0]);
    let minute = parseInt(timeParts[1]);
    let ampm = parts[1];
    
    if (ampm === 'PM' && hour !== 12) hour += 12;
    if (ampm === 'AM' && hour === 12) hour = 0;
    
    currentTime.setHours(hour, minute, 0, 0);
    
    runTest(currentTime, timeStr);
}

function testCurrentTime() {
    let currentTime = new Date();
    let timeStr = currentTime.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' });
    runTest(currentTime, timeStr + ' (Current)');
}

function runTest(currentTime, label) {
    // Update time display
    document.getElementById('timeDisplay').innerHTML = 
        '⏰ Test Time: <span style="color:#1a237e;">' + label + '</span>';
    
    let results = [];
    let frozenCount = 0;
    let total = slots.length;
    
    // Check each slot
    slots.forEach(function(slot) {
        let freezeTime = computeFreezeTime(slot, new Date());
        let isFrozen = currentTime > freezeTime;
        if (isFrozen) frozenCount++;
        
        results.push({
            name: slot.name,
            freezeTime: freezeTime,
            isFrozen: isFrozen,
            freezeTimeStr: freezeTime.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })
        });
    });
    
    // Render results
    let html = '';
    results.forEach(function(r) {
        let cls = r.isFrozen ? 'frozen' : 'editable';
        let icon = r.isFrozen ? '🔒' : '✏️';
        let badge = r.isFrozen ? 
            '<span class="badge badge-freeze">FROZEN</span>' : 
            '<span class="badge badge-edit">EDITABLE</span>';
        html += '<div class="slot ' + cls + '">' +
            '<span><strong>' + r.name + '</strong> → Freezes at ' + r.freezeTimeStr + '</span>' +
            '<span>' + icon + ' ' + badge + '</span>' +
        '</div>';
    });
    document.getElementById('slotResults').innerHTML = html;
    
    // Update status box
    let statusClass = '';
    let statusText = '';
    if (frozenCount === 0) {
        statusClass = 'status-all-edit';
        statusText = '✏️ All slots are EDITABLE (' + total + '/' + total + ')';
    } else if (frozenCount === total) {
        statusClass = 'status-all-freeze';
        statusText = '🔒 All slots are FROZEN (' + frozenCount + '/' + total + ')';
    } else {
        statusClass = 'status-partial';
        statusText = '⚠️ ' + frozenCount + ' slot(s) FROZEN, ' + (total - frozenCount) + ' EDITABLE';
    }
    document.getElementById('statusBox').className = 'status-box ' + statusClass;
    document.getElementById('statusBox').textContent = statusText;
    
    // Update summary
    let summary = '';
    summary += '📊 <strong>' + frozenCount + '</strong> slots frozen out of <strong>' + total + '</strong>\n';
    summary += '📈 ' + Math.round((frozenCount/total)*100) + '% complete\n';
    if (frozenCount === 0) summary += '✅ All slots are open for editing';
    else if (frozenCount === total) summary += '🔒 All slots are frozen';
    else summary += '⏳ ' + (total - frozenCount) + ' slots still editable';
    document.getElementById('summaryBox').innerHTML = summary;
    
    // Update progress bar
    let percentage = Math.round((frozenCount/total)*100);
    document.getElementById('progressFill').style.width = percentage + '%';
    document.getElementById('progressText').textContent = percentage + '% Frozen';
    
    // Update progress bar color
    let fill = document.getElementById('progressFill');
    if (percentage < 33) fill.style.background = 'linear-gradient(90deg, #28a745, #28a745)';
    else if (percentage < 66) fill.style.background = 'linear-gradient(90deg, #ffc107, #ffc107)';
    else fill.style.background = 'linear-gradient(90deg, #dc3545, #dc3545)';
    
    // Console log for debugging
    console.log('=== Test: ' + label + ' ===');
    results.forEach(function(r) {
        console.log(r.name + ' → ' + (r.isFrozen ? '❌ FROZEN' : '✅ EDITABLE') + ' (freezes at ' + r.freezeTimeStr + ')');
    });
    console.log('Summary: ' + frozenCount + '/' + total + ' frozen');
}

// ===== AUTO RUN ON LOAD =====
window.onload = function() {
    testCurrentTime();
    console.log('✅ Freeze Logic Tester Loaded!');
    console.log('📌 Click test buttons to verify sequential freeze logic');
};
