<script>
    // ===== Dynatrace Modal State =====
    var dtApi = '', dtCat = '', dtSource = '', dtError = '', dtHits = '', dtTD = '', dtTime = '';
    
    function openDynatraceModal(msg, api, cat, src, err, hits, td, time) {
        dtApi = api || '';
        dtCat = cat || 'GEN-5';
        dtSource = src || '';
        dtError = err || '';
        dtHits = hits || '';
        dtTD = td || '';
        dtTime = time || '';
        
        document.getElementById('modalMsg').innerHTML = msg;
        document.getElementById('modalContainer').style.display = 'flex';
        document.getElementById('TextArea').value = '';
        document.getElementById('closeName').value = '';
    }
    
    function closeDynatraceModal() {
        document.getElementById('modalContainer').style.display = 'none';
    }
    
    // ===== Close Handler =====
    document.addEventListener('DOMContentLoaded', function() {
        document.getElementById('closeBtn').addEventListener('click', function() {
            var comment = document.getElementById('TextArea').value;
            var closer = document.getElementById('closeName').value;
            var pTime = document.getElementById('pasuseTime').value;
            
            if(comment.trim() == '' || closer == '') {
                alert('⚠️ Please fill Comment and Closed By');
                return;
            }
            
            var url = '<%=request.getContextPath() + "/jsp/ServiceWiseFailureAction.jsp?Api_Name="%>' 
                + encodeURIComponent(dtApi)
                + '&SourceID=' + encodeURIComponent(dtSource)
                + '&TextArea=' + encodeURIComponent(comment)
                + '&pasuseTime=' + encodeURIComponent(pTime)
                + '&ErrorCode=' + encodeURIComponent(dtError)
                + '&closeName=' + encodeURIComponent(closer)
                + '&ServiceCategory=' + encodeURIComponent(dtCat);
            
            closeDynatraceModal();
            location.replace(url);
        });
        
        document.getElementById('detailsBtn').addEventListener('click', function() {
            if(dtCat != 'SI' && dtCat != 'SIServices') {
                var url = '<%=request.getContextPath() + "/showFailureServicesWise.do?action=showFailureServicesWise&url="%>' 
                    + encodeURIComponent(dtApi) 
                    + '&ServiceCategory=' + encodeURIComponent(dtCat);
                window.open(url, '_blank');
            }
        });
        
        document.getElementById('openPopupButton').addEventListener('click', function() {
            window.open('<%=request.getContextPath() + "/jsp/DisplayHundredPercentFailure.jsp"%>', '_blank');
        });
        
        document.getElementById('openPopup').addEventListener('click', function() {
            window.open('DisplayTodayServiceWiseFailure.jsp', 'DataPopup', 'width=800,height=600');
        });
    });
    
    // ===== Timer =====
    function countdownDynatrace(el, minutes, seconds) {
        var element = document.getElementById(el);
        if(!element) return;
        var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
        function update() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var t = new Date(msLeft);
                var m = t.getUTCMinutes();
                var s = t.getUTCSeconds();
                element.innerHTML = (m ? m + ':' : '') + (m ? String(s).padStart(2,'0') : s);
                setTimeout(update, 100);
            }
        }
        update();
    }
    countdownDynatrace('ten-countdown', 5, 0);
    
    // ===== Click outside =====
    document.getElementById('modalContainer').addEventListener('click', function(e) {
        if(e.target === this) closeDynatraceModal();
    });
    
    document.addEventListener('keydown', function(e) {
        if(e.key === 'Escape') closeDynatraceModal();
    });
</script>
