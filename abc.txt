    // ============================================================
    // TOOLTIP
    // ============================================================
    document.addEventListener('DOMContentLoaded', function() {
        var tooltip = document.getElementById('tooltip');
        var tiles = document.querySelectorAll('[data-tip]');
        var timeoutId = null;
        
        tiles.forEach(function(el) {
            el.addEventListener('mouseenter', function(e) {
                tooltip.textContent = this.getAttribute('data-tip');
                tooltip.style.display = 'block';
                var rect = this.getBoundingClientRect();
                tooltip.style.left = (rect.left + rect.width/2) + 'px';
                tooltip.style.top = (rect.top - 10) + 'px';
                if(timeoutId) { clearTimeout(timeoutId); timeoutId = null; }
            });
            el.addEventListener('mousemove', function(e) {
                tooltip.style.left = e.clientX + 'px';
                tooltip.style.top = (e.clientY - 10) + 'px';
            });
            el.addEventListener('mouseleave', function() {
                timeoutId = setTimeout(function() {
                    tooltip.style.display = 'none';
                    timeoutId = null;
                }, 100);
            });
        });
        
        // ============================================================
        // LOADER
        // ============================================================
        setTimeout(function() {
            document.getElementById('loader').classList.add('hidden');
        }, 1500);
        window.addEventListener('load', function() {
            document.getElementById('loader').classList.add('hidden');
        });
        
        // ============================================================
        // CLOCK
        // ============================================================
        function updateClock() {
            var now = new Date();
            document.getElementById('currentTime').textContent = 
                now.toLocaleTimeString('en-IN', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' });
        }
        updateClock();
        setInterval(updateClock, 1000);
        
        // ============================================================
        // STATS
        // ============================================================
        var totalCrit = document.querySelectorAll('.tile-critical, .tile-td, .tile-bd, .tile-hundred-mixed').length || 0;
        var totalWarn = document.querySelectorAll('.tile-warning').length || 0;
        var totalHealth = document.querySelectorAll('.tile-healthy:not(.tile-critical):not(.tile-warning):not(.tile-td):not(.tile-bd):not(.tile-hundred-mixed), .tile-hundred-healthy').length || 0;
        var total = totalCrit + totalWarn + totalHealth;
        var hundredCount = document.querySelectorAll('.tile-td, .tile-bd, .tile-hundred-mixed').length || 0;
        
        document.getElementById('totalServices').textContent = total || 48;
        document.getElementById('criticalCount').textContent = totalCrit;
        document.getElementById('warningCount').textContent = totalWarn;
        document.getElementById('healthyCount').textContent = totalHealth;
        document.getElementById('hundredCount').textContent = hundredCount;

        // MODAL
        // ============================================================
        window.openModal = function(msg, title) {
            document.getElementById('modalBody').innerHTML = msg;
            document.getElementById('modalTitle').textContent = title || 'Service Alert';
            document.getElementById('modalOverlay').classList.add('active');
        };
        
        window.closeModal = function() {
            document.getElementById('modalOverlay').classList.remove('active');
        };
        
        document.getElementById('modalOverlay').addEventListener('click', function(e) {
            if(e.target === this) closeModal();
        });

            if(e.key === 'Escape') closeModal();
        });
    });
