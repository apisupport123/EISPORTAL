<script>
    // ===== Premium Alert Handler =====
    var currentApiName = '';
    var currentFailurePct = '';
    var currentCategory = '';
    var currentSourceId = '';
    var currentErrorCode = '';
    var currentTotalHits = '';
    var currentTDCount = '';
    var currentIssueTime = '';
    
    function openPremiumAlert(msg, apiName, failurePct, category, sourceId, errorCode, totalHits, tdCount, issueTime) {
        currentApiName = apiName || '';
        currentFailurePct = failurePct || '';
        currentCategory = category || 'GEN-5';
        currentSourceId = sourceId || '';
        currentErrorCode = errorCode || '';
        currentTotalHits = totalHits || '';
        currentTDCount = tdCount || '';
        currentIssueTime = issueTime || '';
        
        document.getElementById('modalBody').innerHTML = msg;
        document.getElementById('modalContainer').style.display = 'flex';
        document.getElementById('TextArea').value = '';
        document.getElementById('closeName').value = '';
    }
    
    function closePremiumModal() {
        document.getElementById('modalContainer').style.display = 'none';
    }
    
    // ===== Close Button Handler =====
    document.addEventListener('DOMContentLoaded', function() {
        document.getElementById('closeBtn').addEventListener('click', function() {
            var comment = document.getElementById('TextArea').value;
            var closer = document.getElementById('closeName').value;
            var pTime = document.getElementById('pasuseTime').value;
            
            if(comment.trim() == '' || closer == '' || closer == 'Please Select Name') {
                alert('⚠️ Please fill Comment and select Closed By');
                return false;
            }
            
            var url = '<%=request.getContextPath() + "/jsp/ServiceWiseFailureAction.jsp?Api_Name="%>' 
                + encodeURIComponent(currentApiName) 
                + '&SourceID=' + encodeURIComponent(currentSourceId)
                + '&TextArea=' + encodeURIComponent(comment)
                + '&pasuseTime=' + encodeURIComponent(pTime)
                + '&ErrorCode=' + encodeURIComponent(currentErrorCode)
                + '&closeName=' + encodeURIComponent(closer)
                + '&ServiceCategory=' + encodeURIComponent(currentCategory);
            
            closePremiumModal();
            location.replace(url);
            return false;
        });
        
        // ===== Details Button =====
        document.getElementById('detailsBtn').addEventListener('click', function() {
            if(currentCategory != 'SIServices' && currentCategory != 'SI') {
                var url = '<%=request.getContextPath() + "/showFailureServicesWise.do?action=showFailureServicesWise&url="%>' 
                    + encodeURIComponent(currentApiName) 
                    + '&ServiceCategory=' + encodeURIComponent(currentCategory);
                window.open(url, '_blank');
            }
            return false;
        });
        
        // ===== Popup Buttons =====
        document.getElementById('openPopupButton').addEventListener('click', function() {
            window.open('<%=request.getContextPath() + "/jsp/DisplayHundredPercentFailure.jsp"%>', '_blank');
        });
        
        document.getElementById('openPopup').addEventListener('click', function() {
            window.open('DisplayTodayServiceWiseFailure.jsp', 'DataPopup', 'width=800,height=600');
        });
    });
    
    // ===== Timer =====
    function countdown(elementName, minutes, seconds) {
        var element = document.getElementById(elementName);
        if(!element) return;
        var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
        function updateTimer() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var time = new Date(msLeft);
                var mins = time.getUTCMinutes();
                var secs = time.getUTCSeconds();
                element.innerHTML = (mins ? mins + ':' : '') + (mins ? String(secs).padStart(2,'0') : secs);
                setTimeout(updateTimer, 100);
            }
        }
        updateTimer();
    }
    countdown("ten-countdown", 5, 0);
    
    // ===== Click outside modal to close =====
    document.getElementById('modalContainer').addEventListener('click', function(e) {
        if(e.target === this) closePremiumModal();
    });
    
    // ===== ESC key to close =====
    document.addEventListener('keydown', function(e) {
        if(e.key === 'Escape') closePremiumModal();
    });
</script>
