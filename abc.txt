<script>
    var sourceData = {};
    var activityLog = [];
    var MAX_ACTIVITY = 50;
    var timerSeconds = 600; // 10 minutes
    var timerInterval;
    var refreshInterval;

    function formatTime(seconds) {
        var mins = Math.floor(seconds / 60);
        var secs = seconds % 60;
        return String(mins).padStart(2, '0') + ':' + String(secs).padStart(2, '0');
    }

    function updateTimer() {
        timerSeconds--;
        if (timerSeconds <= 0) {
            timerSeconds = 600;
        }
        document.getElementById('timerDisplay').textContent = formatTime(timerSeconds);
    }

    function addActivity(sourceId, status) {
        var now = new Date();
        var timeStr = now.getHours().toString().padStart(2, '0') + ':' + 
                      now.getMinutes().toString().padStart(2, '0') + ':' + 
                      now.getSeconds().toString().padStart(2, '0');
        
        activityLog.unshift({
            time: timeStr,
            source: sourceId,
            status: status
        });
        
        if (activityLog.length > MAX_ACTIVITY) {
            activityLog.pop();
        }
        renderActivity();
    }

    function renderActivity() {
        var container = document.getElementById('activityList');
        if (activityLog.length === 0) {
            container.innerHTML = `
                <div class="activity-empty">
                    <i class="fas fa-inbox"></i>
                    <span>Waiting for activity...</span>
                </div>
            `;
            return;
        }
        
        var html = '';
        for (var i = 0; i < Math.min(activityLog.length, 30); i++) {
            var item = activityLog[i];
            var statusClass = item.status === 'Y' ? 'on' : 'off';
            var statusText = item.status === 'Y' ? 'ON' : 'OFF';
            html += `
                <div class="activity-item ${statusClass}">
                    <span class="activity-time">${item.time}</span>
                    <span class="activity-source">${item.source}</span>
                    <span class="activity-status ${statusClass}">${statusText}</span>
                </div>
            `;
        }
        container.innerHTML = html;
    }

    function fetchData() {
        fetch('?action=fetchSourceStatus')
            .then(function(response) { return response.json(); })
            .then(function(data) {
                renderSources(data);
                updateStats(data);
                updateActivityLog(data);
                document.getElementById('lastUpdate').textContent = 'Updated: ' + new Date().toLocaleTimeString();
            })
            .catch(function(error) {
                console.error('Error:', error);
            });
    }

    function renderSources(data) {
        var container = document.getElementById('sourcesGrid');
        if (!data || data.length === 0) {
            container.innerHTML = `
                <div style="text-align:center;padding:40px;color:var(--text3);">
                    <i class="fas fa-exclamation-triangle" style="font-size:1.5rem;"></i>
                    <p style="margin-top:8px;">No data available</p>
                </div>
            `;
            return;
        }

        var html = '';
        var activeCount = 0;
        for (var i = 0; i < data.length; i++) {
            var item = data[i];
            var isActive = item.flag === 'Y';
            if (isActive) activeCount++;
            var statusClass = isActive ? 'active' : 'inactive';
            var statusIcon = isActive ? '✅' : '⭕';
            var statusText = isActive ? 'ONLINE' : 'OFFLINE';
            
            html += `
                <div class="source-card ${statusClass}" data-source="${item.source_id}">
                    <span class="badge-time">${isActive ? '🟢' : '🔴'}</span>
                    <span class="source-status-icon">${statusIcon}</span>
                    <div class="source-id">${item.source_id}</div>
                    <div class="source-status">${statusText}</div>
                </div>
            `;
        }
        container.innerHTML = html;
        document.getElementById('sourceCount').textContent = activeCount + '/' + data.length;
    }

    function updateStats(data) {
        var active = 0;
        var inactive = 0;
        for (var i = 0; i < data.length; i++) {
            if (data[i].flag === 'Y') {
                active++;
            } else {
                inactive++;
            }
        }
        document.getElementById('activeCount').textContent = active;
        document.getElementById('inactiveCount').textContent = inactive;
        document.getElementById('totalSources').textContent = data.length;
        var uptime = data.length > 0 ? Math.round((active / data.length) * 100) : 0;
        document.getElementById('uptimePercent').textContent = uptime + '%';
    }

    function updateActivityLog(data) {
        var currentActive = {};
        for (var i = 0; i < data.length; i++) {
            if (data[i].flag === 'Y') {
                currentActive[data[i].source_id] = true;
            }
        }
        
        // Check for changes
        for (var sourceId in currentActive) {
            if (!sourceData[sourceId]) {
                // Newly active
                addActivity(sourceId, 'Y');
            } else if (sourceData[sourceId] !== currentActive[sourceId]) {
                // Status changed
                addActivity(sourceId, currentActive[sourceId] ? 'Y' : 'N');
            }
        }
        
        // Check for sources that went inactive
        for (var oldSource in sourceData) {
            if (sourceData[oldSource] && !currentActive[oldSource]) {
                // Went inactive
                addActivity(oldSource, 'N');
            }
        }
        
        sourceData = currentActive;
    }

    function startDashboard() {
        // Initial fetch
        fetchData();
        
        // Refresh every 10 seconds
        refreshInterval = setInterval(fetchData, 10000);
        
        // Timer countdown
        timerInterval = setInterval(updateTimer, 1000);
        
        // Update timer display initial
        document.getElementById('timerDisplay').textContent = formatTime(timerSeconds);
    }

    // Cleanup on page unload
    window.addEventListener('beforeunload', function() {
        if (refreshInterval) clearInterval(refreshInterval);
        if (timerInterval) clearInterval(timerInterval);
    });

    // Start the dashboard
    startDashboard();
