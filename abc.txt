    private java.util.Date computeDeadline(String taskName, java.sql.Date reportDate) {
        String[] parts = taskName.split("-");
        if (parts.length != 2) return null;
        String endStr = parts[1].trim();
        SimpleDateFormat sdf = new SimpleDateFormat("ha", Locale.US);
        try {
            java.util.Date endTime = sdf.parse(endStr);
            Calendar cal = Calendar.getInstance();
            cal.setTime(reportDate);
            cal.add(Calendar.DATE, 1);
            Calendar timeCal = Calendar.getInstance();
            timeCal.setTime(endTime);
            cal.set(Calendar.HOUR_OF_DAY, timeCal.get(Calendar.HOUR_OF_DAY));
            cal.set(Calendar.MINUTE, timeCal.get(Calendar.MINUTE));
            cal.set(Calendar.SECOND, 0);
            cal.set(Calendar.MILLISECOND, 0);
            cal.add(Calendar.MINUTE, 30);
            return cal.getTime();
        } catch (ParseException e) {
            e.printStackTrace();
            return null;
        }
    }

    function showToast(msg, type) {
        const toastEl = document.getElementById('liveToast');

        toastEl.className = 'toast align-items-center text-white border-0';

        if (!toastInstance) toastInstance = new bootstrap.Toast(toastEl, { autohide: true, delay: 4000 });

    function highlightRow(selectElem, rowNum) {

        if (selectElem.value && selectElem.value.trim() !== '') {
            tr.classList.add('row-highlight');

    function saveAll() {
        if (saveInProgress) return;
        if ($('button.btn-success').prop('disabled')) {
            showToast('Cannot save: View is read-only.', 'warning');
            return;
        }

        var mapping = [];
        var doneByFields = document.getElementsByName('done_by[]');
        var sectionFields = document.getElementsByName('section[]');
        var taskFields = document.getElementsByName('task_name[]');
        for (var i = 0; i < doneByFields.length; i++) {
            var doneVal = doneByFields[i].value;
            if (doneVal && doneVal.trim() !== '') {
                var section = sectionFields[i] ? sectionFields[i].value : '?';
                var task = taskFields[i] ? taskFields[i].value : '?';
                mapping.push('Row ' + (i+1) + ': ' + doneVal + ' (' + section + ' - ' + task + ')');
            }
        }

        var msg = 'The following Done By values will be saved:\n\n';
        if (mapping.length === 0) {
            msg += '⚠️ No Done By values selected.\n';
        } else {
            msg += mapping.join('\n');
        }
        msg += '\n\nProceed with save?';
        if (!confirm(msg)) {
            return;
        }

        saveInProgress = true;
        $('#spinnerOverlay').fadeIn(200);

        var formData = new FormData($('#checklistForm')[0]);

        $.ajax({
            url: window.location.pathname,
            type: 'POST',
            data: formData,
            processData: false,
            contentType: false,
            success: function() {
                showToast('Data saved successfully!', 'success');
                $('#lastSaved').text('Last saved: ' + new Date().toLocaleTimeString());
                setTimeout(function(){ location.reload(); }, 1000);
            },
            error: function(xhr, status, error) { showToast('Error saving: ' + error, 'error'); },
            complete: function() { saveInProgress = false; $('#spinnerOverlay').fadeOut(300); }
        });
    }
