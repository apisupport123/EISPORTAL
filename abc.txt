SELECT s.source_id,
       NVL2(a.source_id, 'Y', 'N') AS flag
FROM (
    SELECT 'SA' source_id FROM dual
    UNION ALL SELECT 'FS' FROM dual
    UNION ALL SELECT 'I4' FROM dual
    UNION ALL SELECT 'Y2' FROM dual
    UNION ALL SELECT 'KA' FROM dual
) s
LEFT JOIN (
    SELECT DISTINCT source_id
    FROM EISAPP.API_LOG
    WHERE source_id IN ('SA','FS','I4','Y2','KA')
      AND request_date_time >= SYSDATE - INTERVAL '10' MINUTE
) a ON s.source_id = a.source_id;


---------------------------------------------

WITH source_list AS (
    SELECT 'SA' source_id FROM dual
    UNION ALL SELECT 'FS' FROM dual
    UNION ALL SELECT 'I4' FROM dual
    UNION ALL SELECT 'Y2' FROM dual
    UNION ALL SELECT 'KA' FROM dual
),
active_sources AS (
    SELECT DISTINCT source_id
    FROM EISAPP.API_LOG
    WHERE source_id IN ('SA','FS','I4','Y2','KA')
      AND request_date_time >= SYSDATE - INTERVAL '10' MINUTE
)
SELECT sl.source_id,
       CASE WHEN asrc.source_id IS NOT NULL THEN 'Y' ELSE 'N' END AS flag
FROM source_list sl
LEFT JOIN active_sources asrc ON sl.source_id = asrc.source_id;
