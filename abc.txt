public class DBUtilityAuto {
    private static Connection connection = null;

    private static String DB_ENVLOCATION = null;
    private static final Object lock = new Object();

    public static void flag() {
        DBConfig.checkAndReload();
       
        synchronized (lock) {
            try {
                Class.forName(DBConfig.getDriver());
               
                // PRIMARY CONNECTION ATTEMPT
                try {
                    connection = DriverManager.getConnection(
                        "jdbc:oracle:thin:@10.191.170.109:1787/REPODB_SRV",
                        DBConfig.getEisappsupportUsername(),
                        DBConfig.getPrimaryPassword()  // From properties file
                    );
                    System.out.println("Connected to PRIMARY DB 1");
                } catch(Exception E) {
                    connection = DriverManager.getConnection(
                        "jdbc:oracle:thin:@10.191.170.110:1787/REPODB_SRV",
                        DBConfig.getEisappsupportUsername(),
                        DBConfig.getPrimaryPassword()  // From properties file
                    );
                    System.out.println("Connected to PRIMARY DB 2");
                }
               
                // GET FLAG STATUS
                String sql = "select * from connection_master";
                try (Statement smt = connection.createStatement();
                     ResultSet rs = smt.executeQuery(sql)) {
                    
                    while (rs.next()) {
                        flagStatus = rs.getString("FLAG");
                        DB_ENVLOCATION = rs.getString("DB_LOCATION");
                        
                        if (flagStatus == null) flagStatus = "PR";
                        if (DB_ENVLOCATION == null) DB_ENVLOCATION = "MI";
                    }
                    
                    if (flagStatus == null || DB_ENVLOCATION == null) {
                        flagStatus = "PR";
                        DB_ENVLOCATION = "MI";
                        System.out.println("No flag data found, using defaults: PR/MI");
                    }
                    
                    System.out.println("Flag Status: " + flagStatus + ", Location: " + DB_ENVLOCATION);
                }
               
            } catch (Exception e) {
                System.out.println("Primary connection failed, trying DR...");
               
                try {
                    // DR CONNECTION ATTEMPT
                    Class.forName(DBConfig.getDriver());
                    try {
                        connection = DriverManager.getConnection(
                            "jdbc:oracle:thin:@10.176.0.109:1787/sidashdb_srv",
                            DBConfig.getEisappsupportUsername(),
                            DBConfig.getPrimaryPassword()  // From properties file
                        );
                        System.out.println("Connected to DR DB 1");
                    } catch(Exception Er) {
                        connection = DriverManager.getConnection(
                            "jdbc:oracle:thin:@10.176.0.110:1787/sidashdb_srv",
                            DBConfig.getEisappsupportUsername(),
                            DBConfig.getPrimaryPassword()  // From properties file
                        );
                        System.out.println("Connected to DR DB 2");
                    }
                   
                    // GET FLAG STATUS FROM DR
                    String sql = "select * from connection_master";
                    try (Statement smt = connection.createStatement();
                         ResultSet rs = smt.executeQuery(sql)) {
                        
                        while (rs.next()) {
                            flagStatus = rs.getString("FLAG");
                            DB_ENVLOCATION = rs.getString("DB_LOCATION");
                            
                            if (flagStatus == null) flagStatus = "DR";
                            if (DB_ENVLOCATION == null) DB_ENVLOCATION = "MI";
                        }
                        
                        if (flagStatus == null || DB_ENVLOCATION == null) {
                            flagStatus = "DR";
                            DB_ENVLOCATION = "MI";
                            System.out.println("No flag data found in DR, using defaults: DR/MI");
                        }
                        
                        System.out.println("Flag Status: " + flagStatus + ", Location: " + DB_ENVLOCATION);
                    }
                   
                } catch(Exception e1) {
                    System.out.println("All connection attempts failed");
                    flagStatus = "PR";
                    DB_ENVLOCATION = "MI";
                    e1.printStackTrace();
                }
            }
        }
    }

    public static Connection getIIBDBConnection() {
        DBConfig.checkAndReload();
       
        flag();
        System.out.println("getIIBDBConnection - flagStatus: " + flagStatus + ", DB_ENVLOCATION: " + DB_ENVLOCATION);
       
        if (flagStatus == null || DB_ENVLOCATION == null) {
            flagStatus = "PR";
            DB_ENVLOCATION = "MI";
            System.out.println("Using default values due to null flag/location");
        }
        
        if(flagStatus.equals("DR") && DB_ENVLOCATION.equals("MI")) {    
            System.out.println("In Hyderabad DR Environment");
            return createIIBHyderabadDRConnection();
        } else if(flagStatus.equals("PR") && DB_ENVLOCATION.equals("MI")) {
            System.out.println("In Hyderabad PR Environment");
            return createIIBHyderabadPRConnection();
        } else {
            System.out.println("In Mumbai Environment");
           
            if(flagStatus.equals("DR")) {
                System.out.println("In Mumbai DR");
                return createIIBMumbaiDRConnection();
            } else if(flagStatus.equals("PR")) {
                System.out.println("In Mumbai PR");
                return createIIBMumbaiPRConnection();
            }
        }
        
        return createIIBMumbaiPRConnection();
    }
