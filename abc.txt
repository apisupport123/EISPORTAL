    // Force complete cache prevention
    response.setHeader("Cache-Control", "no-cache, no-store, must-revalidate, private");

    // ============================================================
    // 1. Declare all variables at page level (outside try)
    // ============================================================
    Timestamp startTs = null;
    Timestamp endTs = null;
    String runTimeDisplay = "No data";
    String fromDateTime = "";
    String toDateTime = "";
    String flagStatus = "DR";

    Map<String, String> brokerMapping = new HashMap<>();
    Map<String, String> portMapping = new HashMap<>();
    Map<String, Integer> brokerOrderMapping = new HashMap<>();
    Map<String, List<String>> brokerIPMapping = new HashMap<>();
    List<Map<String, Object>> brokerDataList = new ArrayList<>();
    List<Map<String, String>> impactedServersList = new ArrayList<>();
    List<Map<String, String>> isolatedServersList = new ArrayList<>();
    Map<String, List<String>> criticalServersByBroker = new LinkedHashMap<>();

    int totalBrokers = 0;
    int criticalBrokers = 0;
    int warningBrokers = 0;
    int healthyBrokers = 0;
    int isolatedBrokersCount = 0;
    int totalServers = 0;
    int impactedServers = 0;
    int isolatedServers = 0;

    Connection reportConn = null;
    try {
        reportConn = DBUtilityAuto.getReportDBConnection();

        // ============================================================
        // 2. Get time parameters from request, default to most recent hour
        // ============================================================
        String fromParam = request.getParameter("fromDateTime");
        String toParam = request.getParameter("toDateTime");

        if (fromParam == null || toParam == null || fromParam.isEmpty() || toParam.isEmpty()) {
            String latestSql = "SELECT MAX(RUN_START) AS MAX_START FROM EXP_BROKER_HOURLY_STATS";
            try (Statement stmt = reportConn.createStatement();
                 ResultSet rs = stmt.executeQuery(latestSql)) {
                if (rs.next()) {
                    Timestamp maxStart = rs.getTimestamp("MAX_START");
                    if (maxStart != null) {
                        Calendar cal = Calendar.getInstance();
                        cal.setTime(maxStart);
                        cal.add(Calendar.HOUR_OF_DAY, 1);
                        Timestamp maxEnd = new Timestamp(cal.getTimeInMillis());
                        startTs = maxStart;
                        endTs = maxEnd;
                    }
                }
            }
            if (startTs == null) {
                Calendar cal = Calendar.getInstance();
                cal.set(Calendar.MINUTE, 0);
                cal.set(Calendar.SECOND, 0);
                cal.set(Calendar.MILLISECOND, 0);
                endTs = new Timestamp(cal.getTimeInMillis());
                cal.add(Calendar.HOUR_OF_DAY, -1);
                startTs = new Timestamp(cal.getTimeInMillis());
            }
        } else {
            SimpleDateFormat isoFmt = new SimpleDateFormat("yyyy-MM-dd'T'HH:mm");
            java.util.Date startDate = isoFmt.parse(fromParam);
            java.util.Date endDate = isoFmt.parse(toParam);
            startTs = new Timestamp(startDate.getTime());
            endTs = new Timestamp(endDate.getTime());
        }

        SimpleDateFormat displayFmt = new SimpleDateFormat("dd-MM-yyyy HH:mm");
        runTimeDisplay = displayFmt.format(new Date(startTs.getTime())) + " to " + displayFmt.format(new Date(endTs.getTime()));
        SimpleDateFormat inputFmt = new SimpleDateFormat("yyyy-MM-dd'T'HH:mm");
        fromDateTime = inputFmt.format(new Date(startTs.getTime()));
        toDateTime = inputFmt.format(new Date(endTs.getTime()));

        // ============================================================
        // 3. Fetch broker mapping and IP mapping
        // ============================================================
        String mappingQuery = "SELECT BROKER_NAME, EXP_PORT, MAPPED_PORT FROM EXP_BROKER_PORT_MAPPING";
        try (Statement stmt = reportConn.createStatement();
             ResultSet rs = stmt.executeQuery(mappingQuery)) {
            while (rs.next()) {
                String broker = rs.getString("BROKER_NAME");
                String expPort = rs.getString("EXP_PORT");
                String mappedPort = rs.getString("MAPPED_PORT");
                if (broker != null && expPort != null && mappedPort != null) {
                    brokerMapping.put(broker.trim(), expPort.trim());
                    portMapping.put(broker.trim(), mappedPort.trim());
                }
            }
        }

        String orderQuery = "SELECT BROKER_NAME, ORDER_NUMBER FROM EXP_BROKER_PORT_MAPPING";
        try (Statement stmt = reportConn.createStatement();
             ResultSet rs = stmt.executeQuery(orderQuery)) {
            while (rs.next()) {
                String broker = rs.getString("BROKER_NAME");
                int order = rs.getInt("ORDER_NUMBER");
                if (broker != null) {
                    brokerOrderMapping.put(broker.trim(), order);
                }
            }
        }

        try (Statement stmt = reportConn.createStatement();
             ResultSet rs = stmt.executeQuery("SELECT FLAG FROM connection_master")) {
            if (rs.next()) {
                flagStatus = rs.getString("FLAG");
                if (flagStatus == null) flagStatus = "DR";
            }
        }

        String ipQuery = "SELECT DISTINCT a.EXP_IP, a.BROKER_NAME FROM EXP_IP_BROKER_DETAILS a ORDER BY a.BROKER_NAME, a.EXP_IP";
        try (Statement stmt = reportConn.createStatement();
             ResultSet rs = stmt.executeQuery(ipQuery)) {
            while (rs.next()) {
                String broker = rs.getString("BROKER_NAME");
                String ip = rs.getString("EXP_IP");
                if (broker == null || ip == null) continue;
                broker = broker.trim();
                ip = ip.trim();
                boolean include = false;
                if ("PR".equals(flagStatus)) {
                    include = ip.startsWith("10.188.24.") || ip.startsWith("10.188.25.");
                } else if ("DR".equals(flagStatus)) {
                    include = ip.startsWith("10.177.40.") || ip.startsWith("10.177.41.");
                }
                if (include) {
                    brokerIPMapping.computeIfAbsent(broker, k -> new ArrayList<>()).add(ip);
                }
            }
            for (List<String> ips : brokerIPMapping.values()) {
                Collections.sort(ips, (ip1, ip2) -> {
                    try {
                        String[] p1 = ip1.split("\\.");
                        String[] p2 = ip2.split("\\.");
                        int o1 = Integer.parseInt(p1[3]);
                        int o2 = Integer.parseInt(p2[3]);
                        return Integer.compare(o1, o2);
                    } catch (Exception e) {
                        return ip1.compareTo(ip2);
                    }
                });
            }
        }

        // ============================================================
        // 4. Fetch broker stats from EXP_BROKER_HOURLY_STATS
        // ============================================================
        brokerDataList.clear();
        String brokerStatsSQL = "SELECT BROKER, PORT, TOTAL_HITS, HEALTHY_SERVERS, WARNING_SERVERS, " +
                "CRITICAL_SERVERS, ISOLATED_SERVERS, TOTAL_SERVERS, STATUS " +
                "FROM EXP_BROKER_HOURLY_STATS WHERE RUN_START = ? AND RUN_END = ?";
        try (PreparedStatement ps = reportConn.prepareStatement(brokerStatsSQL)) {
            ps.setTimestamp(1, startTs);
            ps.setTimestamp(2, endTs);
            try (ResultSet rs = ps.executeQuery()) {
                while (rs.next()) {
                    Map<String, Object> brokerData = new HashMap<>();
                    String broker = rs.getString("BROKER");
                    String port = rs.getString("PORT");
                    int totalHits = rs.getInt("TOTAL_HITS");
                    int healthy = rs.getInt("HEALTHY_SERVERS");
                    int warning = rs.getInt("WARNING_SERVERS");
                    int critical = rs.getInt("CRITICAL_SERVERS");
                    int isolated = rs.getInt("ISOLATED_SERVERS");
                    int totalServersRow = rs.getInt("TOTAL_SERVERS");
                    String status = rs.getString("STATUS");

                    List<String> ipList = brokerIPMapping.getOrDefault(broker, new ArrayList<>());
                    brokerData.put("broker", broker);
                    brokerData.put("port", port);
                    brokerData.put("totalHits", totalHits);
                    brokerData.put("brokerStatus", status);
                    brokerData.put("healthyCount", healthy);
                    brokerData.put("warningCount", warning);
                    brokerData.put("criticalCount", critical);
                    brokerData.put("isolatedCount", isolated);
                    brokerData.put("serverCount", totalServersRow);
                    brokerData.put("ipList", ipList);
                    brokerDataList.add(brokerData);
                }
            }
        }

        // ============================================================
        // 5. Fetch impacted servers from EXP_IMPACTED_SERVERS_HOURLY
        // ============================================================
        impactedServersList.clear();
        isolatedServersList.clear();
        criticalServersByBroker.clear();
        Map<String, Map<String, String>> allServerStats = new HashMap<>();

        String impactedSQL = "SELECT BROKER, IP, HITS, STATUS FROM EXP_IMPACTED_SERVERS_HOURLY " +
                "WHERE RUN_START = ? AND RUN_END = ? ORDER BY STATUS, BROKER";
        try (PreparedStatement ps = reportConn.prepareStatement(impactedSQL)) {
            ps.setTimestamp(1, startTs);
            ps.setTimestamp(2, endTs);
            try (ResultSet rs = ps.executeQuery()) {
                while (rs.next()) {
                    String broker = rs.getString("BROKER");
                    String ip = rs.getString("IP");
                    int hits = rs.getInt("HITS");
                    String status = rs.getString("STATUS");

                    String key = broker + "|" + ip;
                    Map<String, String> stats = new HashMap<>();
                    stats.put("hits", String.valueOf(hits));
                    stats.put("status", status);
                    allServerStats.put(key, stats);

                    Map<String, String> server = new HashMap<>();
                    server.put("broker", broker);
                    server.put("ip", ip);
                    server.put("displayIP", ip.split("\\.")[2] + "." + ip.split("\\.")[3]);
                    server.put("hits", String.valueOf(hits));
                    server.put("status", status);

                    if ("isolated".equals(status)) {
                        isolatedServersList.add(server);
                    } else if ("critical".equals(status) || "warning".equals(status)) {
                        impactedServersList.add(server);
                        if ("critical".equals(status)) {
                            criticalServersByBroker.computeIfAbsent(broker, k -> new ArrayList<>()).add(server.get("displayIP"));
                        }
                    }
                }
            }
        }

        // ============================================================
        // 6. Attach server details to each broker using allServerStats
        // ============================================================
        for (Map<String, Object> brokerData : brokerDataList) {
            String broker = (String) brokerData.get("broker");
            List<String> ipList = (List<String>) brokerData.get("ipList");
            List<Map<String, String>> serverList = new ArrayList<>();
            for (String ip : ipList) {
                String key = broker + "|" + ip;
                Map<String, String> stats = allServerStats.get(key);
                String status = "healthy";
                String hits = "0";
                if (stats != null) {
                    status = stats.get("status");
                    hits = stats.get("hits");
                }
                Map<String, String> serverInfo = new HashMap<>();
                serverInfo.put("ip", ip);
                serverInfo.put("displayIP", ip.split("\\.")[2] + "." + ip.split("\\.")[3]);
                serverInfo.put("status", status);
                serverInfo.put("hits", hits);
                serverList.add(serverInfo);
            }
            brokerData.put("serverList", serverList);
        }

        // ============================================================
        // 7. Sort broker cards: isolated, critical, warning, healthy
        // ============================================================
        brokerDataList.sort((a, b) -> {
            String statusA = (String) a.get("brokerStatus");
            String statusB = (String) b.get("brokerStatus");
            int priorityA = statusA.equals("isolated") ? 0 : statusA.equals("critical") ? 1 : statusA.equals("warning") ? 2 : 3;
            int priorityB = statusB.equals("isolated") ? 0 : statusB.equals("critical") ? 1 : statusB.equals("warning") ? 2 : 3;
            return Integer.compare(priorityA, priorityB);
        });

        // ============================================================
        // 8. Compute counts for stats grid
        // ============================================================
        totalBrokers = brokerDataList.size();
        criticalBrokers = 0;
        warningBrokers = 0;
        healthyBrokers = 0;
        isolatedBrokersCount = 0;
        impactedServers = impactedServersList.size();
        isolatedServers = isolatedServersList.size();

        // Unique server count: union of all IPs across all brokers
        Set<String> uniqueIPs = new HashSet<>();
        for (List<String> ips : brokerIPMapping.values()) {
            uniqueIPs.addAll(ips);
        }
        totalServers = uniqueIPs.size();

        for (Map<String, Object> bd : brokerDataList) {
            String status = (String) bd.get("brokerStatus");
            if ("critical".equals(status)) criticalBrokers++;
            else if ("warning".equals(status)) warningBrokers++;
            else if ("isolated".equals(status)) isolatedBrokersCount++;
            else if ("healthy".equals(status)) healthyBrokers++;
        }

    } catch (Exception e) {
        e.printStackTrace();
    } finally {
        if (reportConn != null) try { reportConn.close(); } catch (SQLException e) {}
    }

    let speechSynthesis = window.speechSynthesis;
    let isVoiceEnabled = true;
    let isMuted = false;
    let isPlaying = false;
    let voices = [];
    let userInteracted = false;
    let voiceInitialized = false;
    let voicePlayCount = 0;
    const MAX_VOICE_PLAYS = 3;

    const criticalServersByBroker = { <%= jsBrokerMap.toString() %> };
    const hasUserInteracted = localStorage.getItem('voiceEnabled') === 'true' || localStorage.getItem('chromeVoiceEnabled') === 'true';

    console.log('Critical servers by broker:', criticalServersByBroker);

    function cleanForPronunciation(text) {
        if (!text) return '';
        let cleaned = text.replace(/EXP/gi, 'E X P');
        cleaned = cleaned.replace(/NP_SPL/gi, 'N P S P L');
        cleaned = cleaned.replace(/_/g, ' ');
        cleaned = cleaned.replace(/\./g, ' dot ');
        return cleaned;
    }

    function buildSequentialMessage() {
        const brokerKeys = Object.keys(criticalServersByBroker);
        if (brokerKeys.length === 0) return '';
        let fullMessage = 'Alert! ';
        let messageParts = [];
        for (let i = 0; i < brokerKeys.length; i++) {
            let broker = brokerKeys[i];
            let servers = criticalServersByBroker[broker];
            let cleanBroker = cleanForPronunciation(broker);
            if (servers.length === 1) {
                messageParts.push(cleanBroker + ' server ' + cleanForPronunciation(servers[0]));
            } else {
                let serverList = servers.map(s => cleanForPronunciation(s)).join(', ');
                messageParts.push(cleanBroker + ' servers ' + serverList);
            }
        }
        if (messageParts.length === 1) return fullMessage + messageParts[0];
        else return fullMessage + messageParts.join(', then ');
    }

    function getTotalCriticalCount() {
        let count = 0;
        for (let broker in criticalServersByBroker) {
            count += criticalServersByBroker[broker].length;
        }
        return count;
    }

    function loadVoices() {
        try { voices = speechSynthesis.getVoices(); } catch(e) {}
    }
    if (speechSynthesis) {
        loadVoices();
        if (speechSynthesis.onvoiceschanged !== undefined) {
            speechSynthesis.onvoiceschanged = loadVoices;
        }
    }

    function getBestVoice() {
        if (voices.length === 0) return null;
        const preferred = ['Google UK English Female','Google US English','Microsoft David','Microsoft Zira','Samantha','Alex','Google UK English Male','en-US','en-GB','English'];
        for (let p of preferred) {
            let voice = voices.find(v => v.name.includes(p) || v.lang.includes(p));
            if (voice) return voice;
        }
        return voices.find(v => v.lang.startsWith('en'));
    }

    function updateVoiceUI() {
        const audioStatus = document.getElementById('audioStatus');
        const testBtn = document.getElementById('testVoiceBtn');
        const muteBtn = document.getElementById('muteVoiceBtn');
        const criticalCount = getTotalCriticalCount();
        if (isVoiceEnabled) {
            testBtn.style.display = 'inline-flex';
            muteBtn.style.display = 'inline-flex';
            if (isMuted) {
                audioStatus.innerHTML = 'Muted <i class="fas fa-volume-mute"></i>';
                audioStatus.style.color = 'var(--gray-500)';
                muteBtn.innerHTML = '<i class="fas fa-volume-up"></i> Unmute';
                muteBtn.style.background = 'var(--success)';
            } else {
                if (criticalCount > 0) {
                    audioStatus.innerHTML = 'Active - ' + criticalCount + ' critical server(s)';
                    audioStatus.style.color = 'var(--danger)';
                } else {
                    audioStatus.innerHTML = 'Active - No critical servers';
                    audioStatus.style.color = 'var(--success)';
                }
                muteBtn.innerHTML = '<i class="fas fa-volume-mute"></i> Mute';
                muteBtn.style.background = 'var(--gray-600)';
            }
        }
    }

    function speakMessage(message, callback) {
        if (!isVoiceEnabled || isMuted || !speechSynthesis || !message) {
            if (callback) callback();
            return false;
        }
        try { speechSynthesis.cancel(); } catch(e) {}
        try {
            const utterance = new SpeechSynthesisUtterance(message);
            utterance.rate = 0.85;
            utterance.pitch = 1.1;
            utterance.volume = 1;
            utterance.lang = 'en-US';
            const bestVoice = getBestVoice();
            if (bestVoice) utterance.voice = bestVoice;
            utterance.onstart = function() { console.log('Speech started'); };
            utterance.onerror = function() { if(callback) callback(); };
            utterance.onend = function() { if(callback) callback(); };
            setTimeout(() => {
                try { speechSynthesis.speak(utterance); } catch(e) { if(callback) callback(); }
            }, 50);
            return true;
        } catch(e) { if(callback) callback(); return false; }
    }

    function playVoiceAlert() {
        if (voicePlayCount >= MAX_VOICE_PLAYS) { console.log('Max voice plays reached.'); return; }
        const criticalCount = getTotalCriticalCount();
        if (criticalCount === 0) return;
        if (!isVoiceEnabled || isMuted) return;
        const message = buildSequentialMessage();
        console.log('Playing voice alert (play #' + (voicePlayCount + 1) + '):', message);
        isPlaying = true;
        voicePlayCount++;
        speakMessage(message, function() { isPlaying = false; updateVoiceUI(); });
        updateVoiceUI();
    }

    function toggleMute() {
        isMuted = !isMuted;
        saveVoicePreferences();
        if (isMuted) try { speechSynthesis.cancel(); } catch(e) {}
        isPlaying = false;
        updateVoiceUI();
    }

            const saved = localStorage.getItem('voiceEnabled');
            if (saved !== null) isVoiceEnabled = saved === 'true';

        } catch(e) {}
    }

    }

        }
    }

    // UI Functions
    function countdown(elementName, minutes, seconds) {
        var element, endTime, time;
        function twoDigits(n) { return (n <= 9 ? "0" + n : n); }
        function updateTimer() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) {
                element.innerHTML = "Refreshing...";
                setTimeout(function() { location.reload(true); }, 1000);
            } else {
                time = new Date(msLeft);
                var hours = time.getUTCHours();
                var mins = time.getUTCMinutes();
                element.innerHTML = (hours ? hours + ':' + twoDigits(mins) : mins) + ':' + twoDigits(time.getUTCSeconds());
                setTimeout(updateTimer, time.getUTCMilliseconds() + 500);
            }
        }
        element = document.getElementById(elementName);
        endTime = (+new Date) + 1000 * (60 * minutes + seconds) + 500;
        updateTimer();
    }

    function cleanForPronunciation(text) {
        if (!text) return '';
        let cleaned = text.replace(/SYS/gi, 'S Y S');
        cleaned = cleaned.replace(/NP_SPL/gi, 'N P S P L');
        cleaned = cleaned.replace(/_/g, ' ');
        cleaned = cleaned.replace(/\./g, ' dot ');
        return cleaned;
    }

    function buildSequentialMessage() {
        const brokerKeys = Object.keys(criticalServersByBroker);
        if (brokerKeys.length === 0) return '';
        let fullMessage = 'Alert! ';
        let messageParts = [];
        for (let i = 0; i < brokerKeys.length; i++) {
            let broker = brokerKeys[i];
            let servers = criticalServersByBroker[broker];
            let cleanBroker = cleanForPronunciation(broker);
            if (servers.length === 1) {
                messageParts.push(cleanBroker + ' server ' + cleanForPronunciation(servers[0]));
            } else {
                let serverList = servers.map(s => cleanForPronunciation(s)).join(', ');
                messageParts.push(cleanBroker + ' servers ' + serverList);
            }
        }
        if (messageParts.length === 1) return fullMessage + messageParts[0];
        else return fullMessage + messageParts.join(', then ');
    }

    function getTotalCriticalCount() {
        let count = 0;
        for (let broker in criticalServersByBroker) {
            count += criticalServersByBroker[broker].length;
        }
        return count;
    }

    function loadVoices() {
        try { voices = speechSynthesis.getVoices(); } catch(e) {}
    }
    if (speechSynthesis) {
        loadVoices();
        if (speechSynthesis.onvoiceschanged !== undefined) {
            speechSynthesis.onvoiceschanged = loadVoices;
        }
    }

    function getBestVoice() {
        if (voices.length === 0) return null;
        const preferred = ['Google UK English Female','Google US English','Microsoft David','Microsoft Zira','Samantha','Alex','Google UK English Male','en-US','en-GB','English'];
        for (let p of preferred) {
            let voice = voices.find(v => v.name.includes(p) || v.lang.includes(p));
            if (voice) return voice;
        }
        return voices.find(v => v.lang.startsWith('en'));
    }

    function updateVoiceUI() {
        const audioStatus = document.getElementById('audioStatus');

        if (isVoiceEnabled) {

            }
        }
    }

    function speakMessage(message, callback) {
        if (!isVoiceEnabled || isMuted || !speechSynthesis || !message) {
            if (callback) callback();
            return false;
        }
        try { speechSynthesis.cancel(); } catch(e) {}
        try {
            const utterance = new SpeechSynthesisUtterance(message);
            utterance.rate = 0.85;
            utterance.pitch = 1.1;
            utterance.volume = 1;
            utterance.lang = 'en-US';
            const bestVoice = getBestVoice();
            if (bestVoice) utterance.voice = bestVoice;
            utterance.onstart = function() { console.log('Speech started'); };
            utterance.onerror = function() { if(callback) callback(); };
            utterance.onend = function() { if(callback) callback(); };
            setTimeout(() => {
                try { speechSynthesis.speak(utterance); } catch(e) { if(callback) callback(); }
            }, 50);
            return true;
        } catch(e) { if(callback) callback(); return false; }
    }

    function playVoiceAlert() {
        if (voicePlayCount >= MAX_VOICE_PLAYS) { console.log('Max voice plays reached.'); return; }
        const criticalCount = getTotalCriticalCount();
        if (criticalCount === 0) return;
        if (!isVoiceEnabled || isMuted) return;
        const message = buildSequentialMessage();
        console.log('Playing voice alert (play #' + (voicePlayCount + 1) + '):', message);
        isPlaying = true;
        voicePlayCount++;
        speakMessage(message, function() { isPlaying = false; updateVoiceUI(); });
        updateVoiceUI();
    }

    function countdown(elementName, minutes, seconds) {
        var element, endTime, time;
        function twoDigits(n) { return (n <= 9 ? "0" + n : n); }
        function updateTimer() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) {
                element.innerHTML = "Refreshing...";
                setTimeout(function() { location.reload(true); }, 1000);
            } else {
                time = new Date(msLeft);
                var hours = time.getUTCHours();
                var mins = time.getUTCMinutes();
                element.innerHTML = (hours ? hours + ':' + twoDigits(mins) : mins) + ':' + twoDigits(time.getUTCSeconds());
                setTimeout(updateTimer, time.getUTCMilliseconds() + 500);
            }
        }
        element = document.getElementById(elementName);
        endTime = (+new Date) + 1000 * (60 * minutes + seconds) + 500;
        updateTimer();
    }

            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var t = new Date(msLeft);
                var m = t.getUTCMinutes();
                var s = t.getUTCSeconds();
                element.innerHTML = (m ? m + ':' : '') + (m ? String(s).padStart(2,'0') : s);
                setTimeout(update, 100);

        }
