    private java.util.Date computeDeadline(String taskName, java.sql.Date reportDate) {
        String[] parts = taskName.split("-");
        if (parts.length != 2) return null;
        String endStr = parts[1].trim();
        SimpleDateFormat sdf = new SimpleDateFormat("ha", Locale.US);
        try {
            java.util.Date endTime = sdf.parse(endStr);
            Calendar cal = Calendar.getInstance();
            cal.setTime(reportDate);
            cal.add(Calendar.DATE, 1);
            Calendar timeCal = Calendar.getInstance();
            timeCal.setTime(endTime);
            cal.set(Calendar.HOUR_OF_DAY, timeCal.get(Calendar.HOUR_OF_DAY));
            cal.set(Calendar.MINUTE, timeCal.get(Calendar.MINUTE));
            cal.set(Calendar.SECOND, 0);
            cal.set(Calendar.MILLISECOND, 0);
            cal.add(Calendar.MINUTE, 30);
            return cal.getTime();
        } catch (ParseException e) {
            e.printStackTrace();
            return null;
        }
    }

    private Map<String, Map<String, Object>> buildDataMap(ResultSet rs) throws SQLException {
        Map<String, Map<String, Object>> map = new HashMap<>();
        while (rs.next()) {
            Map<String, Object> row = new HashMap<>();
            row.put("id", rs.getInt("id"));
            row.put("section", rs.getString("section"));
            row.put("task_name", rs.getString("task_name"));
            row.put("status1", rs.getString("status1"));
            row.put("status2", rs.getString("status2"));
            row.put("status3", rs.getString("status3"));
            row.put("done_by", rs.getString("done_by"));
            row.put("review", rs.getString("review"));
            row.put("remarks", rs.getString("remarks"));
            row.put("updated_at", rs.getTimestamp("updated_at"));
            String key = rs.getString("section") + "|" + rs.getString("task_name");
            map.put(key, row);
        }
        return map;
    }

int hour = nowCal.get(Calendar.HOUR_OF_DAY);

boolean beforeHalfPastNine = (hour < 21) || (hour == 21 && minute < 30);

java.sql.Date todaySQL = new java.sql.Date(nowCal.getTimeInMillis());
Calendar calPrev = Calendar.getInstance();
calPrev.add(Calendar.DATE, -1);
java.sql.Date yesterdaySQL = new java.sql.Date(calPrev.getTimeInMillis());

SimpleDateFormat sdfDisplay = new SimpleDateFormat("dd-MM-yyyy");
SimpleDateFormat sdfParam = new SimpleDateFormat("yyyy-MM-dd");

String viewDateParam = request.getParameter("viewDate");
java.sql.Date displayDate = null;
java.sql.Date saveDate = null;
boolean isCustomView = false;

if (viewDateParam != null && !viewDateParam.isEmpty()) {
    try {
        java.util.Date parsed = sdfParam.parse(viewDateParam);
        displayDate = new java.sql.Date(parsed.getTime());
        saveDate = displayDate;
        isCustomView = true;
    } catch (ParseException e) {
        isCustomView = false;
    }
}

if (!isCustomView) {
    // Default: before 9:30 -> yesterday, after 9:30 -> today
    displayDate = beforeHalfPastNine ? yesterdaySQL : todaySQL;
    saveDate = todaySQL;
}

String displayDateStr = sdfDisplay.format(displayDate);

// Global freeze: if date is not today OR (today but before 9:30)
boolean globalFrozen;
if (isCustomView) {
    globalFrozen = !displayDate.equals(todaySQL) || beforeHalfPastNine;
} else {
    globalFrozen = beforeHalfPastNine;
}

// ===== HANDLE SAVE =====
String action = request.getParameter("action");
String message = "";

if ("save".equals(action)) {
    String[] sections = request.getParameterValues("section[]");
    String[] taskNames = request.getParameterValues("task_name[]");
    String[] status1s = request.getParameterValues("status1[]");
    String[] status2s = request.getParameterValues("status2[]");
    String[] status3s = request.getParameterValues("status3[]");
    String[] doneBys = request.getParameterValues("done_by[]");
    String[] reviews = request.getParameterValues("review[]");
    String[] remarks = request.getParameterValues("remarks[]");

    if (sections != null && sections.length > 0 && !globalFrozen) {
        try {
            conn.setAutoCommit(false);
            String mergeSQL = "MERGE INTO night_checklist t " +
                    "USING (SELECT ? AS report_date, ? AS section, ? AS task_name FROM dual) s " +
                    "ON (t.report_date = s.report_date AND t.section = s.section AND t.task_name = s.task_name) " +
                    "WHEN MATCHED THEN UPDATE SET " +
                    "    status1=?, status2=?, status3=?, done_by=?, review=?, remarks=?, updated_at = SYSTIMESTAMP " +
                    "WHEN NOT MATCHED THEN INSERT " +
                    "    (report_date, section, task_name, status1, status2, status3, done_by, review, remarks, created_at, updated_at) " +
                    "    VALUES (s.report_date, s.section, s.task_name, ?, ?, ?, ?, ?, ?, SYSTIMESTAMP, SYSTIMESTAMP)";
            pstmt = conn.prepareStatement(mergeSQL);
            for (int i = 0; i < sections.length; i++) {
                String section = sections[i];
                String taskName = taskNames[i];
                String status1 = (status1s != null && i < status1s.length) ? status1s[i] : "";
                String status2 = (status2s != null && i < status2s.length) ? status2s[i] : "";
                String status3 = (status3s != null && i < status3s.length) ? status3s[i] : "";
                String doneBy = (doneBys != null && i < doneBys.length) ? doneBys[i] : "";
                String review = (reviews != null && i < reviews.length) ? reviews[i] : "";
                String remark = (remarks != null && i < remarks.length) ? remarks[i] : "";

                pstmt.setDate(1, saveDate);
                pstmt.setString(2, section);
                pstmt.setString(3, taskName);
                pstmt.setString(4, status1);
                pstmt.setString(5, status2);
                pstmt.setString(6, status3);
                pstmt.setString(7, doneBy);
                pstmt.setString(8, review);
                pstmt.setString(9, remark);
                pstmt.setString(10, status1);
                pstmt.setString(11, status2);
                pstmt.setString(12, status3);
                pstmt.setString(13, doneBy);
                pstmt.setString(14, review);
                pstmt.setString(15, remark);
                pstmt.executeUpdate();
            }
            pstmt.close();
            conn.commit();
            message = "SUCCESS: Data saved successfully!";
        } catch (Exception e) {
            try { conn.rollback(); } catch (Exception ex) {}
            e.printStackTrace();
            message = "ERROR: " + e.getMessage();
        } finally {
            try { conn.setAutoCommit(true); } catch (Exception e) {}
        }
    } else {
        message = "Cannot save: View is read-only or no data.";
    }
}

    function showToast(msg, type) {
        const toastEl = document.getElementById('liveToast');

        toastBody.textContent = msg;
        toastEl.className = 'toast align-items-center text-white border-0';
        if (type === 'success') toastEl.classList.add('bg-success');
        else if (type === 'error') toastEl.classList.add('bg-danger');
        else toastEl.classList.add('bg-warning', 'text-dark');
        if (!toastInstance) toastInstance = new bootstrap.Toast(toastEl, { autohide: true, delay: 4000 });
        toastInstance.show();
    }

    function saveAll() {
        if (saveInProgress) return;
        if ($('button.btn-success').prop('disabled')) {
            showToast('Cannot save: View is read-only.', 'warning');
            return;
        }
        saveInProgress = true;
        $('#spinnerOverlay').fadeIn(200);

        var formData = new FormData($('#checklistForm')[0]);
        var formData2 = new FormData($('#checklistForm2')[0]);
        for (var pair of formData2.entries()) formData.append(pair[0], pair[1]);
        formData.set('action', 'save');

        var hasData = false;
        for (var pair of formData.entries()) {
            if (pair[0] === 'section[]' || pair[0] === 'task_name[]') { hasData = true; break; }
        }
        if (!hasData) {
            showToast('No data to save.', 'warning');
            saveInProgress = false;
            $('#spinnerOverlay').fadeOut(200);
            return;
        }

        $.ajax({
            url: window.location.pathname,
            type: 'POST',
            data: formData,
            processData: false,
            contentType: false,
            success: function() {
                showToast('Data saved successfully!', 'success');
                $('#lastSaved').text('Last saved: ' + new Date().toLocaleTimeString());
                setTimeout(function(){ location.reload(); }, 1000);
            },
            error: function(xhr, status, error) { showToast('Error saving: ' + error, 'error'); },
            complete: function() { saveInProgress = false; $('#spinnerOverlay').fadeOut(300); }
        });
    }
