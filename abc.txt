// ===== FAKE DATA =====
const slots = [
    { name: '10PM-11PM', endTime: '11PM', freezeOffset: 45 },
    { name: '11PM-12AM', endTime: '12AM', freezeOffset: 45 },
    { name: '12AM-1AM', endTime: '1AM', freezeOffset: 45 },
    { name: '1AM-2AM', endTime: '2AM', freezeOffset: 45 },
    { name: '2AM-3AM', endTime: '3AM', freezeOffset: 45 },
    { name: '3AM-4AM', endTime: '4AM', freezeOffset: 45 },
    { name: '4AM-5AM', endTime: '5AM', freezeOffset: 45 },
    { name: '5AM-6AM', endTime: '6AM', freezeOffset: 45 },
    { name: '6AM-7AM', endTime: '7AM', freezeOffset: 45 }
];

// ===== FIXED COMPUTE DEADLINE FUNCTION =====
function computeDeadline(taskName, reportDate) {
    const parts = taskName.split('-');
    if (parts.length !== 2) return null;
    let endStr = parts[1].trim();
    
    // Parse time (e.g., "11PM" -> 23:00, "12AM" -> 00:00)
    let hour = 0, minute = 0;
    let isPM = endStr.includes('PM');
    let isAM = endStr.includes('AM');
    let timeStr = endStr.replace(/[AP]M/g, '').trim();
    
    if (timeStr === '12') {
        hour = isPM ? 12 : 0;
    } else {
        hour = parseInt(timeStr);
        if (isPM && hour !== 12) hour += 12;
        if (isAM && hour === 12) hour = 0;
    }
    
    // Create deadline date
    let deadline = new Date(reportDate);
    
    // FIX: Only add +1 day if end time is 12AM (midnight)
    if (hour === 0 && minute === 0) {
        deadline.setDate(deadline.getDate() + 1);
    }
    
    deadline.setHours(hour, minute, 0, 0);
    deadline.setMinutes(deadline.getMinutes() + 45); // Add 45 min freeze
    
    return deadline;
}

// ===== CHECK SLOT STATUS =====
function checkSlots(currentTime) {
    const reportDate = new Date();
    reportDate.setHours(0, 0, 0, 0);
    
    let results = [];
    let debug = [];
    
    slots.forEach(slot => {
        const deadline = computeDeadline(slot.name, reportDate);
        const isFrozen = currentTime > deadline;
        
        results.push({
            name: slot.name,
            deadline: deadline,
            isFrozen: isFrozen,
            freezeTime: deadline.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })
        });
        
        debug.push(`${slot.name} → Freezes at ${deadline.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })} → ${isFrozen ? '❌ FROZEN' : '✅ EDITABLE'}`);
    });
    
    return { results, debug };
}

// ===== RENDER RESULTS =====
function renderResults(currentTime, testName) {
    const { results, debug } = checkSlots(currentTime);
    
    let html = `<div class="status ${results.some(r => r.isFrozen) ? 'status-frozen' : 'status-editable'}">
        <strong>⏰ Current Time:</strong> ${currentTime.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })}
        ${results.some(r => r.isFrozen) ? ' | 🔒 Some slots are FROZEN' : ' | ✏️ All slots are EDITABLE'}
    </div>`;
    
    results.forEach(r => {
        const cls = r.isFrozen ? 'frozen' : 'editable';
        const icon = r.isFrozen ? '🔒' : '✏️';
        const status = r.isFrozen ? 'FROZEN' : 'EDITABLE';
        html += `<div class="slot ${cls}">
            <strong>${r.name}</strong> → Freezes at ${r.freezeTime} → ${icon} ${status}
        </div>`;
    });
    
    document.getElementById('slotStatus').innerHTML = html;
    document.getElementById('debugInfo').textContent = `🧪 Test: ${testName}\n${'='.repeat(50)}\n${debug.join('\n')}`;
}
