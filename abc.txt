    document.addEventListener('DOMContentLoaded', function() {
        var tooltip = document.getElementById('customTooltip');
        var tiles = document.querySelectorAll('.tile[data-tip]');
        var timeoutId = null;
        
        tiles.forEach(function(tile) {
            tile.addEventListener('mouseenter', function(e) {
                var text = this.getAttribute('data-tip');
                var tipClass = this.getAttribute('data-tip-class') || '';
                
                tooltip.textContent = text;
                tooltip.className = tipClass;
                tooltip.style.display = 'block';
                
                var rect = this.getBoundingClientRect();
                var x = rect.left + rect.width / 2;
                var y = rect.top - 10;
                
                tooltip.style.left = x + 'px';
                tooltip.style.top = y + 'px';
                tooltip.style.transform = 'translateX(-50%) translateY(-100%)';
                
                if (timeoutId) {
                    clearTimeout(timeoutId);
                    timeoutId = null;
                }
            });
            
            tile.addEventListener('mousemove', function(e) {
                var x = e.clientX;
                var y = e.clientY - 10;
                tooltip.style.left = x + 'px';
                tooltip.style.top = y + 'px';
                tooltip.style.transform = 'translateX(-50%) translateY(-100%)';
            });
            
            tile.addEventListener('mouseleave', function() {
                timeoutId = setTimeout(function() {
                    tooltip.style.display = 'none';
                    timeoutId = null;
                }, 100);
            });
        });
    });

    function countdownTimer(el, minutes, seconds) {
        var element = document.getElementById(el);
        if(!element) return;
        var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
        function update() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var t = new Date(msLeft);
                var m = t.getUTCMinutes();
                var s = t.getUTCSeconds();
                element.innerHTML = (m ? m + ':' : '') + (m ? String(s).padStart(2,'0') : s);
                setTimeout(update, 100);
            }
        }
        update();
    }
