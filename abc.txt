public String escapeJson(String str) {
    if (str == null) return "";
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < str.length(); i++) {
        char c = str.charAt(i);
        switch (c) {
            case '"': sb.append("\\\""); break;
            case '\\': sb.append("\\\\"); break;
            case '/': sb.append("\\/"); break;
            case '\b': sb.append("\\b"); break;
            case '\f': sb.append("\\f"); break;
            case '\n': sb.append("\\n"); break;
            case '\r': sb.append("\\r"); break;
            case '\t': sb.append("\\t"); break;
            default: sb.append(c); break;
        }
    }
    return sb.toString();
}

<script>
var currentCategory = '';
var queryData = {};

function msg(text, type) {
    var d = document.getElementById('msgDiv');
    d.innerHTML = '<i class="fas fa-' + (type === 'ok' ? 'check-circle' : 'exclamation-circle') + '"></i> ' + text;
    d.className = 'msg ' + type + ' show';
    clearTimeout(d._t);
    d._t = setTimeout(function() { d.classList.remove('show'); }, 3000);
}

function esc(str) {
    if (!str) return '';
    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
}

function updateStats(cats) {
    var total = 0;
    if (cats) { for (var i = 0; i < cats.length; i++) total += cats[i].query_count || 0; }
    document.getElementById('catCount').textContent = cats ? cats.length : 0;
    document.getElementById('qCount').textContent = total;
    document.getElementById('globalStats').textContent = (cats ? cats.length : 0) + ' · ' + total;
}

function loadCats() {
    fetch('?action=fetchCategories')
    .then(function(r) { return r.json(); })
    .then(function(cats) {
        var container = document.getElementById('tabsContainer');
        var select = document.getElementById('catSelect');
        updateStats(cats);

        if (!cats || cats.length === 0) {
            container.innerHTML = '<span style="color:var(--text3);font-size:0.7rem;">No categories</span>';
            select.innerHTML = '<option value="">-- No Categories --</option>';
            document.getElementById('queriesContainer').innerHTML =
                '<div class="empty"><i class="fas fa-folder-open"></i><h3>No Categories</h3><p>Create your first category.</p></div>';
            return;
        }

        var tabs = '';
        var opts = '<option value="">-- Select --</option>';

        for (var i = 0; i < cats.length; i++) {
            var c = cats[i];
            var active = (i === 0 && !currentCategory) ? 'active' : (currentCategory === c.category_name ? 'active' : '');
            var name = c.category_name.replace(/'/g, "\\'");
            var desc = esc(c.category_desc || '').replace(/'/g, "\\'");

            tabs += '<button class="tab ' + active + '" data-cat="' + name + '" onclick="selectCat(\'' + name + '\')">' +
                '<i class="fas fa-folder"></i> ' + esc(c.category_name) +
                '<span class="cnt">' + c.query_count + '</span>' +
                '<span class="act">' +
                '<i class="fas fa-edit e" onclick="event.stopPropagation();openEditCat(\'' + name + '\',\'' + desc + '\')"></i>' +
                '<i class="fas fa-trash d" onclick="event.stopPropagation();delCat(\'' + name + '\')"></i>' +
                '</span>' +
                '</button>';

            opts += '<option value="' + name + '">' + esc(c.category_name) + ' (' + c.query_count + ')</option>';
        }

        container.innerHTML = tabs;
        select.innerHTML = opts;

        if (cats.length > 0 && !currentCategory) {
            selectCat(cats[0].category_name);
        } else if (currentCategory) {
            selectCat(currentCategory);
        }
    })
    .catch(function(e) { console.error(e); msg('Error loading categories', 'err'); });
}

function selectCat(name) {
    currentCategory = name;

    var tabs = document.querySelectorAll('.tab');
    for (var i = 0; i < tabs.length; i++) {
        if (tabs[i].getAttribute('data-cat') === name) {
            tabs[i].classList.add('active');
        } else {
            tabs[i].classList.remove('active');
        }
    }

    fetch('?action=getQueries&category=' + encodeURIComponent(name))
    .then(function(r) { return r.json(); })
    .then(function(qs) {
        var container = document.getElementById('queriesContainer');
        if (!qs || qs.length === 0) {
            container.innerHTML =
                '<div class="empty"><i class="fas fa-code"></i><h3>No Queries</h3><p>Add a query to "' + esc(name) + '".</p></div>';
            return;
        }

        var html = '';
        for (var i = 0; i < qs.length; i++) {
            var q = qs[i];
            var sql = q.sql_text.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/"/g, '&quot;');
            var sqlEscaped = esc(q.sql_text);
            var title = esc(q.query_title || 'Untitled');
            var desc = esc(q.query_description || '');
            var qid = q.query_id;

            html += '<div class="q" data-id="' + qid + '" data-sql="' + sql + '">' +
                '<div class="q-top">' +
                '<div class="q-title">' +
                '<i class="fas fa-terminal"></i> ' + title +
                '<span class="id">#' + qid + '</span>' +
                '</div>' +
                '</div>' +
                '<div class="q-desc">' + desc + '</div>' +
                '<div class="q-sql">' +
                '<pre>' + sqlEscaped + '</pre>' +
                '<button class="cpy" onclick="copySql(this, ' + qid + ')"><i class="fas fa-copy"></i> Copy</button>' +
                '</div>' +
                '<div class="q-bottom">' +
                '<div class="q-meta"><span><i class="far fa-calendar"></i> ' + (q.created_date || 'Unknown') + '</span></div>' +
                '<div class="q-btns">' +
                '<button class="e" onclick="openEditQ(' + qid + ')"><i class="fas fa-edit"></i> Edit</button>' +
                '<button class="d" onclick="delQ(' + qid + ')"><i class="fas fa-trash"></i> Delete</button>' +
                '</div>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;

        // Store query data for later use
        for (var j = 0; j < qs.length; j++) {
            queryData[qs[j].query_id] = qs[j];
        }
    })
    .catch(function(e) { console.error(e); msg('Error loading queries', 'err'); });
}

function copySql(btn, id) {
    var q = queryData[id];
    if (!q) { msg('Query not found', 'err'); return; }
    navigator.clipboard.writeText(q.sql_text).then(function() {
        var orig = btn.innerHTML;
        btn.innerHTML = '<i class="fas fa-check"></i> Copied!';
        btn.classList.add('done');
        setTimeout(function() {
            btn.innerHTML = orig;
            btn.classList.remove('done');
        }, 1500);
    }).catch(function() { alert('Failed to copy'); });
}

function createCat() {
    var name = document.getElementById('newCatName').value.trim();
    var desc = document.getElementById('newCatDesc').value.trim();
    if (!name) { msg('Please enter a category name', 'err'); return; }

    var fd = new FormData();
    fd.append('action', 'createCategory');
    fd.append('categoryName', name);
    fd.append('categoryDesc', desc);

    fetch('', { method: 'POST', body: fd })
    .then(function(r) { return r.json(); })
    .then(function(res) {
        if (res.success) {
            msg(res.message, 'ok');
            document.getElementById('newCatName').value = '';
            document.getElementById('newCatDesc').value = '';
            loadCats();
        } else {
            msg(res.message, 'err');
        }
    })
    .catch(function() { msg('Error creating category', 'err'); });
}

function openEditCat(name, desc) {
    document.getElementById('editCatOld').value = name;
    document.getElementById('editCatName').value = name;
    document.getElementById('editCatDesc').value = desc;
    document.getElementById('editCatModal').classList.add('active');
}

function closeEditCat() {
    document.getElementById('editCatModal').classList.remove('active');
}

function saveEditCat() {
    var old = document.getElementById('editCatOld').value;
    var name = document.getElementById('editCatName').value.trim();
    var desc = document.getElementById('editCatDesc').value.trim();
    if (!name) { msg('Category name cannot be empty', 'err'); return; }

    var fd = new FormData();
    fd.append('action', 'updateCategory');
    fd.append('oldCategoryName', old);
    fd.append('newCategoryName', name);
    fd.append('categoryDesc', desc);

    fetch('', { method: 'POST', body: fd })
    .then(function(r) { return r.json(); })
    .then(function(res) {
        if (res.success) {
            msg(res.message, 'ok');
            closeEditCat();
            if (currentCategory === old) currentCategory = name;
            loadCats();
        } else {
            msg(res.message, 'err');
        }
    })
    .catch(function() { msg('Error updating category', 'err'); });
}

function delCat(name) {
    if (!confirm('Delete category "' + name + '" and ALL its queries?')) return;
    var fd = new FormData();
    fd.append('action', 'deleteCategory');
    fd.append('categoryName', name);

    fetch('', { method: 'POST', body: fd })
    .then(function(r) { return r.json(); })
    .then(function(res) {
        if (res.success) {
            msg(res.message, 'ok');
            currentCategory = '';
            loadCats();
        } else {
            msg(res.message, 'err');
        }
    })
    .catch(function() { msg('Error deleting category', 'err'); });
}

function addQuery() {
    var cat = document.getElementById('catSelect').value;
    var title = document.getElementById('qTitle').value.trim();
    var desc = document.getElementById('qDesc').value.trim();
    var sql = document.getElementById('qSql').value.trim();
    if (!cat) { msg('Please select a category', 'err'); return; }
    if (!title) { msg('Please enter a query title', 'err'); return; }
    if (!sql) { msg('Please enter SQL query', 'err'); return; }

    var fd = new FormData();
    fd.append('action', 'addQuery');
    fd.append('categoryName', cat);
    fd.append('queryTitle', title);
    fd.append('queryDesc', desc);
    fd.append('sqlText', sql);

    fetch('', { method: 'POST', body: fd })
    .then(function(r) { return r.json(); })
    .then(function(res) {
        if (res.success) {
            msg(res.message, 'ok');
            document.getElementById('qTitle').value = '';
            document.getElementById('qDesc').value = '';
            document.getElementById('qSql').value = '';
            loadCats();
            if (currentCategory) selectCat(currentCategory);
        } else {
            msg(res.message, 'err');
        }
    })
    .catch(function() { msg('Error adding query', 'err'); });
}

function openEditQ(id) {
    var q = queryData[id];
    if (!q) { msg('Query not found', 'err'); return; }
    document.getElementById('editQId').value = id;
    document.getElementById('editQTitle').value = q.query_title || '';
    document.getElementById('editQDesc').value = q.query_description || '';
    document.getElementById('editQSql').value = q.sql_text || '';
    document.getElementById('editQueryModal').classList.add('active');
}

function closeEditQuery() {
    document.getElementById('editQueryModal').classList.remove('active');
}

function saveEditQuery() {
    var id = document.getElementById('editQId').value;
    var title = document.getElementById('editQTitle').value.trim();
    var desc = document.getElementById('editQDesc').value.trim();
    var sql = document.getElementById('editQSql').value.trim();
    if (!title) { msg('Query title cannot be empty', 'err'); return; }
    if (!sql) { msg('SQL query cannot be empty', 'err'); return; }

    var fd = new FormData();
    fd.append('action', 'updateQuery');
    fd.append('queryId', id);
    fd.append('queryTitle', title);
    fd.append('queryDesc', desc);
    fd.append('sqlText', sql);

    fetch('', { method: 'POST', body: fd })
    .then(function(r) { return r.json(); })
    .then(function(res) {
        if (res.success) {
            msg(res.message, 'ok');
            closeEditQuery();
            loadCats();
            if (currentCategory) selectCat(currentCategory);
        } else {
            msg(res.message, 'err');
        }
    })
    .catch(function() { msg('Error updating query', 'err'); });
}

function delQ(id) {
    if (!confirm('Delete this query?')) return;
    var fd = new FormData();
    fd.append('action', 'deleteQuery');
    fd.append('queryId', id);

    fetch('', { method: 'POST', body: fd })
    .then(function(r) { return r.json(); })
    .then(function(res) {
        if (res.success) {
            msg(res.message, 'ok');
            loadCats();
            if (currentCategory) selectCat(currentCategory);
        } else {
            msg(res.message, 'err');
        }
    })
    .catch(function() { msg('Error deleting query', 'err'); });
}

document.getElementById('createCatBtn').addEventListener('click', createCat);
document.getElementById('addQueryBtn').addEventListener('click', addQuery);
loadCats();
