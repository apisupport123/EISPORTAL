<script>
    var currentCategory = '';
    var isSearchMode = false;

    function escapeHtml(str) {
        if (!str) return '';
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        });
    }

    function fetchCategories() {
        fetch('?action=fetchCategories')
            .then(function(response) { return response.json(); })
            .then(function(categories) {
                renderCategories(categories);
                updateStats(categories);
                if (categories.length > 0 && !currentCategory) {
                    selectCategory(categories[0].category_name);
                }
            })
            .catch(function(error) {
                console.error('Error:', error);
                document.getElementById('categoriesGrid').innerHTML = '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><p>Error loading categories</p></div>';
            });
    }

    function renderCategories(categories) {
        var container = document.getElementById('categoriesGrid');
        
        if (categories.length === 0) {
            container.innerHTML = '<div class="empty-state"><i class="fas fa-folder-open"></i><p>No categories available</p></div>';
            return;
        }

        var html = '';
        for (var i = 0; i < categories.length; i++) {
            var cat = categories[i];
            var activeClass = (currentCategory === cat.category_name && !isSearchMode) ? 'active' : '';
            var escapedName = cat.category_name.replace(/'/g, "\\'");
            html += '<div class="category-card ' + activeClass + '" onclick="selectCategory(\'' + escapedName + '\')">' +
                '<div class="category-name">' +
                '<i class="fas fa-folder"></i> ' + escapeHtml(cat.category_name) +
                '</div>' +
                '<div class="category-desc">' + escapeHtml(cat.category_desc || 'No description') + '</div>' +
                '<div class="category-stats">' +
                '<span class="query-count"><i class="fas fa-code"></i> ' + cat.query_count + ' queries</span>' +
                '<i class="fas fa-chevron-right" style="font-size: 0.7rem; opacity: 0.5;"></i>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
    }

    function selectCategory(categoryName) {
        if (isSearchMode) {
            clearSearch();
        }
        
        currentCategory = categoryName;
        document.getElementById('currentCategoryBadge').innerHTML = '<i class="fas fa-folder"></i> ' + escapeHtml(categoryName);
        
        var cards = document.querySelectorAll('.category-card');
        for (var i = 0; i < cards.length; i++) {
            cards[i].classList.remove('active');
        }
        
        loadQueries(categoryName);
    }

    function loadQueries(categoryName) {
        var container = document.getElementById('queriesContainer');
        container.innerHTML = '<div class="loading"><i class="fas fa-spinner fa-pulse"></i> Loading queries...</div>';
        
        fetch('?action=getQueries&category=' + encodeURIComponent(categoryName))
            .then(function(response) { return response.json(); })
            .then(function(queries) {
                displayQueries(queries, categoryName);
            })
            .catch(function(error) {
                console.error('Error:', error);
                container.innerHTML = '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><p>Error loading queries</p></div>';
            });
    }

    function displayQueries(queries, categoryName) {
        var container = document.getElementById('queriesContainer');
        
        if (queries.length === 0) {
            container.innerHTML = '<div class="empty-state"><i class="fas fa-code"></i><p>No queries found in "' + escapeHtml(categoryName) + '"</p></div>';
            return;
        }

        var html = '';
        for (var i = 0; i < queries.length; i++) {
            var q = queries[i];
            var escapedSql = escapeHtml(q.sql_text).replace(/'/g, "\\'");
            html += '<div class="query-card">' +
                '<div class="query-header">' +
                '<div class="query-title">' +
                '<i class="fas fa-terminal"></i> ' + escapeHtml(q.query_title) +
                '<span class="category-tag"><i class="fas fa-folder"></i> ' + escapeHtml(categoryName) + '</span>' +
                '</div>' +
                '<button class="copy-btn" onclick="copySQL(this, \'' + escapedSql + '\')">' +
                '<i class="fas fa-copy"></i> Copy SQL' +
                '</button>' +
                '</div>' +
                '<div class="query-desc">' + escapeHtml(q.query_description || 'No description') + '</div>' +
                '<div class="query-code"><pre style="margin:0; white-space:pre-wrap;">' + escapeHtml(q.sql_text) + '</pre></div>' +
                '<div class="query-meta">' +
                '<span><i class="far fa-calendar"></i> Added: ' + (q.created_date || 'Unknown date') + '</span>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
    }

    function searchQueries() {
        var searchTerm = document.getElementById('searchInput').value.trim().toLowerCase();
        
        if (!searchTerm) {
            alert('Please enter a search term');
            return;
        }
        
        isSearchMode = true;
        currentCategory = '';
        document.getElementById('currentCategoryBadge').innerHTML = '<i class="fas fa-search"></i> Search Results';
        
        var cards = document.querySelectorAll('.category-card');
        for (var i = 0; i < cards.length; i++) {
            cards[i].classList.remove('active');
        }
        
        fetch('?action=fetchAllQueries')
            .then(function(response) { return response.json(); })
            .then(function(queries) {
                var results = [];
                for (var i = 0; i < queries.length; i++) {
                    var q = queries[i];
                    if (q.query_title.toLowerCase().includes(searchTerm) ||
                        (q.query_description && q.query_description.toLowerCase().includes(searchTerm)) ||
                        q.category_name.toLowerCase().includes(searchTerm) ||
                        q.sql_text.toLowerCase().includes(searchTerm)) {
                        results.push(q);
                    }
                }
                displaySearchResults(results);
                document.getElementById('resultStats').innerHTML = 'ðŸ” Found ' + results.length + ' matching queries';
            })
            .catch(function(error) {
                console.error('Error:', error);
                document.getElementById('queriesContainer').innerHTML = '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><p>Error searching queries</p></div>';
            });
    }

    function displaySearchResults(queries) {
        var container = document.getElementById('queriesContainer');
        
        if (queries.length === 0) {
            container.innerHTML = '<div class="empty-state"><i class="fas fa-search"></i><p>No matching queries found. Try different keywords.</p></div>';
            return;
        }
        
        var html = '';
        for (var i = 0; i < queries.length; i++) {
            var q = queries[i];
            var escapedSql = escapeHtml(q.sql_text).replace(/'/g, "\\'");
            html += '<div class="query-card">' +
                '<div class="query-header">' +
                '<div class="query-title">' +
                '<i class="fas fa-terminal"></i> ' + escapeHtml(q.query_title) +
                '<span class="category-tag"><i class="fas fa-folder"></i> ' + escapeHtml(q.category_name) + '</span>' +
                '</div>' +
                '<button class="copy-btn" onclick="copySQL(this, \'' + escapedSql + '\')">' +
                '<i class="fas fa-copy"></i> Copy SQL' +
                '</button>' +
                '</div>' +
                '<div class="query-desc">' + escapeHtml(q.query_description || 'No description') + '</div>' +
                '<div class="query-code"><pre style="margin:0; white-space:pre-wrap;">' + escapeHtml(q.sql_text) + '</pre></div>' +
                '<div class="query-meta">' +
                '<span><i class="far fa-calendar"></i> Added: ' + (q.created_date || 'Unknown date') + '</span>' +
                '<span><i class="fas fa-folder"></i> Category: ' + escapeHtml(q.category_name) + '</span>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
    }

    function clearSearch() {
        document.getElementById('searchInput').value = '';
        isSearchMode = false;
        currentCategory = '';
        fetchCategories();
        document.getElementById('currentCategoryBadge').innerHTML = 'Select a category to view queries';
    }

    function updateStats(categories) {
        var statsSpan = document.getElementById('resultStats');
        if (categories && categories.length > 0) {
            var totalQueries = 0;
            for (var i = 0; i < categories.length; i++) {
                totalQueries += categories[i].query_count;
            }
            statsSpan.innerHTML = '<i class="fas fa-folder"></i> ' + categories.length + ' categories · <i class="fas fa-code"></i> ' + totalQueries + ' total queries';
        } else {
            statsSpan.innerHTML = 'Loading categories...';
        }
    }

    function copySQL(btn, sql) {
        navigator.clipboard.writeText(sql).then(function() {
            var originalHtml = btn.innerHTML;
            btn.innerHTML = '<i class="fas fa-check"></i> Copied!';
            btn.classList.add('copied');
            setTimeout(function() {
                btn.innerHTML = originalHtml;
                btn.classList.remove('copied');
            }, 1500);
        }).catch(function() {
            alert('Failed to copy. Please copy manually.');
        });
    }

    document.getElementById('searchBtn').addEventListener('click', searchQueries);
    document.getElementById('clearSearchBtn').addEventListener('click', clearSearch);
    document.getElementById('searchInput').addEventListener('keypress', function(e) {
        if (e.key === 'Enter') {
            searchQueries();
        }
    });
    
    fetchCategories();
