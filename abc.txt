// Helper function to escape JSON strings properly
public String escapeJson(String str) {
    if (str == null) return "";
    StringBuilder sb = new StringBuilder();
    for (int i = 0; i < str.length(); i++) {
        char c = str.charAt(i);
        switch (c) {
            case '"': sb.append("\\\""); break;
            case '\\': sb.append("\\\\"); break;
            case '/': sb.append("\\/"); break;
            case '\b': sb.append("\\b"); break;
            case '\f': sb.append("\\f"); break;
            case '\n': sb.append("\\n"); break;
            case '\r': sb.append("\\r"); break;
            case '\t': sb.append("\\t"); break;
            default: sb.append(c); break;
        }
    }
    return sb.toString();
}

var currentCategory = '';

function showMessage(msg, type) {
    var msgDiv = document.getElementById('messageDiv');
    msgDiv.innerHTML = '<i class="fas fa-' + (type === 'success' ? 'check-circle' : 'exclamation-circle') + '"></i> ' + msg;
    msgDiv.className = 'message ' + type + ' show';
    setTimeout(function() { msgDiv.classList.remove('show'); }, 3000);
}

function escapeHtml(str) {
    if (!str) return '';
    return String(str).replace(/[&<>"']/g, function(m) {
        if (m === '&') return '&amp;';
        if (m === '<') return '&lt;';
        if (m === '>') return '&gt;';
        if (m === '"') return '&quot;';
        if (m === "'") return '&#039;';
        return m;
    });
}

function updateStats(categories) {
    var totalQueries = 0;
    if (categories) {
        for (var i = 0; i < categories.length; i++) {
            totalQueries += categories[i].query_count || 0;
        }
    }
    document.getElementById('categoryCount').textContent = categories ? categories.length : 0;
    document.getElementById('queryCount').textContent = totalQueries;
    document.getElementById('globalStats').textContent = (categories ? categories.length : 0) + ' Categories · ' + totalQueries + ' Queries';
}

function fetchCategories() {
    fetch('?action=fetchCategories')
    .then(function(response) { return response.json(); })
    .then(function(categories) {
        var container = document.getElementById('categoriesTabsContainer');
        var select = document.getElementById('queryCategorySelect');
        
        updateStats(categories);
        
        if (categories.length === 0) {
            container.innerHTML = '<span style="color:var(--text-muted); font-size:0.8rem;">No categories yet</span>';
            select.innerHTML = '<option value="">-- No Categories --</option>';
            document.getElementById('queriesListContainer').innerHTML = 
                '<div class="empty-state"><i class="fas fa-folder-open"></i><h3>No Categories</h3><p>Create your first category to get started.</p></div>';
            return;
        }
        
        var tabsHtml = '';
        var selectHtml = '<option value="">-- Select Category --</option>';
        
        for (var i = 0; i < categories.length; i++) {
            var cat = categories[i];
            var activeClass = (i === 0 && !currentCategory) ? 'active' : (currentCategory === cat.category_name ? 'active' : '');
            var escapedName = cat.category_name.replace(/'/g, "\\'");
            var escapedDesc = escapeHtml(cat.category_desc || '').replace(/'/g, "\\'");
            
            tabsHtml += '<button class="cat-tab ' + activeClass + '" data-category="' + escapedName + '" onclick="selectCategory(\'' + escapedName + '\')">' +
                '<i class="fas fa-folder"></i> ' + escapeHtml(cat.category_name) +
                '<span class="badge">' + cat.query_count + '</span>' +
                '<span class="cat-actions">' +
                '<i class="fas fa-edit edit" onclick="event.stopPropagation(); openEditCategoryModal(\'' + escapedName + '\', \'' + escapedDesc + '\')"></i>' +
                '<i class="fas fa-trash delete" onclick="event.stopPropagation(); deleteCategory(\'' + escapedName + '\')"></i>' +
                '</span>' +
                '</button>';
                
            selectHtml += '<option value="' + escapedName + '">' + escapeHtml(cat.category_name) + ' (' + cat.query_count + ')</option>';
        }
        
        container.innerHTML = tabsHtml;
        select.innerHTML = selectHtml;
        
        if (categories.length > 0 && !currentCategory) {
            selectCategory(categories[0].category_name);
        } else if (currentCategory) {
            selectCategory(currentCategory);
        }
    })
    .catch(function(error) {
        console.error('Error:', error);
        showMessage('Error loading categories', 'error');
    });
}

function selectCategory(categoryName) {
    currentCategory = categoryName;
    
    var tabs = document.querySelectorAll('.cat-tab');
    for (var i = 0; i < tabs.length; i++) {
        if (tabs[i].getAttribute('data-category') === categoryName) {
            tabs[i].classList.add('active');
        } else {
            tabs[i].classList.remove('active');
        }
    }
    
    fetch('?action=getQueries&category=' + encodeURIComponent(categoryName))
    .then(function(response) { return response.json(); })
    .then(function(queries) {
        var container = document.getElementById('queriesListContainer');
        if (queries.length === 0) {
            container.innerHTML = 
                '<div class="empty-state"><i class="fas fa-code"></i><h3>No Queries</h3><p>Add a query to "' + escapeHtml(categoryName) + '" using the form.</p></div>';
            return;
        }
        
        var html = '';
        for (var i = 0; i < queries.length; i++) {
            var q = queries[i];
            var escapedSql = q.sql_text.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/"/g, '&quot;');
            var escapedTitle = escapeHtml(q.query_title || 'Untitled').replace(/'/g, "\\'");
            var escapedDesc = escapeHtml(q.query_description || '').replace(/'/g, "\\'");
            
            html += '<div class="query-card">' +
                '<div class="query-card-header">' +
                '<div class="query-card-left">' +
                '<div class="query-title">' +
                '<i class="fas fa-terminal"></i> ' + escapeHtml(q.query_title || 'Untitled') +
                '<span class="query-id">#' + q.query_id + '</span>' +
                '</div>' +
                '<div class="query-desc">' + escapeHtml(q.query_description || 'No description') + '</div>' +
                '</div>' +
                '</div>' +
                '<div class="query-code-wrapper">' +
                '<pre>' + escapeHtml(q.sql_text) + '</pre>' +
                '<button class="copy-btn" onclick="copySQL(this, \'' + escapedSql + '\')"><i class="fas fa-copy"></i> Copy</button>' +
                '</div>' +
                '<div class="query-card-footer">' +
                '<div class="query-meta">' +
                '<span><i class="far fa-calendar"></i> ' + (q.created_date || 'Unknown') + '</span>' +
                '</div>' +
                '<div class="query-card-actions">' +
                '<button class="edit-btn" onclick="openEditQueryModal(' + q.query_id + ', \'' + escapedTitle + '\', \'' + escapedDesc + '\', \'' + escapedSql + '\')"><i class="fas fa-edit"></i> Edit</button>' +
                '<button class="delete-btn" onclick="deleteQuery(' + q.query_id + ')"><i class="fas fa-trash"></i> Delete</button>' +
                '</div>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
    })
    .catch(function(error) {
        console.error('Error:', error);
        showMessage('Error loading queries', 'error');
    });
}

function createCategory() {
    var categoryName = document.getElementById('newCategoryName').value.trim();
    var categoryDesc = document.getElementById('newCategoryDesc').value.trim();
    if (!categoryName) {
        showMessage('Please enter a category name', 'error');
        return;
    }
    
    var formData = new FormData();
    formData.append('action', 'createCategory');
    formData.append('categoryName', categoryName);
    formData.append('categoryDesc', categoryDesc);
    
    fetch('', {
        method: 'POST',
        body: formData
    })
    .then(function(response) { return response.json(); })
    .then(function(result) {
        if (result.success) {
            showMessage(result.message, 'success');
            document.getElementById('newCategoryName').value = '';
            document.getElementById('newCategoryDesc').value = '';
            fetchCategories();
        } else {
            showMessage(result.message, 'error');
        }
    })
    .catch(function(error) { showMessage('Error creating category', 'error'); });
}

function openEditCategoryModal(categoryName, categoryDesc) {
    document.getElementById('editCategoryOldName').value = categoryName;
    document.getElementById('editCategoryName').value = categoryName;
    document.getElementById('editCategoryDesc').value = categoryDesc;
    document.getElementById('editCategoryModal').classList.add('active');
}

function closeEditCategoryModal() {
    document.getElementById('editCategoryModal').classList.remove('active');
}

function saveCategoryEdit() {
    var oldName = document.getElementById('editCategoryOldName').value;
    var newName = document.getElementById('editCategoryName').value.trim();
    var desc = document.getElementById('editCategoryDesc').value.trim();
    if (!newName) {
        showMessage('Category name cannot be empty', 'error');
        return;
    }
    
    var formData = new FormData();
    formData.append('action', 'updateCategory');
    formData.append('oldCategoryName', oldName);
    formData.append('newCategoryName', newName);
    formData.append('categoryDesc', desc);
    
    fetch('', {
        method: 'POST',
        body: formData
    })
    .then(function(response) { return response.json(); })
    .then(function(result) {
        if (result.success) {
            showMessage(result.message, 'success');
            closeEditCategoryModal();
            if (currentCategory === oldName) {
                currentCategory = newName;
            }
            fetchCategories();
        } else {
            showMessage(result.message, 'error');
        }
    })
    .catch(function(error) { showMessage('Error updating category', 'error'); });
}

function deleteCategory(categoryName) {
    if (!confirm('Delete category "' + categoryName + '" and ALL its queries?')) return;
    
    var formData = new FormData();
    formData.append('action', 'deleteCategory');
    formData.append('categoryName', categoryName);
    
    fetch('', {
        method: 'POST',
        body: formData
    })
    .then(function(response) { return response.json(); })
    .then(function(result) {
        if (result.success) {
            showMessage(result.message, 'success');
            currentCategory = '';
            fetchCategories();
        } else {
            showMessage(result.message, 'error');
        }
    })
    .catch(function(error) { showMessage('Error deleting category', 'error'); });
}

function addQuery() {
    var categoryName = document.getElementById('queryCategorySelect').value;
    var queryTitle = document.getElementById('queryTitle').value.trim();
    var queryDesc = document.getElementById('queryDesc').value.trim();
    var sqlText = document.getElementById('querySql').value.trim();
    if (!categoryName) { showMessage('Please select a category', 'error'); return; }
    if (!queryTitle) { showMessage('Please enter a query title', 'error'); return; }
    if (!sqlText) { showMessage('Please enter SQL query', 'error'); return; }
    
    var formData = new FormData();
    formData.append('action', 'addQuery');
    formData.append('categoryName', categoryName);
    formData.append('queryTitle', queryTitle);
    formData.append('queryDesc', queryDesc);
    formData.append('sqlText', sqlText);
    
    fetch('', {
        method: 'POST',
        body: formData
    })
    .then(function(response) { return response.json(); })
    .then(function(result) {
        if (result.success) {
            showMessage(result.message, 'success');
            document.getElementById('queryTitle').value = '';
            document.getElementById('queryDesc').value = '';
            document.getElementById('querySql').value = '';
            fetchCategories();
            if (currentCategory) selectCategory(currentCategory);
        } else {
            showMessage(result.message, 'error');
        }
    })
    .catch(function(error) { showMessage('Error adding query', 'error'); });
}

function copySQL(btn, sql) {
    navigator.clipboard.writeText(sql).then(function() {
        var originalHtml = btn.innerHTML;
        btn.innerHTML = '<i class="fas fa-check"></i> Copied!';
        btn.classList.add('copied');
        setTimeout(function() {
            btn.innerHTML = originalHtml;
            btn.classList.remove('copied');
        }, 1500);
    }).catch(function() { alert('Failed to copy'); });
}
