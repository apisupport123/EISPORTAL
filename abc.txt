    {"02:00 AM TO 04:00 AM", "04:00 AM", "4", "0"},
    {"04:00 AM TO 06:00 AM", "06:00 AM", "6", "0"},
    {"06:00 AM TO 07:00 AM", "07:00 AM", "7", "0"}
};

int[] freezeMinutes = new int[slots.length];
String[] freezeDisplay = new String[slots.length];

for (int i = 0; i < slots.length; i++) {
    int endH = Integer.parseInt(slots[i][2]);
    int endM = Integer.parseInt(slots[i][3]);
    int totalMin = endH * 60 + endM + 45;
    if (endH >= 0 && endH < 12) {
        totalMin += 1440;
    }
    freezeMinutes[i] = totalMin;
    
    int displayH = (totalMin / 60) % 24;
    int displayM = totalMin % 60;
    freezeDisplay[i] = String.format("%02d:%02d", displayH, displayM);
}

int testMinutes = testH * 60 + testM;

// ============================================================
// SIMPLE FREEZE LOGIC FOR TEST MODE
// ============================================================
String[] slotStatuses = new String[slots.length];
String[] slotStatusLabels = new String[slots.length];
String[] slotStatusClasses = new String[slots.length];

for (int i = 0; i < slots.length; i++) {
    boolean isFrozen = false;
    int freezeMin = freezeMinutes[i];
    
    // 1. ALL SLOTS FREEZE AT 7:45 AM (465 minutes)
    if (testMinutes >= 465) {
        isFrozen = true;
    }
    // 2. Individual slot freeze during night shift (12 AM - 8 AM)
    else if (testH >= 0 && testH < 8) {
        int adjustedTestMinutes = testMinutes + 1440;
        if (adjustedTestMinutes >= freezeMin) {
            isFrozen = true;
        }
    }
    // 3. Read-only mode (8 AM - 4 PM)
    else if (testH >= 8 && testH < 16) {
        isFrozen = true;
    }
    // 4. Editable (4 PM to midnight) - default
    
    if (isFrozen) {
        slotStatuses[i] = "frozen";
        slotStatusLabels[i] = "🔒 Frozen";
        slotStatusClasses[i] = "frozen-slot";
    } else {
        slotStatuses[i] = "editable";
        slotStatusLabels[i] = "✏️ Editable";
        slotStatusClasses[i] = "editable-slot";
    }
}

// Global editable status
boolean isEditable = false;

    // ============================================================
    // computeDeadline - Sequential freeze logic
    // ============================================================
    private java.util.Date computeDeadline(String taskName, java.sql.Date reportDate) {

    var saveInProgress = false;
    var infoPanelVisible = true;
    var timerInterval = null;
    var timerSeconds = 1800;
    var refreshCount = 0;

    function showSpinner(message) {
        var overlay = document.getElementById('spinnerOverlay');
        var msgEl = document.getElementById('spinnerMessage');
        if (msgEl && message) {
            msgEl.textContent = message;
        }
        overlay.style.display = 'flex';
        overlay.style.opacity = '0';
        setTimeout(function() {
            overlay.style.opacity = '1';
        }, 50);
    }

    function hideSpinner() {
        var overlay = document.getElementById('spinnerOverlay');
        overlay.style.opacity = '0';
        setTimeout(function() {
            overlay.style.display = 'none';
        }, 300);
    }

    function toggleInfoPanel() {
        var body = document.getElementById('infoPanelBody');
        var icon = document.getElementById('infoToggleIcon');
        
        if (infoPanelVisible) {
            body.classList.add('hidden');
            icon.className = 'fas fa-chevron-down toggle-icon';
            infoPanelVisible = false;
        } else {
            body.classList.remove('hidden');
            icon.className = 'fas fa-chevron-up toggle-icon';
            infoPanelVisible = true;
        }
    }

    function updateDoneBy(selectElem, rowId) {
        var hiddenField = document.getElementById('done_by_' + rowId);
        if (hiddenField) {
            hiddenField.value = selectElem.value;
        }
        var tr = document.getElementById('row-' + rowId);
        if (!tr) return;
        var indicator = document.getElementById('row-' + rowId + '-indicator');
        if (selectElem.value && selectElem.value.trim() !== '') {
            tr.classList.add('row-highlight');
            if (indicator) indicator.style.display = 'inline';
            selectElem.style.border = '1.5px solid #10b981';
            selectElem.style.backgroundColor = '#f0fdf4';
        } else {
            tr.classList.remove('row-highlight');
            if (indicator) indicator.style.display = 'none';
            selectElem.style.border = '2px solid #ef4444';
            selectElem.style.backgroundColor = '#fef2f2';
        }
        checkDoneByStatus();
    }

    function checkDoneByStatus() {
        var allFilled = true;
        var emptyCount = 0;
        var allSelects = document.querySelectorAll('select.done-by-select');
        allSelects.forEach(function(select) {
            if (select.value && select.value.trim() !== '') {
                select.style.border = '1.5px solid #10b981';
                select.style.backgroundColor = '#f0fdf4';
            } else {
                select.style.border = '2px solid #ef4444';
                select.style.backgroundColor = '#fef2f2';
                allFilled = false;
                emptyCount++;
            }
        });
        return { allFilled: allFilled, emptyCount: emptyCount };
    }

    function startTimer() {
        timerSeconds = 1800;
        refreshCount = 0;
        updateTimerDisplay();
        
        if (timerInterval) clearInterval(timerInterval);
        
        timerInterval = setInterval(function() {
            timerSeconds--;
            updateTimerDisplay();
            
            if (timerSeconds <= 0) {
                clearInterval(timerInterval);
                refreshCount++;
                document.getElementById('refreshCount').textContent = '(' + refreshCount + ')';
                
                var refreshStatus = document.getElementById('refreshStatus');
                if (refreshStatus) refreshStatus.innerHTML = '<i class="fas fa-sync-alt fa-spin"></i> Refreshing...';
                
                setTimeout(function() { location.reload(); }, 1000);
            }
        }, 1000);
    }

    function updateTimerDisplay() {
        var minutes = Math.floor(timerSeconds / 60);
        var seconds = timerSeconds % 60;
        var timeStr = String(minutes).padStart(2, '0') + ':' + String(seconds).padStart(2, '0');
        
        var timerDisplay = document.getElementById('timerDisplay');
        var footerTimer = document.getElementById('footerTimer');
        
        if (timerDisplay) timerDisplay.textContent = timeStr;
        if (footerTimer) footerTimer.textContent = timeStr;
    }

    function saveAll() {
        if (saveInProgress) return;
        if ($('button.btn-save').prop('disabled')) {
            alert('⚠️ Cannot save: View is read-only.');
            return;
        }

        var doneByFields = document.getElementsByName('done_by[]');
        var allFilled = true;
        var emptyCount = 0;
        var firstEmptySelect = null;
        
        for (var i = 0; i < doneByFields.length; i++) {
            if (!doneByFields[i].value || doneByFields[i].value.trim() === '') {
                allFilled = false;
                emptyCount++;
            }
        }
        
        if (!allFilled) {
            var selects = document.querySelectorAll('select.done-by-select');
            for (var j = 0; j < selects.length; j++) {
                if (!selects[j].value || selects[j].value.trim() === '') {
                    selects[j].style.border = '3px solid #ef4444';
                    selects[j].style.backgroundColor = '#fef2f2';
                    if (!firstEmptySelect) firstEmptySelect = selects[j];
                }
            }
            
            if (firstEmptySelect) {
                firstEmptySelect.focus();
                $('html, body').animate({ scrollTop: $(firstEmptySelect).offset().top - 150 }, 300);
            }
            
            alert('⚠️ All "Done By" fields are mandatory!\n\n' + emptyCount + ' row(s) are empty.');
            return;
        }

        if (!confirm('✅ Are you sure you want to save the changes?')) return;

        saveInProgress = true;
        showSpinner('Saving data...');

        var formData = new FormData($('#checklistForm')[0]);

        $.ajax({
            url: window.location.pathname,
            type: 'POST',
            data: formData,
            processData: false,
            contentType: false,
            success: function(response) {
                alert('✅ Data saved successfully!');
                hideSpinner();
                setTimeout(function() {
                    location.reload();
                }, 500);
            },
            error: function(xhr, status, error) { 
                alert('❌ Error saving: ' + error);
                hideSpinner();
                saveInProgress = false;
            },
            complete: function() { 
                saveInProgress = false; 
            }
        });
    }

    function exportExcel() {
        showSpinner('Generating Excel...');
        
        var viewDate = document.getElementById('viewDate').value;
        var url = window.location.pathname + '?exportData=true&viewDate=' + viewDate;
        
        fetch(url)
            .then(function(response) {
                if (!response.ok) throw new Error('Network response was not ok');
                return response.json();
            })
            .then(function(data) {
                if (!data || (!data.timeSlots && !data.nightReports)) {
                    alert('⚠️ No data available to export!');
                    hideSpinner();
                    return;
                }
                exportWithStyledHTML(data);
                hideSpinner();
            })
            .catch(function(error) {
                console.error('Export error:', error);
                alert('❌ Error exporting to Excel: ' + error.message);
                hideSpinner();
            });
    }

    function getStatusBadge(status) {
        if (!status) return '-';
        var statusUpper = status.toUpperCase().trim();
        var className = '';
        if (statusUpper === 'DONE') className = 'status-done';
        else if (statusUpper === 'PENDING') className = 'status-pending';
        else if (statusUpper === 'NA') className = 'status-na';
        else if (statusUpper === 'IN PROGRESS') className = 'status-inprogress';
        else if (statusUpper === 'HANDOVER') className = 'status-handover';
        else if (statusUpper === 'FAILED') className = 'status-failed';
        else if (statusUpper === 'REVIEW') className = 'status-review';
        else return status;
        return '<span class="' + className + '">' + status + '</span>';
    }
