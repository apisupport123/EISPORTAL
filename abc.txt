// ===== COUNTDOWN =====
function countdown(elementName, minutes, seconds) {
    var element = document.getElementById(elementName);
    if(!element) return;
    var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
    function updateTimer() {
        var msLeft = endTime - (+new Date);
        if(msLeft < 1000) {
            element.innerHTML = "Refreshing...";
        } else {
            var time = new Date(msLeft);
            var h = time.getUTCHours();
            var m = time.getUTCMinutes();
            var s = time.getUTCSeconds();
            element.innerHTML = (h ? h + ':' + (m<10?'0':'') + m : m) + ':' + (s<10?'0':'') + s;
            setTimeout(updateTimer, 1000);
        }
    }
    updateTimer();
}

// ===== SET DEFAULT TIME FILTER =====
function setDefaultTimeFilter() {
    var now = new Date();
    var tenMinsAgo = new Date(now.getTime() - 10 * 60 * 1000);
    function fmt(d) {
        var y = d.getFullYear();
        var mo = String(d.getMonth()+1).padStart(2,'0');
        var day = String(d.getDate()).padStart(2,'0');
        var h = String(d.getHours()).padStart(2,'0');
        var mi = String(d.getMinutes()).padStart(2,'0');
        return y + '-' + mo + '-' + day + 'T' + h + ':' + mi;
    }
    var fromEl = document.getElementById('fromDate');
    var toEl = document.getElementById('toDate');
    if(fromEl && !fromEl.value) fromEl.value = fmt(tenMinsAgo);
    if(toEl && !toEl.value) toEl.value = fmt(now);
}
