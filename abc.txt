    String year = new SimpleDateFormat("yy", Locale.US).format(cal.getTime());
    return month + "_" + year;
}

                    <i class="fas fa-spinner fa-spin" style="font-size:1.5rem;display:block;margin-bottom:6px;"></i>
                    <p>Loading source data...</p>
                </div>
            </div>
        </div>

        <!-- ACTIVITY -->
        <div class="activity-panel">
            <div class="panel-head">
                <h2><i class="fas fa-bolt" style="color:var(--cyan);"></i> Live Activity</h2>
                <span style="font-size:0.45rem;color:var(--text3);"><i class="fas fa-circle" style="color:var(--green);font-size:0.25rem;"></i> Real-time</span>
            </div>
            <div class="activity-list" id="activityList">
                <div class="activity-empty"><i class="fas fa-inbox"></i> Waiting for activity...</div>
            </div>
        </div>

    </div>
</div>

<script>
    var sourceData = {};
    var activityLog = [];
    var MAX_ACTIVITY = 30;
    var refreshInterval;
    var isFirstLoad = true;

    function getPartition() {
        var months = ['JAN','FEB','MAR','APR','MAY','JUN','JUL','AUG','SEP','OCT','NOV','DEC'];
        var now = new Date();
        return months[now.getMonth()] + '_' + String(now.getFullYear()).slice(-2);
    }

    function addActivity(source, status) {
        var now = new Date();
        var t = now.getHours().toString().padStart(2,'0') + ':' + 
                now.getMinutes().toString().padStart(2,'0') + ':' + 
                now.getSeconds().toString().padStart(2,'0');
        activityLog.unshift({ time: t, source: source, status: status });
        if (activityLog.length > MAX_ACTIVITY) activityLog.pop();
        renderActivity();
    }

    function renderActivity() {
        var c = document.getElementById('activityList');
        if (activityLog.length === 0) {
            c.innerHTML = '<div class="activity-empty"><i class="fas fa-inbox"></i> Waiting for activity...</div>';
            return;
        }
        var html = '';
        for (var i = 0; i < Math.min(activityLog.length, 20); i++) {
            var item = activityLog[i];
            var cls = item.status === 'Y' ? 'on' : 'off';
            var txt = item.status === 'Y' ? 'ON' : 'OFF';
            html += '<div class="activity-item ' + cls + '">' +
                '<span class="activity-time">' + item.time + '</span>' +
                '<span class="activity-source">' + item.source + '</span>' +
                '<span class="activity-status ' + cls + '">' + txt + '</span>' +
                '</div>';
        }
        c.innerHTML = html;
    }

    function renderSources(data) {
        var c = document.getElementById('sourcesGrid');
        if (!data || data.length === 0) {
            c.innerHTML = '<div style="text-align:center;padding:30px;color:var(--text3);grid-column:1/-1;"><i class="fas fa-database" style="font-size:1.5rem;display:block;margin-bottom:6px;"></i><p>No data available</p></div>';
            return;
        }
        var html = '', active = 0;
        for (var i = 0; i < data.length; i++) {
            var item = data[i];
            var isActive = item.flag === 'Y';
            if (isActive) active++;
            var cls = isActive ? 'active' : 'inactive';
            var status = isActive ? 'ONLINE' : 'OFFLINE';
            
            html += '<div class="source-card ' + cls + '">' +
                '<span class="status-dot"></span>' +
                (isActive ? '<span class="pulse-ring"></span>' : '') +
                '<div class="source-id">' + item.source_id + '</div>' +
                '<div class="source-status">' + status + '</div>' +
                '<span class="source-hits">' + (isActive ? '⚡ ACTIVE' : '⏸️ Idle') + '</span>' +
                '</div>';
        }
        c.innerHTML = html;
        document.getElementById('sourceCount').textContent = active + '/' + data.length;
        return active;
    }

    function updateStats(data) {
        var active = 0, inactive = 0;
        for (var i = 0; i < data.length; i++) {
            if (data[i].flag === 'Y') active++; else inactive++;
        }
        document.getElementById('activeCount').textContent = active;
        document.getElementById('inactiveCount').textContent = inactive;
        document.getElementById('totalSources').textContent = data.length;
        var uptime = data.length > 0 ? Math.round((active / data.length) * 100) : 0;
        document.getElementById('uptimePercent').textContent = uptime + '%';
    }

    function updateActivity(data) {
        var current = {};
        for (var i = 0; i < data.length; i++) {
            if (data[i].flag === 'Y') current[data[i].source_id] = true;
        }
        for (var s in current) {
            if (!sourceData[s]) { addActivity(s, 'Y'); }
            else if (sourceData[s] !== current[s]) { addActivity(s, current[s] ? 'Y' : 'N'); }
        }
        for (var old in sourceData) {
            if (sourceData[old] && !current[old]) { addActivity(old, 'N'); }
        }
        sourceData = current;
    }

    function updateLoadingProgress(progress, status) {
        document.getElementById('loadingProgress').style.width = progress + '%';
        document.getElementById('loadingStatus').textContent = status;
    }

    function fetchData() {
        var startTime = Date.now();
        updateLoadingProgress(10, 'Connecting to database...');
        
        fetch('?action=fetchSourceStatus')
            .then(function(r) { 
                updateLoadingProgress(90, 'Processing data...');
                return r.json(); 
            })
            .then(function(data) {
                var elapsed = Math.round((Date.now() - startTime) / 1000);
                document.getElementById('updateTime').textContent = 'Updated ' + elapsed + 's ago';
                document.getElementById('partitionBadge').textContent = '📁 ' + getPartition();
                
                renderSources(data);
                updateStats(data);
                updateActivity(data);
                
                updateLoadingProgress(100, '✅ Data loaded successfully!');
                setTimeout(function() {
                    document.getElementById('loadingOverlay').classList.add('hide');
                }, 500);
                
                isFirstLoad = false;
            })
            .catch(function(e) {
                console.error(e);
                updateLoadingProgress(0, '❌ Error: ' + e.message);
                if (isFirstLoad) {
                    document.getElementById('sourcesGrid').innerHTML = 
                        '<div style="text-align:center;padding:30px;color:var(--red);grid-column:1/-1;">' +
                        '<i class="fas fa-exclamation-triangle" style="font-size:1.5rem;display:block;margin-bottom:6px;"></i>' +
                        '<p style="font-weight:600;">Connection Error</p>' +
                        '<p style="font-size:0.65rem;color:var(--text3);">' + e.message + '</p>' +
                        '<p style="font-size:0.55rem;color:var(--text3);margin-top:4px;">Please refresh the page</p></div>';
                    setTimeout(function() {
                        document.getElementById('loadingOverlay').classList.add('hide');
                    }, 2000);
                }
            });
    }

    function manualRefresh() {
        var btn = document.getElementById('refreshBtn');
        btn.disabled = true;
        btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Loading...';
        
        document.getElementById('loadingOverlay').classList.remove('hide');
        document.getElementById('loadingProgress').style.width = '0%';
        document.getElementById('loadingStatus').textContent = '🔄 Refreshing data...';
        
        fetchData();
        
        setTimeout(function() {
            btn.disabled = false;
            btn.innerHTML = '<i class="fas fa-sync-alt"></i> Refresh';
        }, 1000);
    }

    function init() {
        document.getElementById('partitionBadge').textContent = '📁 ' + getPartition();
        fetchData();
        
        // Refresh every 5 minutes
        refreshInterval = setInterval(fetchData, 300000);
    }

    window.addEventListener('beforeunload', function() {
        if (refreshInterval) clearInterval(refreshInterval);
    });

    init();
</script>
</body>
</html>
