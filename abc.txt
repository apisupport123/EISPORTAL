// ============================================================
boolean isEditable = false;

// 1. Check if it's a past date (before today)
boolean isPastDate = displayDate.before(todaySQL);

// 2. If past date → NOT editable
if (isPastDate) {
    isEditable = false;
}
// 3. If today or future date → check time
else {
    // If it's today's date (night shift starts at 10 PM, but edit opens at 4 PM)
    if (displayDate.equals(todaySQL)) {
        // After 4 PM → Editable (edit window opens for next day's night shift)
        if (hour >= 16) {
            isEditable = true;
        } else {
            // Before 4 PM → Showing yesterday's night, not editable
            isEditable = false;
        }
    }
    // If it's yesterday's date (being shown before 4 PM)
    else if (displayDate.equals(yesterdaySQL)) {
        // Before 4 PM → Editable (yesterday's night data)
        if (hour < 16) {
            isEditable = true;
        } else {
            isEditable = false;
        }
    }
    // Future dates → Editable (admin can set)
    else if (displayDate.after(todaySQL)) {
        isEditable = true;
    }
}

// For custom view (user selected date)
if (isCustomView) {
    if (displayDate.before(todaySQL)) {
        isEditable = false;
    } else if (displayDate.equals(todaySQL)) {
        isEditable = (hour >= 16);
    } else {
        isEditable = true;

                existingData.put(key, data);
            }
            rs.close();
            pstmt.close();

            String mergeSQL = "MERGE INTO night_checklist t " +
                    "USING (SELECT ? AS report_date, ? AS section, ? AS task_name FROM dual) s " +
                    "ON (t.report_date = s.report_date AND t.section = s.section AND t.task_name = s.task_name) " +
                    "WHEN MATCHED THEN UPDATE SET " +
                    "    status1=?, status2=?, status3=?, done_by=?, review=?, remarks=?, " +
                    "    updated_at = CASE WHEN ? = 1 THEN SYSTIMESTAMP ELSE t.updated_at END " +
                    "WHEN NOT MATCHED THEN INSERT " +
                    "    (report_date, section, task_name, status1, status2, status3, done_by, review, remarks, created_at, updated_at) " +
                    "    VALUES (s.report_date, s.section, s.task_name, ?, ?, ?, ?, ?, ?, SYSTIMESTAMP, SYSTIMESTAMP)";
            pstmt = conn.prepareStatement(mergeSQL);

            for (int i = 0; i < sections.length; i++) {
                String section = sections[i];
                String taskName = taskNames[i];
                String key = section + "||" + taskName;
                String status1 = (status1s != null && i < status1s.length) ? status1s[i] : "PENDING";
                String status2 = (status2s != null && i < status2s.length) ? status2s[i] : "";
                String status3 = (status3s != null && i < status3s.length) ? status3s[i] : "";
                String doneBy = (doneBys != null && i < doneBys.length) ? doneBys[i] : "";
                String review = (reviews != null && i < reviews.length) ? reviews[i] : "";
                String remark = (remarks != null && i < remarks.length) ? remarks[i] : "";

                int dataChanged = 0;
                Map<String, Object> existing = existingData.get(key);
                if (existing == null) {
                    dataChanged = 1;
                } else {
                    String oldStatus1 = (String) existing.get("status1");
                    String oldStatus2 = (String) existing.get("status2");
                    String oldStatus3 = (String) existing.get("status3");
                    String oldDoneBy = (String) existing.get("done_by");
                    String oldReview = (String) existing.get("review");
                    String oldRemark = (String) existing.get("remarks");
                    
                    if (!NVL(oldStatus1,"").equals(NVL(status1,"")) ||
                        !NVL(oldStatus2,"").equals(NVL(status2,"")) ||
                        !NVL(oldStatus3,"").equals(NVL(status3,"")) ||
                        !NVL(oldDoneBy,"").equals(NVL(doneBy,"")) ||
                        !NVL(oldReview,"").equals(NVL(review,"")) ||
                        !NVL(oldRemark,"").equals(NVL(remark,""))) {
                        dataChanged = 1;
                    }
                }

                pstmt.setDate(1, effectiveSaveDate);
                pstmt.setString(2, section);
                pstmt.setString(3, taskName);
                pstmt.setString(4, status1);
                pstmt.setString(5, status2);
                pstmt.setString(6, status3);
                pstmt.setString(7, doneBy);
                pstmt.setString(8, review);
                pstmt.setString(9, remark);
                pstmt.setInt(10, dataChanged);
                pstmt.setString(11, status1);
                pstmt.setString(12, status2);
                pstmt.setString(13, status3);
                pstmt.setString(14, doneBy);
                pstmt.setString(15, review);
                pstmt.setString(16, remark);
                
                pstmt.executeUpdate();
            }
            pstmt.close();
            conn.commit();
            message = "✅ Data saved successfully!";
            showMessage = true;
        } catch (Exception e) {
            try { conn.rollback(); } catch (Exception ex) {}
            e.printStackTrace();
            message = "❌ Error: " + e.getMessage();
            showMessage = true;
        } finally {
            try { conn.setAutoCommit(true); } catch (Exception e) {}
        }
    } else {
        message = "⚠️ Cannot save: View is read-only or no data.";
        showMessage = true;
    }

    function showSpinner(message) {
        var overlay = document.getElementById('spinnerOverlay');
        var msgEl = document.getElementById('spinnerMessage');
        if (msgEl && message) {
            msgEl.textContent = message;
        }
        overlay.style.display = 'flex';
        overlay.style.opacity = '0';
        setTimeout(function() {
            overlay.style.opacity = '1';
        }, 50);
    }

    function hideSpinner() {
        var overlay = document.getElementById('spinnerOverlay');
        overlay.style.opacity = '0';
        setTimeout(function() {
            overlay.style.display = 'none';
        }, 300);
    }

    function toggleInfoPanel() {
        var body = document.getElementById('infoPanelBody');
        var icon = document.getElementById('infoToggleIcon');
        
        if (infoPanelVisible) {
            body.classList.add('hidden');
            icon.className = 'fas fa-chevron-down toggle-icon';
            infoPanelVisible = false;
        } else {
            body.classList.remove('hidden');
            icon.className = 'fas fa-chevron-up toggle-icon';

    function updateDoneBy(selectElem, rowId) {
        var hiddenField = document.getElementById('done_by_' + rowId);
        if (hiddenField) {
            hiddenField.value = selectElem.value;
        }

        if (!tr) return;

        if (selectElem.value && selectElem.value.trim() !== '') {
            tr.classList.add('row-highlight');
            if (indicator) indicator.style.display = 'inline';

    function saveAll() {
        if (saveInProgress) return;
        if ($('button.btn-save').prop('disabled')) {
            alert('⚠️ This view is read-only. You cannot save changes.');
            return;
        }

        var doneByFields = document.getElementsByName('done_by[]');
        var allFilled = true;
        var emptyCount = 0;
        var firstEmptySelect = null;
        
        for (var i = 0; i < doneByFields.length; i++) {
            if (!doneByFields[i].value || doneByFields[i].value.trim() === '') {
                allFilled = false;
                emptyCount++;
            }
        }
        
        if (!allFilled) {
            var selects = document.querySelectorAll('select.done-by-select');
            for (var j = 0; j < selects.length; j++) {
                if (!selects[j].value || selects[j].value.trim() === '') {
                    selects[j].style.border = '3px solid #ef4444';
                    selects[j].style.backgroundColor = '#fef2f2';
                    if (!firstEmptySelect) firstEmptySelect = selects[j];
                }
            }
            
            if (firstEmptySelect) {
                firstEmptySelect.focus();
                $('html, body').animate({ scrollTop: $(firstEmptySelect).offset().top - 150 }, 300);
            }
            
            alert('⚠️ All "Done By" fields are mandatory!\n\n' + emptyCount + ' row(s) are empty.');
            return;
        }

        if (!confirm('✅ Are you sure you want to save the changes?')) return;

        saveInProgress = true;
        showSpinner('Saving data...');

        var formData = new FormData($('#checklistForm')[0]);

        $.ajax({
            url: window.location.pathname,
            type: 'POST',
            data: formData,
            processData: false,
            contentType: false,
            success: function(response) {
                alert('✅ Data saved successfully!');
                hideSpinner();
                setTimeout(function() {
                    location.reload();
                }, 500);
            },
            error: function(xhr, status, error) { 
                alert('❌ Error saving: ' + error);
                hideSpinner();
                saveInProgress = false;
            },
            complete: function() { 
                saveInProgress = false; 
            }
        });
    }

                alert('❌ Error exporting to Excel: ' + error.message);
                hideSpinner();
            });
    }

    function getStatusBadge(status) {
        if (!status) return '-';
        var statusUpper = status.toUpperCase().trim();
        var className = '';
        if (statusUpper === 'DONE') className = 'status-done';
        else if (statusUpper === 'PENDING') className = 'status-pending';
        else if (statusUpper === 'NA') className = 'status-na';
        else if (statusUpper === 'IN PROGRESS') className = 'status-inprogress';
        else if (statusUpper === 'HANDOVER') className = 'status-handover';
        else if (statusUpper === 'FAILED') className = 'status-failed';
        else if (statusUpper === 'REVIEW') className = 'status-review';
        else return status;
        return '<span class="' + className + '">' + status + '</span>';
    }
