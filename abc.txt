    function updateStats() {
        var tiles = document.querySelectorAll('.metric-tile:not(.tile-empty)');
        var total = tiles.length;
        var critical = document.querySelectorAll('.tile-critical').length;
        var healthy = document.querySelectorAll('.tile-healthy').length;
        
        document.getElementById('totalServices').textContent = total || '0';
        document.getElementById('totalCritical').textContent = critical || '0';
        document.getElementById('totalHealthy').textContent = healthy || '0';
    }
    
    function countdownPremium(el, minutes, seconds) {
        var element = document.getElementById(el);
        if(!element) return;
        var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
        function update() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var t = new Date(msLeft);
                var m = t.getUTCMinutes();
                var s = t.getUTCSeconds();
                element.innerHTML = (m ? m + ':' : '') + (m ? String(s).padStart(2,'0') : s);
                setTimeout(update, 100);
            }
        }
        update();
    }
