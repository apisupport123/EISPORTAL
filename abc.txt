WITH FOUND_SOURCES AS(
select  /*+ parallel(04)*/  a.source_id
from eisapp.api_log PARTITION(AUG_26) a where a.source_id in ('AK','CA','CC','CK','CR','FA','FS','I4','IN','KA','MB','NB','ON','RL','SA','SK','SQ','UP','WB','Y2')
--AND a.request_date_time >=SYSDATE - INTERVAL '10' MINUTE
and a.request_date_time between to_date('26-AUG-2026 15:00:00','DD-MON-YYYY HH24:MI:SS') and to_date('26-AUG-2026 16:00:00','DD-MON-YYYY HH24:MI:SS')
    GROUP BY a.source_id
)
SELECT s.source_id,
case when f.source_id is NOT NULL THEN 'Y' ELSE 'N' END AS flag
from (
select 'SA'  as source_id from dual
union all
select 'FS'    from dual
union all
select 'I4'   from dual
union all
select 'Y2'   from dual
union all
select 'KA'  from dual
union all
select 'FA'  from dual
union all
select 'WB'  from dual
union all
select 'ON'  from dual
union all
select 'CC'  from dual
union all
select 'CK'  from dual
union all
select 'CR'  from dual
union all
select 'SQ'  from dual
union all
select 'RL'  from dual
union all
select 'UP'  from dual
union all
select 'CA'  from dual
union all
select 'AK'  from dual
union all
select 'NB'  from dual
union all
select 'IN'  from dual
union all
select 'MB'  from dual
union all
select 'SK'  from dual
) s
LEFT JOIN FOUND_SOURCES f ON f.source_id = s.source_id
ORDER BY s.source_id;
