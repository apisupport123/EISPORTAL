    // ✅ FIXED flag() method - uses local connection with proper closing
    public static void flag() {
        DBConfig.checkAndReload();
       
        synchronized (lock) {
            Connection tempConn = null; // ✅ LOCAL variable - no more leak!
            try {
                Class.forName(DBConfig.getDriver());
               
                // PRIMARY CONNECTION ATTEMPT
                try {
                    tempConn = DriverManager.getConnection(
                        "jdbc:oracle:thin:@10.191.170.109:1787/REPODB_SRV",
                        DBConfig.getEisappsupportUsername(),
                        DBConfig.getPrimaryPassword()
                    );
                    System.out.println("Connected to PRIMARY DB 1");
                } catch(Exception E) {
                    tempConn = DriverManager.getConnection(
                        "jdbc:oracle:thin:@10.191.170.110:1787/REPODB_SRV",
                        DBConfig.getEisappsupportUsername(),
                        DBConfig.getPrimaryPassword()
                    );
                    System.out.println("Connected to PRIMARY DB 2");
                }
               
                // GET FLAG STATUS
                String sql = "select * from connection_master";
                try (Statement smt = tempConn.createStatement();
                     ResultSet rs = smt.executeQuery(sql)) {
                    
                    while (rs.next()) {
                        flagStatus = rs.getString("FLAG");
                        DB_ENVLOCATION = rs.getString("DB_LOCATION");
                        
                        if (flagStatus == null) flagStatus = "PR";
                        if (DB_ENVLOCATION == null) DB_ENVLOCATION = "MI";
                    }
                    
                    if (flagStatus == null || DB_ENVLOCATION == null) {
                        flagStatus = "PR";
                        DB_ENVLOCATION = "MI";
                        System.out.println("No flag data found, using defaults: PR/MI");
                    }
                    
                    System.out.println("Flag Status: " + flagStatus + ", Location: " + DB_ENVLOCATION);
                }
               
            } catch (Exception e) {
                System.out.println("Primary connection failed, trying DR...");
               
                try {
                    // DR CONNECTION ATTEMPT
                    Class.forName(DBConfig.getDriver());
                    try {
                        tempConn = DriverManager.getConnection(
                            "jdbc:oracle:thin:@10.176.0.109:1787/sidashdb_srv",
                            DBConfig.getEisappsupportUsername(),
                            DBConfig.getPrimaryPassword()
                        );
                        System.out.println("Connected to DR DB 1");
                    } catch(Exception Er) {
                        tempConn = DriverManager.getConnection(
                            "jdbc:oracle:thin:@10.176.0.110:1787/sidashdb_srv",
                            DBConfig.getEisappsupportUsername(),
                            DBConfig.getPrimaryPassword()
                        );
                        System.out.println("Connected to DR DB 2");
                    }
                   
                    // GET FLAG STATUS FROM DR
                    String sql = "select * from connection_master";
                    try (Statement smt = tempConn.createStatement();
                         ResultSet rs = smt.executeQuery(sql)) {
                        
                        while (rs.next()) {
                            flagStatus = rs.getString("FLAG");
                            DB_ENVLOCATION = rs.getString("DB_LOCATION");
                            
                            if (flagStatus == null) flagStatus = "DR";
                            if (DB_ENVLOCATION == null) DB_ENVLOCATION = "MI";
                        }
                        
                        if (flagStatus == null || DB_ENVLOCATION == null) {
                            flagStatus = "DR";
                            DB_ENVLOCATION = "MI";
                            System.out.println("No flag data found in DR, using defaults: DR/MI");
                        }
                        
                        System.out.println("Flag Status: " + flagStatus + ", Location: " + DB_ENVLOCATION);
                    }
                   
                } catch(Exception e1) {
                    System.out.println("All connection attempts failed");
                    flagStatus = "PR";
                    DB_ENVLOCATION = "MI";
                    e1.printStackTrace();
                }
            } finally {
                // ✅ CRITICAL FIX: Close the temporary connection
                if (tempConn != null) {
                    try {
                        tempConn.close();
                        System.out.println("Flag check connection closed");
                    } catch (SQLException e) {
                        System.err.println("Error closing flag connection: " + e.getMessage());
                    }
                }
            }
        }
    }

    // ✅ Added utility method to safely close connections (optional)
    public static void closeConnection(Connection conn) {
        if (conn != null) {
            try {
                if (!conn.isClosed()) {
                    conn.close();
                    System.out.println("Connection closed successfully");
                }
            } catch (SQLException e) {
                System.err.println("Error closing connection: " + e.getMessage());
            }
