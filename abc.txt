    var currentCategory = '';
    var isSearchMode = false;
    var queryData = {};

    function esc(str) {
        if (!str) return '';
        return String(str)
            .replace(/&/g, '&amp;')
            .replace(/</g, '&lt;')
            .replace(/>/g, '&gt;')
            .replace(/"/g, '&quot;')
            .replace(/'/g, '&#39;');
    }

    function updateStats(categories) {
        var total = 0;
        if (categories) {
            for (var i = 0; i < categories.length; i++) {
                total += categories[i].query_count || 0;
            }
        }
        var statsSpan = document.getElementById('globalStats');
        var resultSpan = document.getElementById('resultStats');
        if (categories && categories.length > 0) {
            statsSpan.innerHTML = (categories ? categories.length : 0) + ' · ' + total;
            resultSpan.innerHTML = '<i class="fas fa-folder"></i> ' + categories.length + ' categories · <i class="fas fa-code"></i> ' + total + ' queries';
        } else {
            statsSpan.innerHTML = '0 · 0';
            resultSpan.innerHTML = '<i class="fas fa-database"></i> No categories found';
        }
    }

    function fetchCategories() {
        fetch('?action=fetchCategories')
            .then(function(response) { return response.json(); })
            .then(function(categories) {
                renderCategories(categories);
                updateStats(categories);
                if (categories.length > 0 && !currentCategory) {
                    selectCategory(categories[0].category_name);
                }
            })
            .catch(function(error) {
                console.error('Error:', error);
                document.getElementById('categoriesGrid').innerHTML = 
                    '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><h3>Error</h3><p>Failed to load categories</p></div>';
            });
    }

        if (!categories || categories.length === 0) {
            container.innerHTML = 
                '<div class="empty-state"><i class="fas fa-folder-open"></i><h3>No Categories</h3><p>No categories available</p></div>';
            return;
        }

        var html = '';
        for (var i = 0; i < categories.length; i++) {
            var cat = categories[i];
            var activeClass = (currentCategory === cat.category_name && !isSearchMode) ? 'active' : '';
            var name = cat.category_name.replace(/'/g, "\\'");
            html += '<div class="category-card ' + activeClass + '" onclick="selectCategory(\'' + name + '\')">' +
                '<div class="category-name">' +
                '<i class="fas fa-folder"></i> ' + esc(cat.category_name) +
                '</div>' +
                '<div class="category-desc">' + esc(cat.category_desc || 'No description') + '</div>' +
                '<div class="category-stats">' +
                '<span class="query-count"><i class="fas fa-code"></i> ' + cat.query_count + ' queries</span>' +
                '<i class="fas fa-chevron-right" style="font-size: 0.65rem; opacity: 0.4;"></i>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
    }

    function selectCategory(categoryName) {
        if (isSearchMode) {
            clearSearch();
            return;
        }
        
        currentCategory = categoryName;
        document.getElementById('currentCategoryBadge').innerHTML = '<i class="fas fa-folder"></i> ' + esc(categoryName);
        
        var cards = document.querySelectorAll('.category-card');
        for (var i = 0; i < cards.length; i++) {
            cards[i].classList.remove('active');
        }
        // Find and activate the selected card
        var allCards = document.querySelectorAll('.category-card');
        for (var j = 0; j < allCards.length; j++) {
            if (allCards[j].textContent.includes(categoryName)) {
                allCards[j].classList.add('active');
                break;
            }
        }
        
        loadQueries(categoryName);
    }

    function loadQueries(categoryName) {
        var container = document.getElementById('queriesContainer');
        container.innerHTML = '<div class="loading"><i class="fas fa-spinner fa-pulse"></i> Loading queries...</div>';
        
        fetch('?action=getQueries&category=' + encodeURIComponent(categoryName))
            .then(function(response) { return response.json(); })
            .then(function(queries) {
                displayQueries(queries, categoryName);
                // Store query data
                for (var i = 0; i < queries.length; i++) {
                    queryData[queries[i].query_id] = queries[i];
                }
            })
            .catch(function(error) {
                console.error('Error:', error);
                container.innerHTML = 
                    '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><h3>Error</h3><p>Failed to load queries</p></div>';
            });
    }

    function displayQueries(queries, categoryName) {
        var container = document.getElementById('queriesContainer');
        
        if (!queries || queries.length === 0) {
            container.innerHTML = 
                '<div class="empty-state"><i class="fas fa-code"></i><h3>No Queries</h3><p>No queries found in "' + esc(categoryName) + '"</p></div>';
            return;
        }

        var html = '';
        for (var i = 0; i < queries.length; i++) {
            var q = queries[i];
            var sql = q.sql_text.replace(/\\/g, '\\\\').replace(/'/g, "\\'").replace(/"/g, '&quot;');
            var sqlEscaped = esc(q.sql_text);
            
            html += '<div class="query-card" data-id="' + q.query_id + '">' +
                '<div class="query-header">' +
                '<div class="query-title">' +
                '<i class="fas fa-terminal"></i> ' + esc(q.query_title) +
                '<span class="category-tag"><i class="fas fa-folder"></i> ' + esc(categoryName) + '</span>' +
                '</div>' +
                '<button class="copy-btn" onclick="copySQL(this, ' + q.query_id + ')">' +
                '<i class="fas fa-copy"></i> Copy SQL' +
                '</button>' +
                '</div>' +
                '<div class="query-desc">' + esc(q.query_description || 'No description') + '</div>' +
                '<div class="query-code"><pre>' + sqlEscaped + '</pre></div>' +
                '<div class="query-meta">' +
                '<span><i class="far fa-calendar"></i> Added: ' + (q.created_date || 'Unknown') + '</span>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
    }

    function searchQueries() {
        var searchTerm = document.getElementById('searchInput').value.trim().toLowerCase();
        
        if (!searchTerm) {
            alert('Please enter a search term');
            return;
        }
        
        isSearchMode = true;
        currentCategory = '';
        document.getElementById('currentCategoryBadge').innerHTML = '<i class="fas fa-search"></i> Search Results';
        
        var cards = document.querySelectorAll('.category-card');
        for (var i = 0; i < cards.length; i++) {
            cards[i].classList.remove('active');
        }
        
        var container = document.getElementById('queriesContainer');
        container.innerHTML = '<div class="loading"><i class="fas fa-spinner fa-pulse"></i> Searching...</div>';
        
        fetch('?action=fetchAllQueries')
            .then(function(response) { return response.json(); })
            .then(function(queries) {
                var results = [];
                for (var i = 0; i < queries.length; i++) {
                    var q = queries[i];
                    if (q.query_title.toLowerCase().includes(searchTerm) ||
                        (q.query_description && q.query_description.toLowerCase().includes(searchTerm)) ||
                        q.category_name.toLowerCase().includes(searchTerm) ||
                        q.sql_text.toLowerCase().includes(searchTerm)) {
                        results.push(q);
                    }
                }
                displaySearchResults(results);
                document.getElementById('resultStats').innerHTML = '🔍 Found ' + results.length + ' matching queries';
            })
            .catch(function(error) {
                console.error('Error:', error);
                container.innerHTML = 
                    '<div class="empty-state"><i class="fas fa-exclamation-triangle"></i><h3>Error</h3><p>Failed to search queries</p></div>';
            });
    }

    function displaySearchResults(queries) {
        var container = document.getElementById('queriesContainer');
        
        if (!queries || queries.length === 0) {
            container.innerHTML = 
                '<div class="empty-state"><i class="fas fa-search"></i><h3>No Results</h3><p>Try different keywords</p></div>';
            return;
        }
        
        var html = '';
        for (var i = 0; i < queries.length; i++) {
            var q = queries[i];
            var sqlEscaped = esc(q.sql_text);
            
            html += '<div class="query-card" data-id="' + q.query_id + '">' +
                '<div class="query-header">' +
                '<div class="query-title">' +
                '<i class="fas fa-terminal"></i> ' + esc(q.query_title) +
                '<span class="category-tag"><i class="fas fa-folder"></i> ' + esc(q.category_name) + '</span>' +
                '</div>' +
                '<button class="copy-btn" onclick="copySQL(this, ' + q.query_id + ')">' +
                '<i class="fas fa-copy"></i> Copy SQL' +
                '</button>' +
                '</div>' +
                '<div class="query-desc">' + esc(q.query_description || 'No description') + '</div>' +
                '<div class="query-code"><pre>' + sqlEscaped + '</pre></div>' +
                '<div class="query-meta">' +
                '<span><i class="far fa-calendar"></i> Added: ' + (q.created_date || 'Unknown') + '</span>' +
                '<span><i class="fas fa-folder"></i> ' + esc(q.category_name) + '</span>' +
                '</div>' +
                '</div>';
        }
        container.innerHTML = html;
        
        // Store query data
        for (var j = 0; j < queries.length; j++) {
            queryData[queries[j].query_id] = queries[j];
        }
    }

    function clearSearch() {
        document.getElementById('searchInput').value = '';
        isSearchMode = false;
        currentCategory = '';
        document.getElementById('currentCategoryBadge').innerHTML = '<i class="fas fa-folder"></i> Select a category';
        fetchCategories();
        document.getElementById('resultStats').innerHTML = '<i class="fas fa-database"></i> Loading categories...';
    }

    function copySQL(btn, id) {
        var q = queryData[id];
        if (!q) {
            alert('Query data not found');
            return;
        }
        navigator.clipboard.writeText(q.sql_text).then(function() {
            var originalHtml = btn.innerHTML;
            btn.innerHTML = '<i class="fas fa-check"></i> Copied!';
            btn.classList.add('copied');
            setTimeout(function() {
                btn.innerHTML = originalHtml;
                btn.classList.remove('copied');
            }, 1500);
        }).catch(function() {
            alert('Failed to copy. Please copy manually.');
        });
    }

    // Event Listeners
    document.getElementById('searchBtn').addEventListener('click', searchQueries);
    document.getElementById('clearSearchBtn').addEventListener('click', clearSearch);
    document.getElementById('searchInput').addEventListener('keypress', function(e) {
        if (e.key === 'Enter') {
            searchQueries();
        }
    });
    
    // Load
    fetchCategories();
