// ============================================================
// TEST MODE LOGIC - FIXED
// ============================================================
String testMode = request.getParameter("testMode");
String testHour = request.getParameter("testHour");
String testMinute = request.getParameter("testMinute");
String testDate = request.getParameter("testDate");

Calendar nowCal = Calendar.getInstance();
int currentHour = nowCal.get(Calendar.HOUR_OF_DAY);
int currentMinute = nowCal.get(Calendar.MINUTE);
int testH = currentHour;
int testM = currentMinute;
String testTimeDisplay = "Current Time";
boolean isTestMode = false;
String selectedDateDisplay = "";

if ("true".equals(testMode) && testHour != null && testMinute != null) {
    try {
        testH = Integer.parseInt(testHour);
        testM = Integer.parseInt(testMinute);
        isTestMode = true;
        testTimeDisplay = String.format("%02d:%02d", testH, testM);
        
        if (testDate != null && !testDate.isEmpty()) {
            SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd");
            java.util.Date parsed = sdf.parse(testDate);
            selectedDateDisplay = new SimpleDateFormat("dd-MM-yyyy").format(parsed);
        } else {
            selectedDateDisplay = new SimpleDateFormat("dd-MM-yyyy").format(new java.util.Date());
        }
    } catch (Exception e) {}
}

// ============================================================
// FIXED: Compute freeze time using actual logic
// ============================================================
// Slot definitions with their end times
String[][] slots = {
    {"10:00 PM TO 12:00 AM", "12:00 AM", "0", "0"},
    {"12:00 AM TO 02:00 AM", "02:00 AM", "2", "0"},
    {"02:00 AM TO 04:00 AM", "04:00 AM", "4", "0"},
    {"04:00 AM TO 06:00 AM", "06:00 AM", "6", "0"},
    {"06:00 AM TO 07:00 AM", "07:00 AM", "7", "0"}
};

// Calculate freeze time for each slot (end time + 45 minutes)
int[] freezeMinutes = new int[slots.length];
String[] freezeDisplay = new String[slots.length];

for (int i = 0; i < slots.length; i++) {
    int endH = Integer.parseInt(slots[i][2]);
    int endM = Integer.parseInt(slots[i][3]);
    // Add 45 minutes
    int totalMin = endH * 60 + endM + 45;
    // For AM slots (12:00 AM to 7:00 AM), freeze time is next day
    if (endH >= 0 && endH < 12) {
        totalMin += 1440; // Add 24 hours
    }
    freezeMinutes[i] = totalMin;
    
    int displayH = (totalMin / 60) % 24;
    int displayM = totalMin % 60;
    freezeDisplay[i] = String.format("%02d:%02d", displayH, displayM);
}

int testMinutes = testH * 60 + testM;

// FIXED: Calculate slot statuses
String[] slotStatuses = new String[slots.length];
String[] slotStatusLabels = new String[slots.length];
String[] slotStatusClasses = new String[slots.length];

for (int i = 0; i < slots.length; i++) {
    boolean isFrozen = false;
    
    // Check 1: All slots freeze at 7:45 AM
    if (testMinutes >= 7 * 60 + 45) {
        isFrozen = true;
    }
    // Check 2: Individual slot freeze
    else if (testH >= 0 && testH < 8) {
        // Night shift time (12 AM - 8 AM)
        // For AM slots, freeze time is next day, so we need to add 24 hours to test time
        int adjustedTestMinutes = testMinutes + 1440; // Add 24 hours to compare with next day freeze
        if (adjustedTestMinutes >= freezeMinutes[i]) {
            isFrozen = true;
        }
    }
    // Check 3: Read-only mode (8 AM - 4 PM)
    else if (testH >= 8 && testH < 16) {
        isFrozen = true;
    }
    
    if (isFrozen) {
        slotStatuses[i] = "frozen";
        slotStatusLabels[i] = "🔒 Frozen";
        slotStatusClasses[i] = "frozen-slot";
    } else {
        slotStatuses[i] = "editable";
        slotStatusLabels[i] = "✏️ Editable";
        slotStatusClasses[i] = "editable-slot";
    }
}

// Global editable status
boolean isEditable = false;
if (testH >= 16 || testH < 8) {
    isEditable = true;
}

// Build URL for redirect
String baseUrl = request.getContextPath() + "/jsp/NightShiftCheckList_NEW.jsp";
String redirectUrl = baseUrl + "?testMode=true&testHour=" + testH + "&testMinute=" + testM;
if (testDate != null && !testDate.isEmpty()) {
    redirectUrl += "&testDate=" + testDate;
}
