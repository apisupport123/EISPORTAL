    function countdownTimer(el, minutes, seconds) {
        var element = document.getElementById(el);
        if(!element) return;
        var endTime = (+new Date) + 1000 * (60 * minutes + seconds);
        function update() {
            var msLeft = endTime - (+new Date);
            if (msLeft < 1000) { element.innerHTML = "⟳"; }
            else {
                var t = new Date(msLeft);
                var m = t.getUTCMinutes();
                var s = t.getUTCSeconds();
                element.innerHTML = (m ? m + ':' : '') + (m ? String(s).padStart(2,'0') : s);
                setTimeout(update, 100);
            }
        }
        update();
    }
