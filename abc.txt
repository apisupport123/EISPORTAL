	private void insertSI100PercentFailure() throws SQLException {

			preparedStatement = l_objreportdbcon.prepareStatement(SQL_INSERT);

			String l_query = "SELECT function_name as API_NAME, COUNT(1) AS total_hits,\r\n"
					+ "SUM(CASE WHEN ((error_code IS NULL) OR (error_code in( '00','200'))) THEN 1 ELSE 0 END) AS success,\r\n"
					+ "SUM(CASE WHEN error_code NOT IN('00','200') AND error_code IS NOT NULL and substr(ERROR_CODE,1,2)='SI' AND   trim(error_code) not in (select  DISTINCT ERRCODE  from cbs_err_map) THEN 1 ELSE 0 END)AS TD,\r\n"
					+ "round(SUM((CASE WHEN error_code NOT IN('00','200') AND error_code IS NOT NULL and substr(ERROR_CODE,1,2)='SI' AND   trim(error_code) not in (select  DISTINCT ERRCODE  from cbs_err_map) THEN 1 ELSE 0 END) * 100) / COUNT(1), 2) AS TD_PER,\r\n"
					+ "min(req_time) as ISSUE_Time,\r\n" + "SYSDATE AS INSERT_TIME,\r\n"
					+ "LISTAGG(DISTINCT error_code,',')as ERROR_CODE\r\n" + "FROM sbisi.si_trans_db@sidb_link\r\n"
					+ "WHERE\r\n" + "trim(CONCAT(FUNCTION_NAME,error_code )) not in\r\n" + "(\r\n"
					+ "select  trim(CONCAT(api_name,error_code )) from vaild_bd_master\r\n" + ")\r\n"
					+ "and function_name not in ('POSEMICheckOffer')\r\n"
					+ "and req_time >= sysdate - INTERVAL '180' MINUTE\r\n" + "GROUP BY function_name\r\n"
					+ "HAVING round(SUM((CASE WHEN error_code NOT IN('00','200') AND error_code IS NOT NULL and substr(ERROR_CODE,1,2)='SI' AND   trim(error_code) not in (select  DISTINCT ERRCODE  from cbs_err_map) THEN 1 ELSE 0 END) * 100) / COUNT(1), 2)=100";
