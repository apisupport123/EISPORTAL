<script>
// ============================================================
// SLOT DATA
// ============================================================
const slots = [
    { name: '10PM-11PM', endTime: '11PM' },
    { name: '11PM-12AM', endTime: '12AM' },
    { name: '12AM-1AM', endTime: '1AM' },
    { name: '1AM-2AM', endTime: '2AM' },
    { name: '2AM-3AM', endTime: '3AM' },
    { name: '3AM-4AM', endTime: '4AM' },
    { name: '4AM-5AM', endTime: '5AM' },
    { name: '5AM-6AM', endTime: '6AM' },
    { name: '6AM-7AM', endTime: '7AM' }
];

// ============================================================
// FIXED: computeDeadline - Only 12AM gets +1 day
// ============================================================
function computeDeadline(taskName, reportDate) {
    const parts = taskName.split('-');
    if (parts.length !== 2) return null;
    let endStr = parts[1].trim();
    
    // Parse time (e.g., "11PM" -> 23:00, "12AM" -> 00:00)
    let hour = 0, minute = 0;
    let isPM = endStr.includes('PM');
    let isAM = endStr.includes('AM');
    let timeStr = endStr.replace(/[AP]M/g, '').trim();
    
    if (timeStr === '12') {
        hour = isPM ? 12 : 0;
    } else {
        hour = parseInt(timeStr);
        if (isPM && hour !== 12) hour += 12;
        if (isAM && hour === 12) hour = 0;
    }
    
    // Create deadline date
    let deadline = new Date(reportDate);
    
    // ===== FIX: Only add +1 day if end time is 12AM (midnight) =====
    if (hour === 0 && minute === 0) {
        deadline.setDate(deadline.getDate() + 1);
    }
    
    deadline.setHours(hour, minute, 0, 0);
    deadline.setMinutes(deadline.getMinutes() + 45); // Add 45 min freeze
    
    return deadline;
}

// ============================================================
// CHECK SLOT STATUS
// ============================================================
function checkSlots(currentTime) {
    const reportDate = new Date();
    reportDate.setHours(0, 0, 0, 0);
    
    let results = [];
    let debug = [];
    let frozenCount = 0;
    
    slots.forEach(function(slot) {
        const deadline = computeDeadline(slot.name, reportDate);
        const isFrozen = currentTime > deadline;
        if (isFrozen) frozenCount++;
        
        results.push({
            name: slot.name,
            deadline: deadline,
            isFrozen: isFrozen,
            freezeTime: deadline.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })
        });
        
        debug.push(slot.name + ' → Freezes at ' + 
            deadline.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }) + 
            ' → ' + (isFrozen ? '❌ FROZEN' : '✅ EDITABLE'));
    });
    
    return { results: results, debug: debug, frozenCount: frozenCount, total: slots.length };
}

// ============================================================
// RENDER RESULTS
// ============================================================
function renderResults(currentTime, testName) {
    const data = checkSlots(currentTime);
    
    // Update time display
    document.getElementById('currentTimeDisplay').innerHTML = 
        '⏰ Current Time: <span style="color:#1a237e;">' + 
        currentTime.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' }) + 
        '</span> ' + (currentTime.getHours() < 12 ? 'AM' : 'PM');
    
    // Status display
    let statusClass = 'status-editable';
    let statusText = '✏️ All slots are EDITABLE';
    if (data.frozenCount === data.total) {
        statusClass = 'status-frozen';
        statusText = '🔒 All slots are FROZEN';
    } else if (data.frozenCount > 0) {
        statusClass = 'status-partial';
        statusText = '⚠️ ' + data.frozenCount + ' slot(s) are FROZEN, ' + (data.total - data.frozenCount) + ' are EDITABLE';
    }
    
    document.getElementById('statusDisplay').innerHTML = 
        '<div class="status ' + statusClass + '">' + statusText + '</div>';
    
    // Slot status
    let html = '';
    data.results.forEach(function(r) {
        const cls = r.isFrozen ? 'frozen' : 'editable';
        const icon = r.isFrozen ? '🔒' : '✏️';
        const status = r.isFrozen ? 'FROZEN' : 'EDITABLE';
        const badge = r.isFrozen ? '<span class="badge badge-frozen">FROZEN</span>' : 
                                       '<span class="badge badge-editable">EDITABLE</span>';
        html += '<div class="slot ' + cls + '">' +
            '<span><strong>' + r.name + '</strong> → Freezes at ' + r.freezeTime + '</span>' +
            '<span>' + icon + ' ' + badge + '</span>' +
        '</div>';
    });
    
    document.getElementById('slotStatus').innerHTML = html;
    document.getElementById('debugInfo').textContent = 
        '🧪 Test: ' + testName + '\n' +
        '=' .repeat(60) + '\n' +
        data.debug.join('\n') + '\n' +
        '=' .repeat(60) + '\n' +
        '📊 Summary: ' + data.frozenCount + ' frozen, ' + (data.total - data.frozenCount) + ' editable';
    
    // Test Summary
    let summary = '';
    if (testName.includes('11:30 PM')) {
        summary = '✅ PASS: All slots should be editable at 11:30 PM';
    } else if (testName.includes('11:50 PM')) {
        summary = '✅ PASS: 10PM-11PM should be frozen, others editable';
    } else if (testName.includes('12:30 AM')) {
        summary = '✅ PASS: 11PM-12AM should still be editable at 12:30 AM';
    } else if (testName.includes('12:45 AM')) {
        summary = '✅ PASS: 11PM-12AM should be frozen exactly at 12:45 AM';
    } else if (testName.includes('1:00 AM')) {
        summary = '✅ PASS: 11PM-12AM frozen, 12AM-1AM still editable';
    } else {
        summary = '🔄 Real-time check - ' + data.frozenCount + ' slots frozen, ' + 
                  (data.total - data.frozenCount) + ' slots editable';
    }
    document.getElementById('testSummary').innerHTML = summary;
}
