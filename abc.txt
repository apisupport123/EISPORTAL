select  s.source_id,
case when EXISTS ( select  /*+ parallel(04)*/ 1 from eisapp.api_log PARTITION(AUG_26) a where a.source_id=s.source_id
and a.request_date_time>=sysdate - interval '10' minute) then 'Y' else 'N' ENd As flag
from
(select 'SA'  as source_id from dual
union all
select 'FS'    from dual
union all
select 'I4'   from dual
union all
select 'Y2'   from dual
union all
select 'KA'  from dual) s;

Q3.

select 'SA'  as source_id from dual
union all
select 'FS'    from dual
union all
select 'I4'   from dual
union all
select 'Y2'   from dual
union all

case when EXISTS ( select   1 from eisapp.api_log PARTITION(AUG_26) a where a.source_id=s.source_id
